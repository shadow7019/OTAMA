    42→export async function cached<T>(key: string, ttlMs: number, fn: () => Promise<T>): Promise<T> {
    43→  const hit = cache.get(key)
    44→  if (hit && Date.now() - hit.ts < ttlMs) return hit.value as T
    45→  const value = await fn()
    46→  cache.set(key, { ts: Date.now(), value })
    47→  return value
    48→}

    52→async function fetchText(url: string, timeoutMs = 12_000): Promise<string> {
    53→  const ctrl = new AbortController()
    54→  const t = setTimeout(() => ctrl.abort(), timeoutMs)
    55→  try {
    56→    const res = await fetch(url, {
    57→      signal: ctrl.signal,
    58→      cache: 'no-store',
    59→      headers: { 'User-Agent': UA, Accept: '*/*' },
    60→    })
    61→    if (!res.ok) throw new Error(`HTTP ${res.status} for ${url}`)
    62→    return await res.text()
    63→  } finally {
    64→    clearTimeout(t)
    65→  }
    66→}
    67→
    68→async function fetchJson<T>(url: string, timeoutMs = 12_000): Promise<T> {
    69→  const text = await fetchText(url, timeoutMs)
    70→  if (text.trimStart().startsWith('<')) throw new Error(`Got HTML instead of JSON from ${url}`)
    71→  return JSON.parse(text) as T
    72→}

    76→/**
    77→ * Some trackers (apibay = ThePirateBay API, EZTV) sit behind Cloudflare and
    78→ * fingerprint TLS: Node's fetch/https stacks get 403-challenged while curl
    79→ * passes. Use curl as the primary transport for those hosts with a fetch
    80→ * fallback for environments without curl.
    81→ */
    82→export async function cfGetText(url: string, timeoutMs = 12_000): Promise<string> {
    83→  const maxTime = Math.ceil(timeoutMs / 1000)
    84→  try {
    85→    const { execFile } = await import('node:child_process')
    86→    const out = await new Promise<string>((resolve, reject) => {
    87→      execFile(
    88→        'curl',
    89→        ['-s', '-L', '--compressed', '--max-time', String(maxTime), '-A', UA, '-H', 'Accept: */*', url],
    90→        { timeout: (maxTime + 2) * 1000, maxBuffer: 8 * 1024 * 1024 },
    91→        (err, stdout) => (err ? reject(err) : resolve(stdout)),
    92→      )
    93→    })
    94→    if (out) return out
    95→  } catch { /* curl missing or failed -> fall back to fetch */ }
    96→  return fetchText(url, timeoutMs)
    97→}
    98→
    99→export async function cfGetJson<T>(url: string, timeoutMs = 12_000): Promise<T> {
   100→  const text = await cfGetText(url, timeoutMs)
   101→  if (text.trimStart().startsWith('<')) throw new ProviderError(`Blocked by Cloudflare: ${new URL(url).host}`)
   102→  return JSON.parse(text) as T
   103→}

   159→  const data = await fetchJson<{ metas?: CineMetaRaw[] }>(url, 15_000)
   160→  return (data.metas || []).map((m) => normalizeCine(m, kind))

   198→export function humanSize(bytes?: number | string): string {
   199→  const n = typeof bytes === 'string' ? parseInt(bytes, 10) : bytes
   200→  if (!n || isNaN(n)) return ''
   201→  const units = ['B', 'KB', 'MB', 'GB', 'TB']
   202→  let i = 0
   203→  let v = n
   204→  while (v >= 1024 && i < units.length - 1) {
   205→    v /= 1024
   206→    i++
   207→  }
   208→  return `${v.toFixed(v >= 100 || i === 0 ? 0 : 1)} ${units[i]}`
   209→}

   211→const QUALITY_RE = /\b(2160p|1440p|1080p|1080i|720p|576p|480p|4k|uhd|web[ .-]?dl|web[ .-]?rip|brrip|bluray|bdrip|hdrip|dvdrip|hdtv|cam|ts)\b/i

   213→export function detectQuality(name: string): string | undefined {
   214→  const m = QUALITY_RE.exec(name)
   215→  if (!m) return undefined
   216→  const q = m[1].toUpperCase()
   217→  if (q === '4K' || q === 'UHD') return '2160p'
   218→  return q.replace(/\s/g, '')
   219→}

   221→/**
   222→ * Codec detection from a release name. HEVC/x265 releases typically CANNOT be
   223→ * decoded by browsers (Chrome only plays H.265 with hardware support), which
   224→ * is the #1 cause of the "stuck on buffering" report — so OTAMA surfaces and
   225→ * deprioritises them.
   226→ */
   227→export function detectCodecFromName(name: string): 'h264' | 'hevc' | undefined {
   228→  const s = (name || '').toLowerCase()
   229→  if (/\b(x\s?265|h\.?265|hevc)\b/.test(s)) return 'hevc'
   230→  if (/\b(x\s?264|h\.?264|avc)\b/.test(s)) return 'h264'
   231→  return undefined
   232→}
   233→
   234→/**
   235→ * Browser playability of a release name — 0 plays everywhere, higher ranks
   236→ * may not play: 1 MKV/MOV (Chromium-only), 2 AVI/TS (<video> can't demux),
   237→ * 3 HEVC/x265 (browsers cannot decode without hardware support).
   238→ */
   239→export function playabilityRank(name: string, codec?: string): 0 | 1 | 2 | 3 {
   240→  const c = (codec as 'h264' | 'hevc' | undefined) || detectCodecFromName(name)
   241→  if (c === 'hevc') return 3
   242→  const ext = (name.match(/\.(mp4|m4v|mkv|mov|avi|ts|webm)\b/i) || [])[1]?.toLowerCase()
   243→  if (ext === 'avi' || ext === 'ts') return 2 // container not supported by <video>
   244→  if (ext === 'mkv' || ext === 'mov') return 1 // plays in Chromium/Chrome, not Firefox/Safari
   245→  return 0
   246→}
   247→
   248→/**
   249→ * Sort torrent options so that browser-playable releases come first, then by
   250→ * seed count. Keeps the popular x265-HEVC packs from hijacking the top slot.
   251→ */
   252→export function sortTorrentsPlayableFirst(list: TorrentOption[]): TorrentOption[] {
   253→  return [...list].sort((a, b) => {
   254→    const pa = playabilityRank(a.title, a.codec)
   255→    const pb = playabilityRank(b.title, b.codec)
   256→    if (pa !== pb) return pa - pb
   257→    return (b.seeds || 0) - (a.seeds || 0)
   258→  })
   259→}

   261→export function normalizeApibayRow(row: ApibayRow): TpbItem | null {
   262→  if (!row.info_hash || !row.name) return null
   263→  const cat = row.category || '0'
   264→  return {
   265→    id: row.id || row.info_hash,
   266→    name: row.name,
   267→    hash: row.info_hash.toLowerCase(),
   268→    size: humanSize(row.size),
   269→    sizeBytes: parseInt(row.size || '0', 10) || 0,
   270→    seeds: parseInt(row.seeders || '0', 10) || 0,
   271→    leechers: parseInt(row.leechers || '0', 10) || 0,
   272→    category: TPB_CATEGORIES[cat] || 'Other',

   274→    added: row.added ? new Date(parseInt(row.added, 10) * 1000).toISOString() : '',
   275→    username: row.username || 'anonymous',
   276→    status: row.status,
   277→    imdb: row.imdb || undefined,
   278→    quality: detectQuality(row.name),
   279→  }

   306→/**
   307→ * Race a provider against a soft deadline — best-effort sources (TGx mirror
   308→ * walks can take a minute when every mirror is CF-blocked) must not stall the
   309→ * whole detail panel that is waiting on Promise.all.
   310→ */
   311→function withDeadline<T>(p: Promise<T>, ms: number, fallback: T): Promise<T> {
   312→  return Promise.race([
   313→    p.catch(() => fallback),
   314→    new Promise<T>((resolve) => setTimeout(() => resolve(fallback), ms)),
   315→  ])
   316→}

   318→/**
   319→ * Find movie torrents across ALL video sources:
   320→ * Torrentio (multi-site, file-exact) + TPB + YTS + 1337x + SolidTorrents
   321→ * + RARBG archive + LimeTorrents + TorrentDownloads + TorrentGalaxy.
   322→ */
   323→export async function findMovieTorrents(imdbId?: string, title?: string, year?: number): Promise<TorrentOption[]> {
   324→  return cached(`movtorrent:${imdbId || '-'}|${title || '-'}|${year || '-'}`, 10 * 60_000, async () => {
   325→    const { ytsMovieTorrents } = await import('./yts')
   326→    const { leetxSearch } = await import('./leetx')
   327→    const { torrentioStreams } = await import('./torrentio')
   328→    const { solidSearch } = await import('./solidtorrents')
   329→    const { rarbgSearch } = await import('./rarbg')
   330→    const { limeSearch } = await import('./limetorrents')
   331→    const { torrentDownloadsSearch } = await import('./torrentdownloads')
   332→    const { tgxSearch } = await import('./torrentgalaxy')
   333→
   334→    const tpbTask = (async () => {
   335→      let rows: TpbItem[] = []
   336→      if (imdbId) {
   337→        try {
   338→          rows = await apibaySearch(imdbId)
   339→        } catch { /* fall through */ }
   340→      }
   341→      if (rows.length === 0 && title) {
   342→        const q = year ? `${title} ${year}` : title
   343→        try {
   344→          rows = await apibaySearch(q)
   345→        } catch { /* fall through */ }
   346→      }
   347→      const videoCats = new Set(['201', '202', '207', '208', '209', '299'])
   348→      return rows
   349→        .filter((r) => videoCats.has(r.categoryCode) || r.imdb)
   350→        .map(apibayToTorrentOption)
   351→    })()
   352→
   353→    const q = title ? `${title}${year ? ` ${year}` : ''}` : ''
   354→    const [torrentio, tpb, yts, leetx, solid, rarbg, lime, td, tgx] = await Promise.all([
   355→      imdbId
   356→        ? torrentioStreams('movie', imdbId).catch(() => [] as TorrentOption[])
   357→        : Promise.resolve([] as TorrentOption[]),
   358→      tpbTask,
   359→      ytsMovieTorrents(imdbId, title, year).catch(() => [] as TorrentOption[]),
   360→      q
   361→        ? leetxSearch(q, { category: 'movies', resolve: 8 }).catch(() => [] as TorrentOption[])
   362→        : Promise.resolve([] as TorrentOption[]),
   363→      q
   364→        ? solidSearch(q, { videoOnly: true }).catch(() => [] as TorrentOption[])
   365→        : Promise.resolve([] as TorrentOption[]),
   366→      q
   367→        ? rarbgSearch(q).catch(() => [] as TorrentOption[])
   368→        : Promise.resolve([] as TorrentOption[]),
   369→      q
   370→        ? limeSearch(q, { category: 'movies' }).catch(() => [] as TorrentOption[])
   371→        : Promise.resolve([] as TorrentOption[]),
   372→      q
   373→        ? torrentDownloadsSearch(q, { resolve: 8 }).catch(() => [] as TorrentOption[])
   374→        : Promise.resolve([] as TorrentOption[]),
   375→      q
   376→        ? withDeadline(tgxSearch(q, { resolve: 6 }).catch(() => [] as TorrentOption[]), 16_000, [] as TorrentOption[])
   377→        : Promise.resolve([] as TorrentOption[]),
   378→    ])
   379→
   380→    const seen = new Set<string>()
   381→    const merged: TorrentOption[] = []
   382→    for (const t of [...torrentio, ...yts, ...tpb, ...rarbg, ...lime, ...leetx, ...solid, ...td, ...tgx]) {
   383→      const key = (t.source || t.hash || '').toLowerCase()
   384→      if (!key || seen.has(key)) continue
   385→      seen.add(key)
   386→      merged.push(t)
   387→    }
   388→    // Browser-playable (H.264/x264, MP4/MKV) releases first, seeds second.
   389→    return sortTorrentsPlayableFirst(merged).slice(0, 40)
   390→  })
   391→}

   399→/**
   400→ * Does a poster URL actually resolve to an image? Cinemeta's metahub serves a
   401→ * small HTML error page (content-type text/html) for titles without artwork,
   402→ * which used to render as an ugly grey "initials" placeholder card.
   403→ */
   404→export async function posterWorks(url: string): Promise<boolean> {
   405→  if (!url) return false
   406→  if (!/^https?:\/\//.test(url)) return true
   407→  const hit = posterCache.get(url)
   408→  if (hit !== undefined) return hit
   409→  let ok = true
   410→  try {
   411→    const res = await fetch(url, {
   412→      method: 'HEAD',
   413→      signal: AbortSignal.timeout(6_000),
   414→      headers: { 'User-Agent': UA },
   415→    })
   416→    const ct = (res.headers.get('content-type') || '').toLowerCase()
   417→    if (!res.ok || !ct.startsWith('image/')) ok = false
   418→  } catch {
   419→    // network hiccup — give the poster the benefit of the doubt
   420→    ok = true
   421→  }
   422→  if (posterCache.size < 5000) posterCache.set(url, ok)
   423→  return ok
   424→}

   426→/** Drop broken poster URLs from a list (poster falls back to branded card). */
   427→export async function stripBrokenPosters<T extends { poster?: string }>(items: T[]): Promise<T[]> {
   428→  return Promise.all(
   429→    items.map(async (it) => {
   430→      if (!it.poster) return it
   431→      const ok = await posterWorks(it.poster)
   432→      return ok ? it : { ...it, poster: undefined }
   433→    }),
   434→  )
   435→}

   485→    .replace(/&nbsp;/g, ' ')

   496→  hash: string
   497→  seeds: number
   498→  leechers: number
   499→  size: string
   500→  date: string
   501→  page: string
   502→  info: string
   503→}
   504→
   505→export function parseNyaaRss(xml: string): NyaaItem[] {
   506→  const items: NyaaItem[] = []
   507→  const itemRe = /<item>([\s\S]*?)<\/item>/g
   508→  let m: RegExpExecArray | null
   509→  const tag = (block: string, name: string): string => {
   510→    const re = new RegExp(`<${name}[^>]*>([\\s\\S]*?)</${name}>`, 'i')
   511→    const t = re.exec(block)
   512→    return t ? decodeEntities(cdata(t[1]).trim()) : ''
   513→  }
   514→  while ((m = itemRe.exec(xml))) {
   515→    const block = m[1]
   516→    const hash = tag(block, 'nyaa:infoHash').toLowerCase()
   517→    const title = tag(block, 'title')
   518→    if (!hash || !title) continue
   519→    items.push({
   520→      title,
   521→      hash,
   522→      seeds: parseInt(tag(block, 'nyaa:seeders'), 10) || 0,
   523→      leechers: parseInt(tag(block, 'nyaa:leechers'), 10) || 0,
   524→      size: tag(block, 'nyaa:size'),
   525→      date: tag(block, 'pubDate'),
   526→      page: tag(block, 'guid'),
   527→      info: tag(block, 'description'),
   528→    })
   529→  }
   530→  return items
   531→}

   533→export async function nyaaSearch(
   534→  q: string,
   535→  opts: { sort?: 'seeders' | 'date' } = {},
   536→): Promise<TorrentOption[]> {
   537→  const sort = opts.sort === 'date' ? 's=id&o=desc' : 's=seeders&o=desc'
   538→  const url = `${NYAA}/?page=rss&q=${encodeURIComponent(q)}&c=1_2&f=0&${sort}`
   539→  const xml = await fetchText(url, 12_000)
   540→  return parseNyaaRss(xml).map((r) => ({
   541→    hash: r.hash,
   542→    title: r.title,
   543→    quality: detectQuality(r.title),
   544→    size: r.size,
   545→    seeds: r.seeds,
   546→    leechers: r.leechers,
   547→    provider: 'nyaa' as const,
   548→    source: r.hash,
   549→    date: r.date ? new Date(r.date).toISOString() : undefined,
   550→  }))
   551→}

   612→export async function seriesDetail(imdbId: string): Promise<{ item: MetaItem & { cast?: string[]; awards?: string }; seasons: { season: number; episodes: EpisodeInfo[] }[] }> {
   613→  const cine = await cineMeta('series', imdbId)
   614→  let item: MetaItem & { cast?: string[]; awards?: string } = cine
   615→  const seasonsMap = new Map<number, EpisodeInfo[]>()
   616→
   617→  // Try TVMaze for reliable episode lists
   618→  try {
   619→    const show = await tvmazeLookupByImdb(imdbId)
   620→    if (show) {
   621→      const shows = await fetchJson<TvMazeShow & { _embedded?: { episodes?: TvMazeEpisode[] } }>(
   622→        `${TVMAZE}/shows/${show.id}?embed=episodes`,
   623→        15_000,
   624→      )
   625→      item = { ...normalizeTvMaze(shows), backdrop: cine.backdrop, summary: cine.summary || normalizeTvMaze(shows).summary, cast: cine.cast, awards: cine.awards }
   626→      for (const ep of shows._embedded?.episodes || []) {
   627→        if (ep.number == null) continue
   628→        const list = seasonsMap.get(ep.season) || []
   629→        list.push({
   630→          season: ep.season,
   631→          episode: ep.number,
   632→          title: ep.name,
   633→          overview: ep.summary ? ep.summary.replace(/<[^>]+>/g, '') : undefined,
   634→          airDate: ep.airdate || undefined,
   635→          thumbnail: ep.image?.medium || undefined,
   636→        })
   637→        seasonsMap.set(ep.season, list)
   638→      }
   639→    }
   640→  } catch {
   641→    // fallback: Cinemeta videos
   642→  }
   643→
   644→  if (seasonsMap.size === 0 && cine.videos?.length) {
   645→    for (const v of cine.videos) {
   646→      if (v.season == null || v.episode == null || v.season < 1) continue
   647→      const list = seasonsMap.get(v.season) || []
   648→      list.push({
   649→        season: v.season,
   650→        episode: v.episode,
   651→        title: v.name || v.title,
   652→        overview: v.overview,
   653→        airDate: v.released,
   654→        thumbnail: v.thumbnail,
   655→      })
   656→      seasonsMap.set(v.season, list)
   657→    }
   658→  }
   659→
   660→  const seasons = [...seasonsMap.entries()]
   661→    .map(([season, episodes]) => ({ season, episodes: episodes.sort((a, b) => a.episode - b.episode) }))
   662→    .sort((a, b) => a.season - b.season)
   663→
   664→  return { item, seasons }
   665→}

   667→/** Episode torrents: Torrentio (multi-site, file-exact) + EZTV + TPB + 1337x + SolidTorrents. */
   668→export async function findEpisodeTorrents(
   669→  imdbId?: string,
   670→  title?: string,
   671→  season?: number,
   672→  episode?: number,
   673→): Promise<TorrentOption[]> {
   674→  // Torrentio first — aggregates EZTV/TPB/1337x/TGX/RARBG/KAT and returns the
   675→  // exact video file index for the episode (fast + fewest dead torrents).
   676→  if (imdbId && /^tt\d+$/.test(imdbId)) {
   677→    try {
   678→      const { torrentioStreams } = await import('./torrentio')
   679→      const tor = await cached(
   680→        `torrentio:ep:${imdbId}:${season ?? ''}:${episode ?? ''}`,
   681→        5 * 60_000,
   682→        () => torrentioStreams('series', imdbId, season, episode),
   683→      )
   684→      if (season && episode) {
   685→        const exact = sortTorrentsPlayableFirst(tor.filter((t) => t.season == null || t.season === season)).slice(0, 15)
   686→        if (exact.length) return exact
   687→      } else {
   688→        const sorted = sortTorrentsPlayableFirst(tor).slice(0, 24)
   689→        if (sorted.length) return sorted
   690→      }
   691→    } catch { /* Torrentio unavailable -> EZTV */ }
   692→  }
   693→
   694→  // whole-show EZTV lookup (cached)
   695→  if (imdbId && /^\d+$/.test(imdbId.replace(/^tt/, ''))) {
   696→    try {
   697→      const list = await cached(`eztv:${imdbId}`, 10 * 60_000, () => eztvByImdb(imdbId.replace(/^tt/, '')))
   698→      const sorted = sortTorrentsPlayableFirst(list)
   699→      if (season && episode) {
   700→        const exact = sorted.filter((t) => t.season === season && t.episode === episode)
   701→        if (exact.length) return exact.slice(0, 12)
   702→      } else if (season) {
   703→        const pack = sorted.filter((t) => t.season === season && (t.episode == null || t.episode === 0))
   704→        if (pack.length) return pack.slice(0, 12)
   705→      } else {
   706→        return sorted.slice(0, 24)
   707→      }
   708→    } catch { /* EZTV unavailable -> TPB */ }
   709→  }
   710→  if (!title) return []
   711→  const pad = (n?: number, w = 2) => (n != null ? String(n).padStart(w, '0') : '')
   712→  const queries: string[] = []
   713→  if (season && episode) {
   714→    queries.push(`${title} S${pad(season)}E${pad(episode)}`)
   715→    queries.push(`${title} ${season}x${pad(episode)}`)
   716→  } else if (season) {
   717→    queries.push(`${title} S${pad(season)}`)
   718→  } else {
   719→    queries.push(title)
   720→  }
   721→  for (const q of queries) {
   722→    try {
   723→      const rows = await apibaySearch(q)
   724→      const tvCats = new Set(['205', '208'])
   725→      const opts = rows
   726→        .filter((r) => tvCats.has(r.categoryCode) || /\bS\d{1,2}E\d{1,2}\b/i.test(r.name))
   727→        .map(apibayToTorrentOption)
   728→      const sorted = sortTorrentsPlayableFirst(opts).slice(0, 12)
   729→      if (sorted.length) return sorted
   730→    } catch { /* try next */ }
   731→  }
   732→  // 1337x fallback (works when TPB/EZTV are blocked)
   733→  if (season && episode) {
   734→    try {
   735→      const { leetxSearch } = await import('./leetx')
   736→      const opts = await leetxSearch(`${title} S${pad(season)}E${pad(episode)}`, { category: 'tv', resolve: 8 })
   737→      if (opts.length) return opts.slice(0, 12)
   738→    } catch { /* next fallback */ }
   739→  }
   740→  // SolidTorrents fallback (DHT index, reachable when trackers are blocked)
   741→  if (season && episode) {
   742→    try {
   743→      const { solidSearch } = await import('./solidtorrents')
   744→      const opts = await solidSearch(`${title} S${pad(season)}E${pad(episode)}`, { videoOnly: true })
   745→      const sorted = sortTorrentsPlayableFirst(opts).slice(0, 12)
   746→      if (sorted.length) return sorted
   747→    } catch { /* next fallback */ }
   748→  }
   749→  // RARBG archive fallback (JSON, infohash included)
   750→  try {
   751→    const { rarbgSearch } = await import('./rarbg')
   752→    const opts = sortTorrentsPlayableFirst(await rarbgSearch(`${title} ${season && episode ? `S${pad(season)}E${pad(episode)}` : ''}`.trim()))
   753→      .filter((t) => /\bS\d{1,2}E\d{1,2}\b/i.test(t.title) || !season)
   754→    if (opts.length) return opts.slice(0, 12)
   755→  } catch { /* next fallback */ }
   756→  // LimeTorrents TV fallback (hash embedded in rows)
   757→  if (season && episode) {
   758→    try {
   759→      const { limeSearch } = await import('./limetorrents')
   760→      const opts = await limeSearch(`${title} S${pad(season)}E${pad(episode)}`, { category: 'tv' })
   761→      if (opts.length) return opts.slice(0, 12)
   762→    } catch { /* next fallback */ }
   763→  }
   764→  // TorrentGalaxy fallback (best-effort, mirror chain) — soft deadline so a
   765→  // fully-blocked mirror chain can never stall the episode panel
   766→  if (season && episode) {
   767→    try {
   768→      const { tgxSearch } = await import('./torrentgalaxy')
   769→      const opts = await withDeadline(tgxSearch(`${title} S${pad(season)}E${pad(episode)}`, { resolve: 6 }), 16_000, [] as TorrentOption[])
   770→      if (opts.length) return opts.slice(0, 12)
   771→    } catch { /* give up */ }
   772→  }
   773→  return []
   774→}
