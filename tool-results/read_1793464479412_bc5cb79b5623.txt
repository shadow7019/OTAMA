    43→/**
    44→ * A torrent "server" = one ranked source returned by /api/torrent/search
    45→ * (ThePirateBay, Torrentio/1337x, YTS, EZTV …). We download it on our own
    46→ * backend and stream it to the native <video> element — no third-party
    47→ * iframe, therefore no pop-ups, no tab redirects, no injected ads.
    48→ */
    49→interface TorrentServer {
    50→  id: string;
    51→  releaseName: string;
    52→  seeders: number;
    53→  peers: number;
    54→  sizeBytes: number;
    55→  provider: string;
    56→  quality: string;
    57→  fileIdx: number | null;
    58→  /** verified = exact title/episode match · pack = season pack (right file auto-picked) · weak = title-only match */
    59→  match: "verified" | "pack" | "weak";
    60→  /** audio/language tags parsed from the release name (sub / eng / hindi / dual…) */
    61→  audioTags: string[];
    62→  score: number;
    63→}
    64→
    65→interface SearchHints {
    66→  absEpisode: number | null;
    67→  relEpisode: number | null;
    68→}
    69→
    70→type HealthStatus = "checking" | "live" | "dead";
    71→

   116→/** Switch servers when the swarm gives us nothing at all. */
   117→const NO_PEER_SWITCH_MS = 18000;
   118→/** Switch when peers exist but data never arrives. */
   119→const STALL_SWITCH_MS = 40000;
   120→
   121→const fmtSize = (b: number) =>
   122→  b >= 1e9 ? `${(b / 1e9).toFixed(1)} GB` : b >= 1e6 ? `${Math.round(b / 1e6)} MB` : b > 0 ? `${Math.round(b / 1e3)} KB` : "—";
   123→const fmtSpeed = (bps: number) =>
   124→  bps >= 1e6 ? `${(bps / 1e6).toFixed(1)} MB/s` : `${Math.round(bps / 1e3)} KB/s`;
   125→const fmtSeeders = (n: number) => (n >= 1000 ? `${(n / 1000).toFixed(1)}k` : String(n));

   127→export function PlayerModal({ target, actions, onClose }: PlayerModalProps) {
   128→  // remounted per target via key — initial state is safe to derive from props
   129→  const type = target ? getMediaType(target.item) : "movie";
   130→  const itemId = target?.item.id ?? null;
   131→
   132→  /* ---------------- servers (torrent sources) ---------------- */
   133→  const [servers, setServers] = useState<TorrentServer[]>([]);
   134→  const [serversState, setServersState] = useState<"loading" | "ready" | "error">("loading");
   135→  const [searchNonce, setSearchNonce] = useState(0); // bump = manual re-search
   136→  const [activeIdx, setActiveIdx] = useState<number | null>(null);
   137→  const [manualIdx, setManualIdx] = useState<number | "archive" | "trailer" | null>(null);
   138→  const [tried, setTried] = useState<number[]>([]);
   139→  const [exhausted, setExhausted] = useState(false);
   140→  const [videoState, setVideoState] = useState<"loading" | "playing" | "error">("loading");
   141→  const [needsTap, setNeedsTap] = useState(false);
   142→  const [status, setStatus] = useState<TorrentStatus | null>(null);
   143→  const [videoNonce, setVideoNonce] = useState(0);
   144→  const [banner, setBanner] = useState<string | null>(null);
   145→  /** automatic re-searches done for the CURRENT episode (reset on ep change) */
   146→  const [autoRetries, setAutoRetries] = useState(0);
   147→  /** anime numbering hints from the search — steer file picking in batches */
   148→  const [hints, setHints] = useState<SearchHints | null>(null);
   149→  /** per-server health from the auto-checker (infoHash → status) */
   150→  const [health, setHealth] = useState<Record<string, HealthStatus>>({});
   151→  const [healthBusy, setHealthBusy] = useState(false);
   152→  /** bump → (re)run the health auto-checker */
   153→  const [healthNonce, setHealthNonce] = useState(0);
   154→
   155→  /* ---------------- archive.org direct fallback (movies) ---------------- */
   156→  const [archive, setArchive] = useState<ArchiveMatchClient | null>(null);
   157→  const [archiveState, setArchiveState] = useState<"loading" | "found" | "none">("loading");
   158→
   159→  /* ---------------- TV navigation ---------------- */
   160→  const [season, setSeason] = useState(() => target?.season ?? 1);
   161→  const [episode, setEpisode] = useState(() => target?.episode ?? 1);
   162→  const [episodes, setEpisodes] = useState<Episode[]>([]);
   163→  const [epLoading, setEpLoading] = useState(() => type === "tv");
   164→  const [details, setDetails] = useState<MediaDetails | null>(null);
   165→  const [showEpisodes, setShowEpisodes] = useState(false);
   166→
   167→  const mounted = useMounted();
   168→  const addContinue = useMediaStore((s) => s.addContinue);
   169→  const toggleWatchlist = useMediaStore((s) => s.toggleWatchlist);
   170→  const watchlist = useMediaStore((s) => s.watchlist);
   171→  const serverLang = useMediaStore((s) => s.serverLang);
   172→  const setServerLang = useMediaStore((s) => s.setServerLang);
   173→  const deadServers = useMediaStore((s) => s.deadServers);
   174→  const markDeadServers = useMediaStore((s) => s.markDeadServers);
   175→  const reviveServer = useMediaStore((s) => s.reviveServer);
   176→
   177→  // refs so callbacks/effects always read the latest values
   178→  const serversRef = useRef(servers);
   179→  serversRef.current = servers;
   180→  const activeIdxRef = useRef(activeIdx);
   181→  activeIdxRef.current = activeIdx;
   182→  const triedRef = useRef(tried);
   183→  triedRef.current = tried;
   184→  const videoStateRef = useRef(videoState);
   185→  videoStateRef.current = videoState;
   186→  const streamStartRef = useRef(0);
   187→  const currentIdRef = useRef<string | null>(null);
   188→  const videoRef = useRef<HTMLVideoElement | null>(null);
   189→
   190→  // When the browser blocks autoplay we surface a tap-to-play overlay instead
   191→  // of a stuck black frame (video data is already buffered at that point).
   192→  const handleCanPlay = (e: React.SyntheticEvent<HTMLVideoElement>) => {
   193→    setVideoState((s) => (s === "loading" ? "playing" : s));
   194→    const v = e.currentTarget;
   195→    if (v.paused) {
   196→      v.play().then(() => setNeedsTap(false)).catch(() => setNeedsTap(true));
   197→    }
   198→  };
   199→
   200→  const activeServer = activeIdx !== null ? servers[activeIdx] ?? null : null;
   201→  const isArchive = manualIdx === "archive" && archiveState === "found" && !!archive;
   202→  const showTrailer = manualIdx === "trailer";
   203→
   204→  const trailer: Video | null = useMemo(() => (details ? trailerOf(details) : null), [details]);
   205→
   206→  const stopId = useCallback((id: string | null) => {
   207→    if (!id) return;
   208→    fetch("/api/torrent/stop", {
   209→      method: "POST",
   210→      headers: { "Content-Type": "application/json" },
   211→      body: JSON.stringify({ id }),
   212→      keepalive: true,
   213→    }).catch(() => {});
   214→  }, []);
   215→
   216→  const serverLabel = useCallback((s: TorrentServer) => `${s.quality} · ${s.provider}`, []);
   217→
   218→  /* ---------------- server ordering: language filter, dead last ---------------- */
   219→  const orderedServers = useMemo(() => {
   220→    const all = servers.map((_, i) => i);
   221→    const filtered = all.filter((i) => {
   222→      const tags = servers[i].audioTags ?? [];
   223→      return serverLang === "auto" || tags.includes(serverLang);
   224→    });
   225→    // a stale language preference (from another title) must never empty the list
   226→    const base = filtered.length ? filtered : all;
   227→    const alive = base.filter((i) => health[servers[i].id] !== "dead" && !isServerDead(deadServers, servers[i].id));
   228→    const dead = base.filter((i) => !alive.includes(i));
   229→    return [...alive, ...dead];
   230→  }, [servers, serverLang, health, deadServers]);
   231→  const orderedRef = useRef(orderedServers);
   232→  orderedRef.current = orderedServers;
   233→
   234→  /** language tags present among the current servers, in stable display order */
   235→  const langChips = useMemo(() => {
   236→    const order = [
   237→      "sub", "eng", "hindi", "tamil", "telugu", "malayalam", "kannada",
   238→      "bengali", "marathi", "punjabi", "dual", "multi", "jap", "raw", "spa", "por", "chi",
   239→    ];
   240→    const present = new Set<string>();
   241→    for (const s of servers) for (const t of s.audioTags ?? []) present.add(t);
   242→    return order.filter((t) => present.has(t));
   243→  }, [servers]);
   244→
   245→  /* ---------------- failover ---------------- */
   246→  const failover = useCallback(
   247→    (reason: string) => {
   248→      const cur = activeIdxRef.current;
   249→      const list = serversRef.current;
   250→      const pool = orderedRef.current.filter((i) => i !== cur && !triedRef.current.includes(i));
   251→      // healthy servers first; a health-marked dead one is a last resort
   252→      const next = pool.find((i) => health[serversRef.current[i]?.id] !== "dead") ?? pool[0];
   253→      const triedNext = new Set(triedRef.current);
   254→      if (cur !== null) triedNext.add(cur);
   255→      triedRef.current = [...triedNext];
   256→      setTried([...triedNext]);
   257→      const failed = cur !== null ? serversRef.current[cur] : null;
   258→      if (failed) stopId(failed.id);
   259→      if (next !== undefined && next >= 0) {
   260→        setActiveIdx(next);
   261→        setVideoState("loading");
   262→        setStatus(null);
   263→        streamStartRef.current = Date.now();
   264→        setBanner(`${reason} — switching to server ${next + 1} (${serverLabel(list[next])})`);
   265→      } else {
   266→        setExhausted(true);
   267→        setBanner(reason);
   268→      }
   269→    },
   270→    [stopId, serverLabel, health]
   271→  );
   272→
   273→  /* ---------------- load servers for this title/episode ---------------- */
   274→  useEffect(() => {
   275→    if (!target) return;
   276→    let cancelled = false;
   277→    setServers([]);
   278→    setServersState("loading");
   279→    setTried([]);
   280→    triedRef.current = [];
   281→    setActiveIdx(null);
   282→    setManualIdx(null);
   283→    setExhausted(false);
   284→    setVideoState("loading");
   285→    setStatus(null);
   286→    setBanner(null);
   287→
   288→    const params = new URLSearchParams({
   289→      type,
   290→      id: String(target.item.id),
   291→      title: getTitle(target.item),
   292→      year: getYear(target.item),
   293→    });
   294→    if (type === "tv") {
   295→      params.set("season", String(season));
   296→      params.set("episode", String(episode));
   297→    }
   298→    fetch(`/api/torrent/search?${params.toString()}`)
   299→      .then((r) => r.json())
   300→      .then((j: { results?: TorrentServer[]; hints?: SearchHints }) => {
   301→        if (cancelled) return;
   302→        setServers(Array.isArray(j.results) ? j.results : []);
   303→        setHints(j.hints ?? null);
   304→        setHealth({});
   305→        setHealthNonce((n) => n + 1); // kick off the health auto-checker
   306→        setServersState("ready");
   307→      })
   308→      .catch(() => {
   309→        if (!cancelled) setServersState("error");
   310→      });
   311→
   312→    // Otama Direct: movies → public-domain copy; TV episodes → episode lookup
   313→    // (Indian daily shows surface on archive.org hours after broadcast, long
   314→    // before the public trackers list them)
   315→    setArchiveState("loading");
   316→    const archiveUrl =
   317→      type === "movie"
   318→        ? `/api/archive?title=${encodeURIComponent(getTitle(target.item))}&year=${encodeURIComponent(getYear(target.item))}`
   319→        : `/api/archive?type=tv&id=${target.item.id}&season=${season}&episode=${episode}&title=${encodeURIComponent(getTitle(target.item))}`;
   320→    fetch(archiveUrl)
   321→      .then((r) => r.json())
   322→      .then((j: { match: ArchiveMatchClient | null }) => {
   323→        if (cancelled) return;
   324→        setArchive(j.match);
   325→        setArchiveState(j.match ? "found" : "none");
   326→      })
   327→      .catch(() => {
   328→        if (!cancelled) setArchiveState("none");
   329→      });
   330→    return () => {
   331→      cancelled = true;
   332→    };
   333→  }, [target, type, season, episode, searchNonce]);
   334→  // automatic retries are per-episode: a manual "Search again" must not reset them
   335→  useEffect(() => {
   336→    setAutoRetries(0);
   337→  }, [target, type, season, episode]);
   338→
   339→  /* ---------------- auto-pick the best untried server ----------------
   340→   * Walks the ordered list (language filter applied, dead servers last) so
   341→   * the user's language choice and the health checker steer auto-play. */
   342→  useEffect(() => {
   343→    if (serversState !== "ready") return;
   344→    if (manualIdx !== null || activeIdx !== null || exhausted) return;
   345→    const idx = orderedServers.find((i) => !tried.includes(i));
   346→    if (idx !== undefined) {
   347→      setActiveIdx(idx);
   348→      setVideoState("loading");
   349→      streamStartRef.current = Date.now();
   350→    } else {
   351→      setExhausted(true);
   352→    }
   353→  }, [serversState, orderedServers, tried, activeIdx, manualIdx, exhausted]);
   354→
   355→  /* ---------------- server health auto-checker ----------------
   356→   * Probes every found server against the swarm (DHT + trackers) so live
   357→   * servers get a ✓ badge and dead ones drop to the bottom (and out of
   358→   * future sessions, via the persisted dead list). Re-verifies every 10 min
   359→   * while the player stays open — good servers come back automatically. */
   360→  useEffect(() => {
   361→    if (serversState !== "ready" || servers.length === 0 || healthNonce === 0) return;
   362→    let cancelled = false;
   363→    const batch = servers.slice(0, 10).map((s) => ({ id: s.id, name: s.releaseName }));
   364→    setHealthBusy(true);
   365→    setHealth(() => Object.fromEntries(batch.map((s) => [s.id, "checking" as HealthStatus])));
   366→    fetch("/api/torrent/health", {
   367→      method: "POST",
   368→      headers: { "Content-Type": "application/json" },
   369→      body: JSON.stringify({ servers: batch }),
   370→    })
   371→      .then((r) => r.json())
   372→      .then((j: { results?: { id: string; status: "live" | "dead" }[] }) => {
   373→        if (cancelled) return;
   374→        const rows = j.results ?? [];
   375→        setHealth(() => Object.fromEntries(rows.map((r) => [r.id, r.status as HealthStatus])));
   376→        const deadIds = rows.filter((r) => r.status === "dead").map((r) => r.id);
   377→        if (deadIds.length) markDeadServers(deadIds);
   378→        for (const r of rows) if (r.status === "live") reviveServer(r.id);
   379→      })
   380→      .catch(() => {
   381→        if (!cancelled) setHealth({});
   382→      })
   383→      .finally(() => {
   384→        if (!cancelled) setHealthBusy(false);
   385→      });
   386→    return () => {
   387→      cancelled = true;
   388→    };
   389→  }, [serversState, servers, healthNonce, markDeadServers, reviveServer]);
   390→
   391→  // periodic re-verification while the player stays open (auto-updater)
   392→  useEffect(() => {
   393→    if (serversState !== "ready" || servers.length === 0) return;
   394→    const iv = setInterval(() => {
   395→      if (document.visibilityState !== "visible") return;
   396→      setHealthNonce((n) => n + 1);
   397→    }, 10 * 60_000);
   398→    return () => clearInterval(iv);
   399→  }, [serversState, servers.length]);
   400→
   401→  /* ---------------- Otama Direct for episodes with no torrent sources ----------------
   402→   * Daily-show episodes (Bigg Boss Hindi…) are usually absent from public
   403→   * trackers but uploaded to archive.org within hours of broadcast. When the
   404→   * torrent scan comes up empty and a direct episode stream exists, play it. */
   405→  useEffect(() => {
   406→    if (type !== "tv" || serversState !== "ready") return;
   407→    if (servers.length > 0 || manualIdx !== null || exhausted) return;
   408→    if (archiveState !== "found" || !archive) return;
   409→    setManualIdx("archive");
   410→    setVideoState("loading");
   411→  }, [type, serversState, servers.length, manualIdx, exhausted, archiveState, archive]);
   412→
   413→  /* ---------------- stream URL ---------------- */
   414→  const streamUrl = useMemo(() => {
   415→    if (!activeServer || !target) return null;
   416→    const file = activeServer.fileIdx != null ? `&file=${activeServer.fileIdx}` : "";
   417→    // season/episode steer file selection inside season packs and year
   418→    // steers multi-year movie collections — so the exact requested content
   419→    // plays instead of the pack's biggest file. For anime the broadcast-order
   420→    // / fansub-relative numbers steer file picking inside fansub batches
   421→    // ("Bleach - 367.mkv" = S02E01).
   422→    const steer =
   423→      type === "tv"
   424→        ? `&season=${season}&episode=${episode}`
   425→        : `&year=${getYear(target.item)}`;
   426→    const absQ = hints?.absEpisode ? `&abs=${hints.absEpisode}` : "";
   427→    const relQ = hints?.relEpisode ? `&rel=${hints.relEpisode}` : "";
   428→    return `/api/torrent/stream?id=${activeServer.id}&dn=${encodeURIComponent(activeServer.releaseName)}${file}${steer}${absQ}${relQ}`;
   429→  }, [activeServer, type, season, episode, target?.item, hints]);
   430→
   431→  useEffect(() => {
   432→    streamStartRef.current = Date.now();
   433→    setStatus(null);
   434→    setNeedsTap(false);
   435→  }, [streamUrl, videoNonce]);
   436→
   437→  /* ---------------- live status + watchdog failover ---------------- */
   438→  useEffect(() => {
   439→    if (!streamUrl || !activeServer) {
   440→      setStatus(null);
   441→      return;
   442→    }
   443→    let cancelled = false;
   444→    const tick = async () => {
   445→      if (cancelled) return;
   446→      try {
   447→        const r = await fetch(`/api/torrent/status?id=${activeServer.id}`);
   448→        const j: TorrentStatus = await r.json();
   449→        if (cancelled) return;
   450→        setStatus(j);
   451→        if (j.found === false) {
   452→          failover("This server dropped the stream");
   453→          return;
   454→        }
   455→        if (videoStateRef.current === "playing") return;
   456→        const elapsed = Date.now() - streamStartRef.current;
   457→        if (elapsed > NO_PEER_SWITCH_MS && (j.peers ?? 0) === 0 && (j.progress ?? 0) === 0) {
   458→          failover("No peers on this server");
   459→        } else if (elapsed > STALL_SWITCH_MS && videoStateRef.current === "loading") {
   460→          // data may be flowing but the wrong pieces — this server can't start in time
   461→          failover("This server is too slow to start");
   462→        }
   463→      } catch {
   464→        /* transient — next tick retries */
   465→      }
   466→    };
   467→    tick();
   468→    const iv = setInterval(tick, 2000);
   469→    return () => {
   470→      cancelled = true;
   471→      clearInterval(iv);
   472→    };
   473→  }, [streamUrl, activeServer, failover]);
   474→
   475→  /* ---------------- stop torrents on switch / unmount ---------------- */
   476→  useEffect(() => {
   477→    const id = activeServer?.id ?? null;
   478→    if (currentIdRef.current && currentIdRef.current !== id) stopId(currentIdRef.current);
   479→    currentIdRef.current = id;
   480→  }, [activeServer?.id, stopId]);
   481→  useEffect(() => {
   482→    return () => {
   483→      if (currentIdRef.current) stopId(currentIdRef.current);
   484→    };
   485→  }, [stopId]);
   486→
   487→  /* ---------------- record progress for Continue Watching ---------------- */
   488→  useEffect(() => {
   489→    if (!target || !mounted) return;
   490→    const t = getMediaType(target.item);
   491→    addContinue(target.item, t === "tv" ? season : undefined, t === "tv" ? episode : undefined);
   492→  }, [target, mounted, addContinue, season, episode]);
   493→
   494→  /* ---------------- details (season list + trailer fallback) ---------------- */
   495→  useEffect(() => {
   496→    if (!target) return;
   497→    let cancelled = false;
   498→    tmdb<MediaDetails>(`${type}/${target.item.id}`)
   499→      .then((d) => !cancelled && setDetails(d))
   500→      .catch(() => {});
   501→    return () => {
   502→      cancelled = true;
   503→    };
   504→  }, [target, type]);
   505→
   506→  useEffect(() => {
   507→    if (!target || type !== "tv") return;
   508→    let cancelled = false;
   509→    tmdb<{ episodes: Episode[] }>(`tv/${target.item.id}/season/${season}`)
   510→      .then((res) => {
   511→        if (!cancelled) {
   512→          setEpisodes(res.episodes ?? []);
   513→          setEpLoading(false);
   514→        }
   515→      })
   516→      .catch(() => {
