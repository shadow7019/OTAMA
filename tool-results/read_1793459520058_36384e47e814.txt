    41→
    42→function parseCount(raw: string | undefined | null): number {
    43→  if (!raw) return 0;
    44→  const m = raw.replace(/,/g, "").trim().match(/^([\d.]+)\s*([KM])?$/i);
    45→  if (!m) return 0;
    46→  const n = parseFloat(m[1]);
    47→  const mult = m[2]?.toUpperCase() === "K" ? 1e3 : m[2]?.toUpperCase() === "M" ? 1e6 : 1;
    48→  return Math.round((isFinite(n) ? n : 0) * mult);
    49→}
    50→
    51→function parseSizeBytes(raw: string | undefined | null): number {
    52→  if (!raw) return 0;
    53→  const m = raw.replace(/,/g, "").match(/([\d.]+)\s*(B|KB|MB|GB|TB)/i);
    54→  if (!m) return 0;
    55→  const n = parseFloat(m[1]);
    56→  const unit = m[2].toUpperCase();
    57→  const mult = unit === "TB" ? 1e12 : unit === "GB" ? 1e9 : unit === "MB" ? 1e6 : unit === "KB" ? 1e3 : 1;
    58→  return Math.round((isFinite(n) ? n : 0) * mult);
    59→}
    60→
    61→function detectQuality(name: string): string {
    62→  const n = name.toLowerCase();
    63→  if (/\b(2160p|4k)\b/.test(n)) return "2160p";
    64→  if (/\b1080p\b/.test(n)) return "1080p";
    65→  if (/\b720p\b/.test(n)) return "720p";
    66→  if (/\b480p\b|\bdvdrip\b|\bdvd\b/.test(n)) return "480p";
    67→  if (/\b(360p|sd)\b/.test(n)) return "SD";
    68→  return "unknown";
    69→}

    71→async function fetchJson<T>(url: string, timeoutMs = 8000): Promise<T | null> {
    72→  try {
    73→    const res = await fetch(url, {
    74→      headers: { accept: "application/json", "user-agent": "Mozilla/5.0 (X11; Linux x86_64) Otama/1.0" },
    75→      signal: AbortSignal.timeout(timeoutMs),
    76→    });
    77→    if (!res.ok) return null;
    78→    return (await res.json()) as T;
    79→  } catch {
    80→    return null;
    81→  }
    82→}

    86→/**
    87→ * Name-based sanity check applied even to IMDb-keyed providers: indexers
    88→ * sometimes attach wrong IMDb ids to uploads, so a release whose own name
    89→ * contradicts the request (wrong year / wrong episode) is dropped no matter
    90→ * which provider supplied it.
    91→ */
    92→function sanityOk(
    93→  p: ParsedRelease,
    94→  type: "movie" | "tv",
    95→  year: string,
    96→  season: number,
    97→  episode: number
    98→): boolean {
    99→  if (type === "movie") {
   100→    if (p.year !== null && year && Math.abs(p.year - Number(year)) > 1) return false;
   101→    return true;
   102→  }
   103→  if (p.season !== null && p.season !== season) return false;
   104→  if (p.episode !== null && p.episode !== episode) return false;
   105→  return true;
   106→}

   118→/** IMDb-keyed — but indexers mislabel ids, so names are still cross-checked. */
   119→async function providerTorrentio(
   120→  type: "movie" | "tv",
   121→  imdbId: string,
   122→  season: number,
   123→  episode: number,
   124→  year: string
   125→): Promise<TorrentSource[]> {
   126→  // Stremio addon stream ids are single segments: `tt…:season:episode`.
   127→  // (slash-separated paths 404 since the 2026 addon API tightening)
   128→  const path =
   129→    type === "movie"
   130→      ? `stream/movie/${imdbId}.json`
   131→      : `stream/series/${imdbId}:${season}:${episode}.json`;
   132→  const data = await fetchJson<{ streams?: TorrentioStream[] }>(
   133→    `https://torrentio.strem.fun/${path}`
   134→  );
   135→  const streams = data?.streams ?? [];
   136→  const out: TorrentSource[] = [];
   137→  for (const s of streams) {
   138→    if (!s.infoHash || !/^[0-9a-f]{40}$/i.test(s.infoHash)) continue;
   139→    const titleLines = (s.title ?? "").split("\n");
   140→    const releaseName = (titleLines[0] || s.behaviorHints?.filename || "Torrent").trim();
   141→    // indexers occasionally attach a wrong IMDb id — cross-check the name
   142→    const parsed = parseRelease(releaseName);
   143→    if (!sanityOk(parsed, type, year, season, episode)) continue;
   144→    const metaLine = titleLines.find((l) => l.includes("👤")) ?? "";
   145→    const seeders = parseCount(metaLine.match(/👤\s*([\d.,]+\s*[KM]?)/)?.[1]);
   146→    const sizeBytes = parseSizeBytes(metaLine.match(/💾\s*([\d.,]+\s*[KMGTB]+)/i)?.[1]);
   147→    const provider = metaLine.match(/⚙️\s*([^\n]+)/)?.[1]?.trim() || "Torrentio";
   148→    const hintName = s.name ?? "";
   149→    const quality =
   150→      /4k/i.test(hintName) && !/1080p/.test(releaseName)
   151→        ? "2160p"
   152→        : detectQuality(releaseName);
   153→    out.push({
   154→      id: s.infoHash.toLowerCase(),
   155→      releaseName,
   156→      seeders,
   157→      peers: 0,
   158→      sizeBytes,
   159→      provider,
   160→      quality,
   161→      fileIdx: typeof s.fileIdx === "number" ? s.fileIdx : null,
   162→      match: "verified",
   163→      score: 0,
   164→    });
   165→  }
   166→  return out;
   167→}

   191→/** Air date (YYYY-MM-DD) of one episode — short TTL so brand-new daily-show
   192→ *  episodes (Bigg Boss S20E55…) get their date as soon as TMDB publishes it. */
   193→const seasonCache = new Map<string, { dates: Map<number, string | null>; ts: number }>();
   194→const SEASON_TTL = 30 * 60 * 1000;
   195→
   196→export async function getEpisodeAirDate(
   197→  tmdbId: number,
   198→  season: number,
   199→  episode: number
   200→): Promise<string | null> {
   201→  const key = `tv:${tmdbId}:${season}`;
   202→  let hit = seasonCache.get(key);
   203→  if (!hit || Date.now() - hit.ts > SEASON_TTL) {
   204→    try {
   205→      const data = await tmdbServer<{
   206→        episodes?: { episode_number: number; air_date: string | null }[];
   207→      }>(`tv/${tmdbId}/season/${season}`);
   208→      const dates = new Map<number, string | null>();
   209→      for (const e of data.episodes ?? []) dates.set(e.episode_number, e.air_date ?? null);
   210→      hit = { dates, ts: Date.now() };
   211→      seasonCache.set(key, hit);
   212→    } catch {
   213→      return null;
   214→    }
   215→  }
   216→  return hit.dates.get(episode) ?? null;
   217→}

   237→/**
   238→ * Keyword provider — answers on title words alone, so every result is
   239→ * strictly validated here against the requested year (movies) or
   240→ * season/episode (TV) before it may enter the candidate list.
   241→ */
   242→async function providerPirateBay(
   243→  type: "movie" | "tv",
   244→  ctx: BayContext
   245→): Promise<TorrentSource[]> {
   246→  const { title, originalTitle, year, season, episode } = ctx;
   247→  const queries =
   248→    type === "movie"
   249→      ? [`${title} ${year ?? ""}`.trim(), title]
   250→      : [
   251→          `${title} s${pad(season)}e${pad(episode)}`,
   252→          // language variant — Indian versions ("Bigg Boss Telugu") overflow the
   253→          // bare-title result window on trackers
   254→          ...(ctx.showLang ? [`${title} ${ctx.showLang}`] : []),
   255→          `${title} ${season}x${pad(episode)}`,
   256→          title, // dated daily-show names carry no SxxExx — match by name + air date
   257→        ];
   258→
   259→  const seen = new Map<string, TorrentSource>();
   260→  const collect = (rows: ApibayResult[]) => {
   261→    for (const r of rows) {
   262→      if (!r.info_hash || !/^[0-9a-f]{40}$/i.test(r.info_hash)) continue;
   263→      const name = (r.name ?? "Torrent").trim();
   264→      if (name === "No results") continue;
   265→      const p = parseRelease(name);
   266→      const match =
   267→        type === "movie"
   268→          ? matchMovie(p, title, originalTitle, year)
   269→          : matchTv(p, season, episode, title, originalTitle, {
   270→              airDate: ctx.airDate,
   271→              showLang: ctx.showLang,
   272→              absEpisode: ctx.absEpisode,
   273→            });
   274→      if (!match) continue; // same name, different film/episode — reject
   275→      const cPen = type === "tv" ? countryPenalty(p, ctx.countries) : 0;
   276→      // an explicit country tag that contradicts the show ("The.Office.US"
   277→      // when the TMDB show is GB) is a different show — hard reject
   278→      if (cPen < 0) continue;
   279→      const key = r.info_hash.toLowerCase();
   280→      if (seen.has(key)) continue;
   281→      seen.set(key, {
   282→        id: key,
   283→        releaseName: name,
   284→        seeders: parseCount(r.seeders),
   285→        peers: parseCount(r.leechers),
   286→        sizeBytes: parseSizeBytes(r.size),
   287→        provider: "ThePirateBay",
   288→        quality: detectQuality(name),
   289→        fileIdx: null,
   290→        match,
   291→        countryPenalty: cPen,
   292→        score: 0,
   293→      });
   294→    }
   295→  };
   296→
   297→  // apibay answers tiny JSON quickly; sequential keeps request count low
   298→  for (const q of queries) {
   299→    if (seen.size >= 5) break;
   300→    const rows = await fetchJson<ApibayResult[]>(
   301→      `https://apibay.org/q.php?q=${encodeURIComponent(q)}&cat=`,
   302→      7000
   303→    );
   304→    if (rows && Array.isArray(rows)) collect(rows);
   305→  }
   306→
   307→  // episode-level search came up empty → try season packs for TV
   308→  if (type === "tv" && seen.size === 0) {
   309→    const packQueries = [`${title} s${pad(season)}`, `${title} season ${season}`];
   310→    for (const q of packQueries) {
   311→      const rows = await fetchJson<ApibayResult[]>(
   312→        `https://apibay.org/q.php?q=${encodeURIComponent(q)}&cat=`,
   313→        7000
   314→      );
   315→      if (rows && Array.isArray(rows)) collect(rows);
   316→      if (seen.size > 0) break;
   317→    }
   318→  }
   319→  return [...seen.values()];
   320→}

   324→  name: string;
   325→  seeders: number;
   326→  peers: number;
   327→  sizeBytes: number;

   330→/**
   331→ * BitSearch (bitsearch.eu) — HTML search, the last live indexer when the
   332→ * JSON APIs are down or blocked. It indexes Indian release groups
   333→ * (1TamilBlasters, TamilMV, 3MovieRulz…) that upload daily-show episodes
   334→ * ("Bigg Boss (2026) S09E101 …") hours after broadcast. Every row is
   335→ * validated with the same strict matcher as the keyword providers.
   336→ */
   337→async function providerBitSearch(type: "movie" | "tv", ctx: BayContext): Promise<TorrentSource[]> {
   338→  const { title, originalTitle, year, season, episode } = ctx;
   339→  const queries =
   340→    type === "movie"
   341→      ? [`${title} ${year ?? ""}`.trim(), title]
   342→      : [
   343→          `${title} s${pad(season)}e${pad(episode)}`,
   344→          `${title} season ${season}`, // dated daily shows + season packs
   345→        ];
   346→
   347→  const cards = new Map<string, BitsearchCard>();
   348→  const scrape = async (q: string) => {
   349→    try {
   350→      const res = await fetch(`https://bitsearch.eu/search?q=${encodeURIComponent(q)}`, {
   351→        headers: {
   352→          accept: "text/html,application/xhtml+xml",
   353→          "user-agent":
   354→            "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0.0.0 Safari/537.36",
   355→        },
   356→        signal: AbortSignal.timeout(9000),
   357→        redirect: "follow",
   358→      });
   359→      if (!res.ok) return;
   360→      const html = (await res.text()).replace(/\s+/g, " ");
   361→      const re = /\/download\/torrent\/([0-9a-fA-F]{40})\?title=([^">]+)/g;
   362→      for (const m of html.matchAll(re)) {
   363→        const hash = m[1].toLowerCase();
   364→        if (cards.has(hash)) continue;
   365→        let name = m[2];
   366→        try { name = decodeURIComponent(name); } catch { /* raw text fallback */ }
   367→        name = name
   368→          .replace(/&amp;/g, "&")
   369→          .replace(/&#x3D;|&#61;/gi, "=")
   370→          .replace(/&quot;/gi, '"')
   371→          .replace(/&#39;/g, "'")
   372→          .trim();
   373→        const win = html.slice(Math.max(0, m.index - 2200), m.index);
   374→        const seeders = Number(win.match(/font-medium">(\d+)<\/span>\s*<span>\s*seeders/)?.[1] ?? 0);
   375→        const peers = Number(win.match(/font-medium">(\d+)<\/span>\s*<span>\s*leechers/)?.[1] ?? 0);
   376→        const sizeBytes = parseSizeBytes(
   377→          win.match(/fa-download"><\/i>\s*<span>\s*([\d.]+\s*[KMGT]?B)</i)?.[1] ?? null
   378→        );
   379→        cards.set(hash, { hash, name, seeders, peers, sizeBytes });
   380→      }
   381→    } catch {
   382→      // blocked / timed out — the other providers still answer
   383→    }
   384→  };
   385→
   386→  await Promise.all(queries.map(scrape));
   387→
   388→  const out: TorrentSource[] = [];
   389→  for (const c of cards.values()) {
   390→    const p = parseRelease(c.name);
   391→    const match =
   392→      type === "movie"
   393→        ? matchMovie(p, title, originalTitle, year)
   394→        : matchTv(p, season, episode, title, originalTitle, {
   395→            airDate: ctx.airDate,
   396→            showLang: ctx.showLang,
   397→            absEpisode: ctx.absEpisode,
   398→          });
   399→    if (!match) continue; // same name, wrong season/episode/year — reject
   400→    if (type === "tv" && countryPenalty(p, ctx.countries) < 0) continue;
   401→    out.push({
   402→      id: c.hash,
   403→      releaseName: c.name,
   404→      seeders: c.seeders,
   405→      peers: c.peers,
   406→      sizeBytes: c.sizeBytes,
   407→      provider: "BitSearch",
   408→      quality: detectQuality(c.name),
   409→      fileIdx: null,
   410→      match,
   411→      score: 0,
   412→    });
   413→  }
   414→  return out;
   415→}

   491→/** IMDb-keyed, but the endpoint fuzzy-matches — pin to the exact IMDb code. */
   492→async function providerYts(imdbId: string, year: string, timeoutMs = 7000): Promise<TorrentSource[]> {
   493→  const data = await fetchJson<{ data?: { movies?: YtsMovie[] } }>(
   494→    `https://yts.mx/api/v2/list_movies.json?query_term=${encodeURIComponent(imdbId)}&limit=5`,
   495→    timeoutMs
   496→  );
   497→  const movies = (data?.data?.movies ?? []).filter(
   498→    (mv) => (mv.imdb_code ?? "").toLowerCase() === imdbId.toLowerCase()
   499→  );
   500→  const out: TorrentSource[] = [];
   501→  for (const mv of movies) {
   502→    // sanity: YTS metadata year must agree with TMDB (±1)
   503→    if (year && mv.year && Math.abs(mv.year - Number(year)) > 1) continue;
   504→    for (const t of mv.torrents ?? []) {
   505→      if (!t.hash || !/^[0-9a-f]{40}$/i.test(t.hash)) continue;
   506→      out.push({
   507→        id: t.hash.toLowerCase(),
   508→        releaseName: `${mv.title ?? "Movie"} ${mv.year ?? ""} ${t.quality ?? ""} YTS`.replace(/\s+/g, " ").trim(),
   509→        seeders: t.seeds ?? 0,
   510→        peers: t.peers ?? 0,
   511→        sizeBytes: parseSizeBytes(t.size),
   512→        provider: "YTS",
   513→        quality: t.quality ?? detectQuality(`${t.size}`),
   514→        fileIdx: null,
   515→        match: "verified",
   516→        score: 0,
   517→      });
   518→    }
   519→  }
   520→  return out;
   521→}

   533→/** EZTV rotates domains; the first mirror that answers wins. */
   534→const EZTV_HOSTS = ["eztvx.to", "eztv.wf", "eztv.tf"];
   535→
   536→async function eztvPage(
   537→  numericId: string,
   538→  page: number
   539→): Promise<{ torrents?: EztvTorrent[] } | null> {
   540→  for (const host of EZTV_HOSTS) {
   541→    const data = await fetchJson<{ torrents?: EztvTorrent[] }>(
   542→      `https://${host}/api/get-torrents?imdb_id=${numericId}&limit=100&page=${page}`,
   543→      7000
   544→    );
   545→    if (data) return data; // any well-formed answer — even empty — ends the chain
   546→  }
   547→  return null;
   548→}
   549→
   550→/**
   551→ * IMDb-keyed per episode. The API rows carry season/episode metadata and
   552→ * ignore the season/episode query params, so we scan a couple of recency
   553→ * pages; the filename is cross-checked so a mislabeled row can't leak wrong
   554→ * episodes into same-named shows.
   555→ */
   556→async function providerEztv(
   557→  imdbId: string,
   558→  season: number,
   559→  episode: number
   560→): Promise<TorrentSource[]> {
   561→  const out: TorrentSource[] = [];
   562→  const numericId = imdbId.replace(/^tt/i, "");
   563→  for (let page = 1; page <= 2 && out.length < 8; page++) {
   564→    const data = await eztvPage(numericId, page);
   565→    const rows = data?.torrents ?? [];
   566→    for (const r of rows) {
   567→      if (!r.hash || !/^[0-9a-f]{40}$/i.test(r.hash)) continue;
   568→      if (r.season && season && r.season !== season) continue;
   569→      if (r.episode && episode && r.episode !== episode) continue;
   570→      const filename = (r.filename ?? "Episode").trim();
   571→      const p = parseRelease(filename);
   572→      const wrongEpisode =
   573→        (p.season !== null && p.season !== season) ||
   574→        (p.episode !== null && p.episode !== episode);
   575→      if (wrongEpisode) continue;
   576→      out.push({
   577→        id: r.hash.toLowerCase(),
   578→        releaseName: filename,
   579→        seeders: r.seeds ?? 0,
   580→        peers: r.peers ?? 0,
   581→        sizeBytes: r.size_bytes ?? 0,
   582→        provider: "EZTV",
   583→        quality: detectQuality(filename),
   584→        fileIdx: null,
   585→        match: "verified",
   586→        score: 0,
   587→      });
   588→      if (out.length >= 8) break;
   589→    }
   590→    if (rows.length < 100) break; // no more pages for this show
   591→  }
   592→  return out;
   593→}

   595→/* ---------- anime: nyaa.si (fansub indexers) ---------- */
   596→
   597→interface NyaaItem {
   598→  title: string;
   599→  infoHash: string;
   600→  seeders: number;
   601→  leechers: number;
   602→  sizeBytes: number;
   603→  /** upload date (epoch ms) — identifies fresh sequel-episode fansubs */
   604→  pubDate: number | null;
   605→}
   606→
   607→/** RSS search on nyaa.si — the primary public index for fansubbed anime. */
   608→async function nyaaSearch(query: string): Promise<NyaaItem[]> {
   609→  try {
   610→    const res = await fetch(
   611→      `https://nyaa.si/?page=rss&f=0&c=0_0&q=${encodeURIComponent(query)}`,
   612→      {
   613→        headers: {
   614→          accept: "application/rss+xml",
   615→          "user-agent": "Mozilla/5.0 (X11; Linux x86_64) Otama/1.0",
   616→        },
   617→        signal: AbortSignal.timeout(9000),
   618→      }
   619→    );
   620→    if (!res.ok) return [];
   621→    const xml = await res.text();
   622→    const items: NyaaItem[] = [];
   623→    for (const chunk of xml.split("<item>").slice(1)) {
   624→      const pick = (re: RegExp) => chunk.match(re)?.[1]?.trim() ?? "";
   625→      const title = pick(/<title>([\s\S]*?)<\/title>/);
   626→      const infoHash = pick(/<nyaa:infoHash>([0-9a-fA-F]{40})<\/nyaa:infoHash>/).toLowerCase();
   627→      if (!title || !infoHash) continue;
   628→      // "1.4 GiB" → "1.4 GB" so the shared size parser understands it
   629→      const sizeRaw = pick(/<nyaa:size>([\d.]+\s*[KMGTP]i?B)<\/nyaa:size>/).replace(/i/g, "");
   630→      const pubRaw = pick(/<pubDate>([\s\S]*?)<\/pubDate>/);
   631→      const pubMs = pubRaw ? new Date(pubRaw).getTime() : NaN;
   632→      items.push({
   633→        title,
   634→        infoHash,
   635→        seeders: parseInt(pick(/<nyaa:seeders>(\d+)<\/nyaa:seeders>/), 10) || 0,
   636→        leechers: parseInt(pick(/<nyaa:leechers>(\d+)<\/nyaa:leechers>/), 10) || 0,
   637→        sizeBytes: parseSizeBytes(sizeRaw),
   638→        pubDate: Number.isFinite(pubMs) ? pubMs : null,
   639→      });
   640→    }
   641→    return items;
   642→  } catch {
   643→    return [];
   644→  }
   645→}

   647→/** "892-1085" / "1078.5-1085" / "E01-E52" batch range inside a fansub
   648→ *  release name. */
   649→export function batchRange(name: string): [number, number] | null {
   650→  const m = name.match(/(\d{1,4})(?:\.\d)?\s*(?:[-–—~]|to)\s*E?(\d{1,4})\b/i);
   651→  if (!m) return null;
   652→  const a = Number(m[1]);
   653→  const b = Number(m[2]);
   654→  return b > a && b - a <= 400 ? [a, b] : null;
   655→}
   656→
   657→/**
   658→ * Nyaa — fansubbed anime releases. Names use broadcast-order episode numbers
   659→ * ("One Piece - 1085"), season/episode tags ("S02E05") or batch ranges
   660→ * ("Season 20 (892-1085) Batch"), all handled via the strict matcher plus an
   661→ * explicit absolute-episode check. Only enabled for Japanese titles.
   662→ */
   663→async function providerNyaa(type: "movie" | "tv", ctx: BayContext): Promise<TorrentSource[]> {
   664→  const { title, originalTitle, season, episode, absEpisode, relEpisode, isAnime } = ctx;
   665→  if (!isAnime) return [];
   666→  const wanted = absEpisode ?? episode;
   667→  const relWanted = relEpisode ?? null;
   668→  const queries =
   669→    type === "movie"
   670→      ? [`${title} ${ctx.year ?? ""}`.trim(), title, ...(originalTitle ? [originalTitle] : [])]
   671→      : [
   672→          `${title} ${wanted}`,
   673→          // merged sequel seasons are fansub-numbered from 1 ("Sennen
   674→          // Kessen-hen - 45"), which differs from the broadcast-order number
   675→          ...(relWanted != null && relWanted !== wanted ? [`${title} ${relWanted}`] : []),
   676→          ...(originalTitle ? [`${originalTitle} ${wanted}`] : []),
   677→          `${title} s${pad(season)}e${pad(episode)}`,
   678→        ];
   679→
   680→  const seen = new Map<string, TorrentSource>();
   681→  for (const q of queries) {
   682→    if (seen.size >= 6) break;
   683→    const rows = await nyaaSearch(q);
   684→    for (const r of rows) {
   685→      if (seen.has(r.infoHash)) continue;
   686→      const p = parseRelease(r.title);
   687→      let match: MatchClass | null;
   688→      if (type === "movie") {
   689→        match = matchMovie(p, title, originalTitle, ctx.year);
   690→      } else {
   691→        match = matchTv(p, season, episode, title, originalTitle, {
   692→          airDate: ctx.airDate,
   693→          absEpisode,
   694→          relEpisode: relWanted,
   695→          pubDate: r.pubDate,
   696→        });
   697→        if (!match) {
   698→          // multi-episode batch: keep it as a pack — the stream route picks
   699→          // the right file inside the torrent by episode number. The relative
   700→          // number is only accepted inside a SUBTITLED release (>= 2 unconsumed
   701→          // title words — "Sennen Kessen-hen 1-52 Batch"), so the base show's
   702→          // batches ("Bleach 001~366", "BLEACH Kai 1~366") can never answer
   703→          // a sequel-season request.
   704→          const br = batchRange(r.title);
   705→          if (br) {
   706→            const inAbs = wanted >= br[0] && wanted <= br[1];
   707→            const inRel =
   708→              relWanted != null && relWanted >= br[0] && relWanted <= br[1];
   709→            if (inAbs || (inRel && subtitleWords(p, [title, originalTitle]).length >= 2)) {
   710→              match = "pack";
   711→            }
   712→          }
   713→        }
   714→      }
   715→      if (!match) continue;
   716→      seen.set(r.infoHash, {
   717→        id: r.infoHash,
   718→        releaseName: r.title,
   719→        seeders: r.seeders,
   720→        peers: r.leechers,
   721→        sizeBytes: r.sizeBytes,
   722→        provider: "Nyaa",
   723→        quality: detectQuality(r.title),
   724→        fileIdx: null,
   725→        match,
   726→        score: 0,
   727→      });
   728→    }
   729→  }
   730→  return [...seen.values()];
   731→}

   733→/* ---------- scoring ---------- */
   734→
   735→export function scoreTorrent(t: TorrentSource): number {
   736→  let s = Math.min(Math.log10(t.seeders + 1) * 4, 13);
   737→  if (t.quality === "1080p") s += 5;
   738→  else if (t.quality === "720p") s += 4;
   739→  else if (t.quality === "2160p") s += 2;
   740→  else if (t.quality === "480p") s += 1;
   741→  const n = t.releaseName.toLowerCase();
   742→  if (/x265|h\.?265|hevc/.test(n)) s -= 4; // HEVC support is spotty in browsers
   743→  if (/av1/.test(n)) s += 1;
   744→  if (/x264|h\.?264|xvid/.test(n)) s += 2;
   745→  if (/\b(cam|ts|telesync|screener|dvdscr|hdts|\btc\b)\b/.test(n)) s -= 12;
   746→  if (/\b3d\b/.test(n)) s -= 6;
   747→  if (/\.mp4\b/.test(n) || /yts/i.test(t.releaseName)) s += 1.5;
   748→  if (t.sizeBytes > 0 && t.sizeBytes < 120 * 1024 * 1024) s -= 2;
   749→  if (t.sizeBytes > 25 * 1024 * 1024 * 1024) s -= 3;
   750→  return Math.round(s * 10) / 10;
   751→}
   752→
   753→/** Bonus/penalty applied for how tightly a release matches the request. */
   754→const MATCH_RANK: Record<MatchClass, number> = {
   755→  verified: 8,
   756→  pack: 2.5,
   757→  weak: -6,
   758→};
   759→
   760→function merge(sources: TorrentSource[]): TorrentSource[] {
   761→  const best = new Map<string, TorrentSource>();
   762→  for (const t of sources) {
   763→    const prev = best.get(t.id);
   764→    if (!prev || t.seeders > prev.seeders) best.set(t.id, t);
   765→  }
   766→  const list = [...best.values()];
   767→  for (const t of list) {
   768→    t.score =
   769→      Math.round((scoreTorrent(t) + MATCH_RANK[t.match] + (t.countryPenalty ?? 0)) * 10) / 10;
   770→  }
   771→  // only fall back to unverified (no-year) releases when nothing better exists
   772→  const confident = list.filter((t) => t.match !== "weak");
   773→  const final = confident.length ? confident : list;
   774→  return final.sort((a, b) => b.score - a.score || b.seeders - a.seeders).slice(0, 14);
   775→}

   788→export async function searchTorrentSources(args: SearchArgs): Promise<TorrentSource[]> {
   789→  const { type, tmdbId, title, season, episode } = args;
   790→  const [imdbId, meta] = await Promise.all([getImdbId(type, tmdbId), getTitleMeta(type, tmdbId)]);
   791→  // TMDB's own year is canonical (guards against stale/rounded client data)
   792→  const year = meta.year || args.year || "";
   793→  // Indian daily shows (Bigg Boss…) name episodes by broadcast date and tag
   794→  // language versions — both are needed to match them without mixing content
   795→  const airDate = type === "tv" ? await getEpisodeAirDate(tmdbId, season, episode) : null;
   796→  const showLang = meta.originalLanguage ? LANG_WORD[meta.originalLanguage] ?? null : null;
   797→  // "Bigg Boss Tamil" uploads are named "Bigg Boss (2026) S09E101 [Tamil …]":
   798→  // the language moves into the tag area, so match against the base title and
   799→  // let the release's language tag pick the right version (mismatch = reject)
   800→  const matchTitle =
   801→    type === "tv" && showLang
   802→      ? title.replace(new RegExp(`\\s+${showLang}$`, "i"), "").trim() || title
   803→      : title;
   804→
   805→  // Japanese titles are anime: fansub indexers (Nyaa) get queried and the
   806→  // broadcast-order episode number is offered to the matchers. Sequel seasons
   807→  // folded into the base show ("Bleach" S02 = TYBW) additionally carry the
   808→  // fansub-relative episode number.
   809→  const isAnime = (meta.originalLanguage ?? "") === "ja";
   810→  const absEpisode = type === "tv" && isAnime ? await getAbsoluteEpisode(tmdbId, season, episode) : null;
   811→  const relEpisode = type === "tv" && isAnime && season >= 2 ? episode : null;
   812→
   813→  const jobs: Promise<TorrentSource[]>[] = [];
   814→  if (matchTitle) {
   815→    const bayCtx: BayContext = {
   816→      title: matchTitle,
   817→      originalTitle: meta.originalTitle ?? undefined,
   818→      year: year || undefined,
   819→      countries: meta.countries,
   820→      season,
   821→      episode,
   822→      airDate,
   823→      showLang,
   824→      absEpisode,
   825→      relEpisode,
   826→      isAnime,
   827→    };
   828→    jobs.push(providerPirateBay(type, bayCtx));
   829→    jobs.push(providerBitSearch(type, bayCtx));
   830→    jobs.push(providerTorrentsCsv(type, bayCtx));
   831→    if (isAnime) jobs.push(providerNyaa(type, bayCtx));
   832→  }
   833→  if (imdbId) {
   834→    jobs.push(providerTorrentio(type, imdbId, season, episode, year));
   835→    if (type === "movie") jobs.push(providerYts(imdbId, year));
   836→    else jobs.push(providerEztv(imdbId, season, episode));
   837→  }
   838→
   839→  const settled = await Promise.allSettled(jobs);
   840→  const all: TorrentSource[] = [];
   841→  for (const r of settled) if (r.status === "fulfilled") all.push(...r.value);
   842→  return merge(all);
   843→}

   845→/** Cheap availability probe for the auto-content pipeline (Torrentio only). */
   846→const availCache = new Map<string, { n: number; ts: number }>();
   847→const AVAIL_TTL = 30 * 60 * 1000;
   848→
   849→export async function quickAvailability(
   850→  type: "movie" | "tv",
   851→  imdbId: string,
   852→  season: number,
   853→  episode: number
   854→): Promise<number> {
   855→  const key = `${type}:${imdbId}:${season}:${episode}`;
   856→  const hit = availCache.get(key);
   857→  if (hit && Date.now() - hit.ts < AVAIL_TTL) return hit.n;
   858→  const data = await fetchJson<{ streams?: TorrentioStream[] }>(
   859→    type === "movie"
   860→      ? `https://torrentio.strem.fun/stream/movie/${imdbId}.json`
   861→      : `https://torrentio.strem.fun/stream/series/${imdbId}:${season}:${episode}.json`,
   862→    8000
   863→  );
   864→  const n = (data?.streams ?? []).length;
   865→  availCache.set(key, { n, ts: Date.now() });
   866→  return n;
   867→}
