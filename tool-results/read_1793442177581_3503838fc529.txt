    41→    41→function parseCount(raw: string | undefined | null): number {
    42→    42→  if (!raw) return 0;
    43→    43→  const m = raw.replace(/,/g, "").trim().match(/^([\d.]+)\s*([KM])?$/i);
    44→    44→  if (!m) return 0;
    45→    45→  const n = parseFloat(m[1]);
    46→    46→  const mult = m[2]?.toUpperCase() === "K" ? 1e3 : m[2]?.toUpperCase() === "M" ? 1e6 : 1;
    47→    47→  return Math.round((isFinite(n) ? n : 0) * mult);
    48→    48→}
    49→    49→
    50→    50→function parseSizeBytes(raw: string | undefined | null): number {
    51→    51→  if (!raw) return 0;
    52→    52→  const m = raw.replace(/,/g, "").match(/([\d.]+)\s*(B|KB|MB|GB|TB)/i);
    53→    53→  if (!m) return 0;
    54→    54→  const n = parseFloat(m[1]);
    55→    55→  const unit = m[2].toUpperCase();
    56→    56→  const mult = unit === "TB" ? 1e12 : unit === "GB" ? 1e9 : unit === "MB" ? 1e6 : unit === "KB" ? 1e3 : 1;
    57→    57→  return Math.round((isFinite(n) ? n : 0) * mult);
    58→    58→}

    60→    60→function detectQuality(name: string): string {
    61→    61→  const n = name.toLowerCase();
    62→    62→  if (/\b(2160p|4k)\b/.test(n)) return "2160p";
    63→    63→  if (/\b1080p\b/.test(n)) return "1080p";
    64→    64→  if (/\b720p\b/.test(n)) return "720p";
    65→    65→  if (/\b480p\b|\bdvdrip\b|\bdvd\b/.test(n)) return "480p";
    66→    66→  if (/\b(360p|sd)\b/.test(n)) return "SD";
    67→    67→  return "unknown";
    68→    68→}

    91→    91→function sanityOk(

    94→    94→  year: string,
    95→    95→  season: number,
    96→    96→  episode: number
    97→    97→): boolean {

   100→   100→    return true;
   101→   101→  }

   103→   103→  if (p.episode !== null && p.episode !== episode) return false;
   104→   104→  return true;
   105→   105→}

   118→   118→async function providerTorrentio(
   119→   119→  type: "movie" | "tv",
   120→   120→  imdbId: string,
   121→   121→  season: number,
   122→   122→  episode: number,
   123→   123→  year: string
   124→   124→): Promise<TorrentSource[]> {
   125→   125→  // Stremio addon stream ids are single segments: `tt…:season:episode`.
   126→   126→  // (slash-separated paths 404 since the 2026 addon API tightening)
   127→   127→  const path =
   128→   128→    type === "movie"
   129→   129→      ? `stream/movie/${imdbId}.json`
   130→   130→      : `stream/series/${imdbId}:${season}:${episode}.json`;
   131→   131→  const data = await fetchJson<{ streams?: TorrentioStream[] }>(
   132→   132→    `https://torrentio.strem.fun/${path}`
   133→   133→  );
   134→   134→  const streams = data?.streams ?? [];
   135→   135→  const out: TorrentSource[] = [];
   136→   136→  for (const s of streams) {
   137→   137→    if (!s.infoHash || !/^[0-9a-f]{40}$/i.test(s.infoHash)) continue;
   138→   138→    const titleLines = (s.title ?? "").split("\n");
   139→   139→    const releaseName = (titleLines[0] || s.behaviorHints?.filename || "Torrent").trim();
   140→   140→    // indexers occasionally attach a wrong IMDb id — cross-check the name
   141→   141→    const parsed = parseRelease(releaseName);
   142→   142→    if (!sanityOk(parsed, type, year, season, episode)) continue;
   143→   143→    const metaLine = titleLines.find((l) => l.includes("👤")) ?? "";
   144→   144→    const seeders = parseCount(metaLine.match(/👤\s*([\d.,]+\s*[KM]?)/)?.[1]);
   145→   145→    const sizeBytes = parseSizeBytes(metaLine.match(/💾\s*([\d.,]+\s*[KMGTB]+)/i)?.[1]);
   146→   146→    const provider = metaLine.match(/⚙️\s*([^\n]+)/)?.[1]?.trim() || "Torrentio";
   147→   147→    const hintName = s.name ?? "";
   148→   148→    const quality =
   149→   149→      /4k/i.test(hintName) && !/1080p/.test(releaseName)
   150→   150→        ? "2160p"
   151→   151→        : detectQuality(releaseName);
   152→   152→    out.push({
   153→   153→      id: s.infoHash.toLowerCase(),
   154→   154→      releaseName,
   155→   155→      seeders,
   156→   156→      peers: 0,
   157→   157→      sizeBytes,
   158→   158→      provider,
   159→   159→      quality,
   160→   160→      fileIdx: typeof s.fileIdx === "number" ? s.fileIdx : null,
   161→   161→      match: "verified",
   162→   162→      score: 0,
   163→   163→    });
   164→   164→  }
   165→   165→  return out;
   166→   166→}

   201→   201→  let hit = seasonCache.get(key);
   202→   202→  if (!hit || Date.now() - hit.ts > SEASON_TTL) {

   208→   208→      for (const e of data.episodes ?? []) dates.set(e.episode_number, e.air_date ?? null);
   209→   209→      hit = { dates, ts: Date.now() };
   210→   210→      seasonCache.set(key, hit);
   211→   211→    } catch {
   212→   212→      return null;
   213→   213→    }

   234→   234→ * Keyword provider — answers on title words alone, so every result is
   235→   235→ * strictly validated here against the requested year (movies) or
   236→   236→ * season/episode (TV) before it may enter the candidate list.

   238→   238→async function providerPirateBay(
   239→   239→  type: "movie" | "tv",
   240→   240→  ctx: BayContext
   241→   241→): Promise<TorrentSource[]> {
   242→   242→  const { title, originalTitle, year, season, episode } = ctx;
   243→   243→  const queries =
   244→   244→    type === "movie"
   245→   245→      ? [`${title} ${year ?? ""}`.trim(), title]
   246→   246→      : [
   247→   247→          `${title} s${pad(season)}e${pad(episode)}`,
   248→   248→          // language variant — Indian versions ("Bigg Boss Telugu") overflow the
   249→   249→          // bare-title result window on trackers
   250→   250→          ...(ctx.showLang ? [`${title} ${ctx.showLang}`] : []),
   251→   251→          `${title} ${season}x${pad(episode)}`,
   252→   252→          title, // dated daily-show names carry no SxxExx — match by name + air date
   253→   253→        ];
   254→   254→
   255→   255→  const seen = new Map<string, TorrentSource>();
   256→   256→  const collect = (rows: ApibayResult[]) => {
   257→   257→    for (const r of rows) {
   258→   258→      if (!r.info_hash || !/^[0-9a-f]{40}$/i.test(r.info_hash)) continue;
   259→   259→      const name = (r.name ?? "Torrent").trim();
   260→   260→      if (name === "No results") continue;
   261→   261→      const p = parseRelease(name);
   262→   262→      const match =
   263→   263→        type === "movie"
   264→   264→          ? matchMovie(p, title, originalTitle, year)
   265→   265→          : matchTv(p, season, episode, title, originalTitle, {
   266→   266→              airDate: ctx.airDate,
   267→   267→              showLang: ctx.showLang,
   268→   268→              absEpisode: ctx.absEpisode,
   269→   269→            });
   270→   270→      if (!match) continue; // same name, different film/episode — reject
   271→   271→      const cPen = type === "tv" ? countryPenalty(p, ctx.countries) : 0;
   272→   272→      // an explicit country tag that contradicts the show ("The.Office.US"
   273→   273→      // when the TMDB show is GB) is a different show — hard reject
   274→   274→      if (cPen < 0) continue;
   275→   275→      const key = r.info_hash.toLowerCase();
   276→   276→      if (seen.has(key)) continue;
   277→   277→      seen.set(key, {
   278→   278→        id: key,
   279→   279→        releaseName: name,
   280→   280→        seeders: parseCount(r.seeders),
   281→   281→        peers: parseCount(r.leechers),
   282→   282→        sizeBytes: parseSizeBytes(r.size),
   283→   283→        provider: "ThePirateBay",
   284→   284→        quality: detectQuality(name),
   285→   285→        fileIdx: null,
   286→   286→        match,
   287→   287→        countryPenalty: cPen,
   288→   288→        score: 0,
   289→   289→      });
   290→   290→    }
   291→   291→  };
   292→   292→
   293→   293→  // apibay answers tiny JSON quickly; sequential keeps request count low
   294→   294→  for (const q of queries) {
   295→   295→    if (seen.size >= 5) break;
   296→   296→    const rows = await fetchJson<ApibayResult[]>(
   297→   297→      `https://apibay.org/q.php?q=${encodeURIComponent(q)}&cat=`,
   298→   298→      7000
   299→   299→    );
   300→   300→    if (rows && Array.isArray(rows)) collect(rows);
   301→   301→  }
   302→   302→
   303→   303→  // episode-level search came up empty → try season packs for TV
   304→   304→  if (type === "tv" && seen.size === 0) {
   305→   305→    const packQueries = [`${title} s${pad(season)}`, `${title} season ${season}`];
   306→   306→    for (const q of packQueries) {
   307→   307→      const rows = await fetchJson<ApibayResult[]>(
   308→   308→        `https://apibay.org/q.php?q=${encodeURIComponent(q)}&cat=`,
   309→   309→        7000
   310→   310→      );
   311→   311→      if (rows && Array.isArray(rows)) collect(rows);
   312→   312→      if (seen.size > 0) break;
   313→   313→    }
   314→   314→  }
   315→   315→  return [...seen.values()];
   316→   316→}

   326→   326→/**
   327→   327→ * BitSearch (bitsearch.eu) — HTML search, the last live indexer when the
   328→   328→ * JSON APIs are down or blocked. It indexes Indian release groups
   329→   329→ * (1TamilBlasters, TamilMV, 3MovieRulz…) that upload daily-show episodes
   330→   330→ * ("Bigg Boss (2026) S09E101 …") hours after broadcast. Every row is
   331→   331→ * validated with the same strict matcher as the keyword providers.
   332→   332→ */
   333→   333→async function providerBitSearch(type: "movie" | "tv", ctx: BayContext): Promise<TorrentSource[]> {
   334→   334→  const { title, originalTitle, year, season, episode } = ctx;
   335→   335→  const queries =
   336→   336→    type === "movie"
   337→   337→      ? [`${title} ${year ?? ""}`.trim(), title]
   338→   338→      : [
   339→   339→          `${title} s${pad(season)}e${pad(episode)}`,
   340→   340→          `${title} season ${season}`, // dated daily shows + season packs
   341→   341→        ];
   342→   342→
   343→   343→  const cards = new Map<string, BitsearchCard>();
   344→   344→  const scrape = async (q: string) => {
   345→   345→    try {
   346→   346→      const res = await fetch(`https://bitsearch.eu/search?q=${encodeURIComponent(q)}`, {
   347→   347→        headers: {
   348→   348→          accept: "text/html,application/xhtml+xml",
   349→   349→          "user-agent":
   350→   350→            "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0.0.0 Safari/537.36",
   351→   351→        },
   352→   352→        signal: AbortSignal.timeout(9000),
   353→   353→        redirect: "follow",
   354→   354→      });
   355→   355→      if (!res.ok) return;
   356→   356→      const html = (await res.text()).replace(/\s+/g, " ");
   357→   357→      const re = /\/download\/torrent\/([0-9a-fA-F]{40})\?title=([^">]+)/g;
   358→   358→      for (const m of html.matchAll(re)) {
   359→   359→        const hash = m[1].toLowerCase();
   360→   360→        if (cards.has(hash)) continue;
   361→   361→        let name = m[2];
   362→   362→        try { name = decodeURIComponent(name); } catch { /* raw text fallback */ }
   363→   363→        name = name
   364→   364→          .replace(/&amp;/g, "&")
   365→   365→          .replace(/&#x3D;|&#61;/gi, "=")
   366→   366→          .replace(/&quot;/gi, '"')
   367→   367→          .replace(/&#39;/g, "'")
   368→   368→          .trim();
   369→   369→        const win = html.slice(Math.max(0, m.index - 2200), m.index);
   370→   370→        const seeders = Number(win.match(/font-medium">(\d+)<\/span>\s*<span>\s*seeders/)?.[1] ?? 0);
   371→   371→        const peers = Number(win.match(/font-medium">(\d+)<\/span>\s*<span>\s*leechers/)?.[1] ?? 0);
   372→   372→        const sizeBytes = parseSizeBytes(
   373→   373→          win.match(/fa-download"><\/i>\s*<span>\s*([\d.]+\s*[KMGT]?B)</i)?.[1] ?? null
   374→   374→        );
   375→   375→        cards.set(hash, { hash, name, seeders, peers, sizeBytes });
   376→   376→      }
   377→   377→    } catch {
   378→   378→      // blocked / timed out — the other providers still answer
   379→   379→    }
   380→   380→  };
   381→   381→
   382→   382→  await Promise.all(queries.map(scrape));
   383→   383→
   384→   384→  const out: TorrentSource[] = [];
   385→   385→  for (const c of cards.values()) {
   386→   386→    const p = parseRelease(c.name);
   387→   387→    const match =
   388→   388→      type === "movie"
   389→   389→        ? matchMovie(p, title, originalTitle, year)
   390→   390→        : matchTv(p, season, episode, title, originalTitle, {
   391→   391→            airDate: ctx.airDate,
   392→   392→            showLang: ctx.showLang,
   393→   393→            absEpisode: ctx.absEpisode,
   394→   394→          });
   395→   395→    if (!match) continue; // same name, wrong season/episode/year — reject
   396→   396→    if (type === "tv" && countryPenalty(p, ctx.countries) < 0) continue;
   397→   397→    out.push({
   398→   398→      id: c.hash,
   399→   399→      releaseName: c.name,
   400→   400→      seeders: c.seeders,
   401→   401→      peers: c.peers,
   402→   402→      sizeBytes: c.sizeBytes,
   403→   403→      provider: "BitSearch",
   404→   404→      quality: detectQuality(c.name),
   405→   405→      fileIdx: null,
   406→   406→      match,
   407→   407→      score: 0,
   408→   408→    });
   409→   409→  }
   410→   410→  return out;
   411→   411→}

   570→   570→        (p.episode !== null && p.episode !== episode);
   571→   571→      if (wrongEpisode) continue;

   573→   573→        id: r.hash.toLowerCase(),
   574→   574→        releaseName: filename,
   575→   575→        seeders: r.seeds ?? 0,
   576→   576→        peers: r.peers ?? 0,
   577→   577→        sizeBytes: r.size_bytes ?? 0,

   580→   580→        fileIdx: null,
   581→   581→        match: "verified",
   582→   582→        score: 0,
   583→   583→      });

   586→   586→    if (rows.length < 100) break; // no more pages for this show

   594→   594→  title: string;

   601→   601→/** RSS search on nyaa.si — the primary public index for fansubbed anime. */
   602→   602→async function nyaaSearch(query: string): Promise<NyaaItem[]> {
   603→   603→  try {
   604→   604→    const res = await fetch(
   605→   605→      `https://nyaa.si/?page=rss&f=0&c=0_0&q=${encodeURIComponent(query)}`,
   606→   606→      {
   607→   607→        headers: {
   608→   608→          accept: "application/rss+xml",
   609→   609→          "user-agent": "Mozilla/5.0 (X11; Linux x86_64) Otama/1.0",
   610→   610→        },
   611→   611→        signal: AbortSignal.timeout(9000),
   612→   612→      }
   613→   613→    );
   614→   614→    if (!res.ok) return [];
   615→   615→    const xml = await res.text();
   616→   616→    const items: NyaaItem[] = [];
   617→   617→    for (const chunk of xml.split("<item>").slice(1)) {
   618→   618→      const pick = (re: RegExp) => chunk.match(re)?.[1]?.trim() ?? "";
   619→   619→      const title = pick(/<title>([\s\S]*?)<\/title>/);
   620→   620→      const infoHash = pick(/<nyaa:infoHash>([0-9a-fA-F]{40})<\/nyaa:infoHash>/).toLowerCase();
   621→   621→      if (!title || !infoHash) continue;
   622→   622→      // "1.4 GiB" → "1.4 GB" so the shared size parser understands it
   623→   623→      const sizeRaw = pick(/<nyaa:size>([\d.]+\s*[KMGTP]i?B)<\/nyaa:size>/).replace(/i/g, "");
   624→   624→      items.push({
   625→   625→        title,
   626→   626→        infoHash,
   627→   627→        seeders: parseInt(pick(/<nyaa:seeders>(\d+)<\/nyaa:seeders>/), 10) || 0,
   628→   628→        leechers: parseInt(pick(/<nyaa:leechers>(\d+)<\/nyaa:leechers>/), 10) || 0,
   629→   629→        sizeBytes: parseSizeBytes(sizeRaw),
   630→   630→      });
   631→   631→    }
   632→   632→    return items;
   633→   633→  } catch {
   634→   634→    return [];
   635→   635→  }
   636→   636→}

   638→   638→/** "892-1085" / "1078.5-1085" batch range inside a fansub release name. */
   639→   639→function batchRange(name: string): [number, number] | null {
   640→   640→  const m = name.match(/(\d{1,4})(?:\.\d)?\s*(?:[-–—~]|to)\s*(\d{1,4})\b/i);
   641→   641→  if (!m) return null;
   642→   642→  const a = Number(m[1]);
   643→   643→  const b = Number(m[2]);
   644→   644→  return b > a && b - a <= 400 ? [a, b] : null;
   645→   645→}
   646→   646→
   647→   647→/**
   648→   648→ * Nyaa — fansubbed anime releases. Names use broadcast-order episode numbers
   649→   649→ * ("One Piece - 1085"), season/episode tags ("S02E05") or batch ranges
   650→   650→ * ("Season 20 (892-1085) Batch"), all handled via the strict matcher plus an
   651→   651→ * explicit absolute-episode check. Only enabled for Japanese titles.
   652→   652→ */
   653→   653→async function providerNyaa(type: "movie" | "tv", ctx: BayContext): Promise<TorrentSource[]> {
   654→   654→  const { title, originalTitle, season, episode, absEpisode, isAnime } = ctx;
   655→   655→  if (!isAnime) return [];
   656→   656→  const wanted = absEpisode ?? episode;
   657→   657→  const queries =
   658→   658→    type === "movie"
   659→   659→      ? [`${title} ${ctx.year ?? ""}`.trim(), title, ...(originalTitle ? [originalTitle] : [])]
   660→   660→      : [
   661→   661→          `${title} ${wanted}`,
   662→   662→          ...(originalTitle ? [`${originalTitle} ${wanted}`] : []),
   663→   663→          `${title} s${pad(season)}e${pad(episode)}`,
   664→   664→        ];
   665→   665→
   666→   666→  const seen = new Map<string, TorrentSource>();
   667→   667→  for (const q of queries) {
   668→   668→    if (seen.size >= 6) break;
   669→   669→    const rows = await nyaaSearch(q);
   670→   670→    for (const r of rows) {
   671→   671→      if (seen.has(r.infoHash)) continue;
   672→   672→      const p = parseRelease(r.title);
   673→   673→      let match: MatchClass | null;
   674→   674→      if (type === "movie") {
   675→   675→        match = matchMovie(p, title, originalTitle, ctx.year);
   676→   676→      } else {
   677→   677→        match = matchTv(p, season, episode, title, originalTitle, { absEpisode });
   678→   678→        if (!match) {
   679→   679→          // multi-episode batch: keep it as a pack — the stream route picks
   680→   680→          // the right file inside the torrent by episode number
   681→   681→          const br = batchRange(r.title);
   682→   682→          if (br && wanted >= br[0] && wanted <= br[1]) match = "pack";
   683→   683→        }
   684→   684→      }
   685→   685→      if (!match) continue;
   686→   686→      seen.set(r.infoHash, {
   687→   687→        id: r.infoHash,
   688→   688→        releaseName: r.title,
   689→   689→        seeders: r.seeders,
   690→   690→        peers: r.leechers,
   691→   691→        sizeBytes: r.sizeBytes,
   692→   692→        provider: "Nyaa",
   693→   693→        quality: detectQuality(r.title),
   694→   694→        fileIdx: null,
   695→   695→        match,
   696→   696→        score: 0,
   697→   697→      });
   698→   698→    }
   699→   699→  }
   700→   700→  return [...seen.values()];
   701→   701→}

   703→   703→/* ---------- scoring ---------- */
   704→   704→
   705→   705→export function scoreTorrent(t: TorrentSource): number {
   706→   706→  let s = Math.min(Math.log10(t.seeders + 1) * 4, 13);
   707→   707→  if (t.quality === "1080p") s += 5;
   708→   708→  else if (t.quality === "720p") s += 4;
   709→   709→  else if (t.quality === "2160p") s += 2;
   710→   710→  else if (t.quality === "480p") s += 1;
   711→   711→  const n = t.releaseName.toLowerCase();
   712→   712→  if (/x265|h\.?265|hevc/.test(n)) s -= 4; // HEVC support is spotty in browsers
   713→   713→  if (/av1/.test(n)) s += 1;
   714→   714→  if (/x264|h\.?264|xvid/.test(n)) s += 2;
   715→   715→  if (/\b(cam|ts|telesync|screener|dvdscr|hdts|\btc\b)\b/.test(n)) s -= 12;
   716→   716→  if (/\b3d\b/.test(n)) s -= 6;
   717→   717→  if (/\.mp4\b/.test(n) || /yts/i.test(t.releaseName)) s += 1.5;
   718→   718→  if (t.sizeBytes > 0 && t.sizeBytes < 120 * 1024 * 1024) s -= 2;
   719→   719→  if (t.sizeBytes > 25 * 1024 * 1024 * 1024) s -= 3;
   720→   720→  return Math.round(s * 10) / 10;
   721→   721→}

   723→   723→/** Bonus/penalty applied for how tightly a release matches the request. */
   724→   724→const MATCH_RANK: Record<MatchClass, number> = {
   725→   725→  verified: 8,
   726→   726→  pack: 2.5,
   727→   727→  weak: -6,
   728→   728→};
   729→   729→
   730→   730→function merge(sources: TorrentSource[]): TorrentSource[] {
   731→   731→  const best = new Map<string, TorrentSource>();
   732→   732→  for (const t of sources) {
   733→   733→    const prev = best.get(t.id);
   734→   734→    if (!prev || t.seeders > prev.seeders) best.set(t.id, t);
   735→   735→  }
   736→   736→  const list = [...best.values()];
   737→   737→  for (const t of list) {
   738→   738→    t.score =
   739→   739→      Math.round((scoreTorrent(t) + MATCH_RANK[t.match] + (t.countryPenalty ?? 0)) * 10) / 10;
   740→   740→  }
   741→   741→  // only fall back to unverified (no-year) releases when nothing better exists
   742→   742→  const confident = list.filter((t) => t.match !== "weak");
   743→   743→  const final = confident.length ? confident : list;
   744→   744→  return final.sort((a, b) => b.score - a.score || b.seeders - a.seeders).slice(0, 14);
   745→   745→}

   758→   758→export async function searchTorrentSources(args: SearchArgs): Promise<TorrentSource[]> {
   759→   759→  const { type, tmdbId, title, season, episode } = args;
   760→   760→  const [imdbId, meta] = await Promise.all([getImdbId(type, tmdbId), getTitleMeta(type, tmdbId)]);
   761→   761→  // TMDB's own year is canonical (guards against stale/rounded client data)
   762→   762→  const year = meta.year || args.year || "";
   763→   763→  // Indian daily shows (Bigg Boss…) name episodes by broadcast date and tag
   764→   764→  // language versions — both are needed to match them without mixing content
   765→   765→  const airDate = type === "tv" ? await getEpisodeAirDate(tmdbId, season, episode) : null;
   766→   766→  const showLang = meta.originalLanguage ? LANG_WORD[meta.originalLanguage] ?? null : null;
   767→   767→  // "Bigg Boss Tamil" uploads are named "Bigg Boss (2026) S09E101 [Tamil …]":
   768→   768→  // the language moves into the tag area, so match against the base title and
   769→   769→  // let the release's language tag pick the right version (mismatch = reject)
   770→   770→  const matchTitle =
   771→   771→    type === "tv" && showLang
   772→   772→      ? title.replace(new RegExp(`\\s+${showLang}$`, "i"), "").trim() || title
   773→   773→      : title;
   774→   774→
   775→   775→  // Japanese titles are anime: fansub indexers (Nyaa) get queried and the
   776→   776→  // broadcast-order episode number is offered to the matchers
   777→   777→  const isAnime = (meta.originalLanguage ?? "") === "ja";
   778→   778→  const absEpisode = type === "tv" && isAnime ? await getAbsoluteEpisode(tmdbId, season, episode) : null;
   779→   779→
   780→   780→  const jobs: Promise<TorrentSource[]>[] = [];
   781→   781→  if (matchTitle) {
   782→   782→    const bayCtx: BayContext = {
   783→   783→      title: matchTitle,
   784→   784→      originalTitle: meta.originalTitle ?? undefined,
   785→   785→      year: year || undefined,
   786→   786→      countries: meta.countries,
   787→   787→      season,
   788→   788→      episode,
   789→   789→      airDate,
   790→   790→      showLang,
   791→   791→      absEpisode,
   792→   792→      isAnime,
   793→   793→    };
   794→   794→    jobs.push(providerPirateBay(type, bayCtx));
   795→   795→    jobs.push(providerBitSearch(type, bayCtx));
   796→   796→    jobs.push(providerTorrentsCsv(type, bayCtx));
   797→   797→    if (isAnime) jobs.push(providerNyaa(type, bayCtx));
   798→   798→  }
   799→   799→  if (imdbId) {
   800→   800→    jobs.push(providerTorrentio(type, imdbId, season, episode, year));
   801→   801→    if (type === "movie") jobs.push(providerYts(imdbId, year));
   802→   802→    else jobs.push(providerEztv(imdbId, season, episode));
   803→   803→  }
   804→   804→
   805→   805→  const settled = await Promise.allSettled(jobs);
   806→   806→  const all: TorrentSource[] = [];
   807→   807→  for (const r of settled) if (r.status === "fulfilled") all.push(...r.value);
   808→   808→  return merge(all);
