     1→"use client";
     2→
     3→import { useEffect, useMemo, useState } from "react";
     4→import {
     5→  X,
     6→  ChevronLeft,
     7→  ChevronRight,
     8→  Plus,
     9→  Check,
    10→  List,
    11→  Info,
    12→  Server,
    13→  ExternalLink,
    14→  Loader2,
    15→  AlertTriangle,
    16→  RefreshCw,
    17→  ShieldCheck,
    18→  Zap,
    19→} from "lucide-react";
    20→import {
    21→  MediaDetails,
    22→  MediaItem,
    23→  Episode,
    24→  getTitle,
    25→  getMediaType,
    26→  getYear,
    27→  img,
    28→  tmdb,
    29→  PLAYER_SOURCES,
    30→  AUTO_SOURCE_ID,
    31→  ARCHIVE_SOURCE_ID,
    32→} from "@/lib/tmdb";
    33→import { useMediaStore, useMounted } from "@/hooks/use-media-store";
    34→import { MediaActions } from "./types";
    35→
    36→interface PlayerModalProps {
    37→  target: { item: MediaItem; season?: number; episode?: number } | null;
    38→  actions: MediaActions;
    39→  onClose: () => void;
    40→}
    41→
    42→type HealthState = "checking" | "up" | "popup" | "down";
    43→interface Health {
    44→  state: HealthState;
    45→  ms?: number;
    46→}
    47→type FrameState = "loading" | "ready" | "timeout";
    48→
    49→interface ArchiveMatchClient {
    50→  identifier: string;
    51→  title: string;
    52→  year: number | null;
    53→  creator: string;
    54→  description: string;
    55→  license: string;
    56→  fileUrl: string;
    57→  fileSize: number;
    58→}
    59→
    60→/** How long to wait for the embed iframe before auto-failing-over. */
    61→const LOAD_TIMEOUT_MS = 12000;
    62→
    63→const HEALTH_DOT: Record<HealthState, string> = {
    64→  checking: "bg-zinc-600 animate-pulse",
    65→  up: "bg-emerald-400",
    66→  popup: "bg-amber-400",
    67→  down: "bg-red-500",
    68→};
    69→
    70→const HEALTH_TIP: Record<HealthState, string> = {
    71→  checking: "Checking server…",
    72→  up: "Server is up",
    73→  popup: "Server is up but may block embedding — use Pop out",
    74→  down: "Server not responding right now",
    75→};
    76→
    77→export function PlayerModal({ target, actions, onClose }: PlayerModalProps) {
    78→  // remounted per target via key — initial state is safe to derive from props
    79→  const type = target ? getMediaType(target.item) : "movie";
    80→  const itemId = target?.item.id ?? null;
    81→
    82→  // Auto mode is the default every session: the player health-checks all servers
    83→  // and picks the fastest healthy one (ad-free direct stream first when available).
    84→  // The persisted manual choice is honoured as auto's preferred-first hint, not as
    85→  // a forced pick — so a dead saved server can never break playback again.
    86→  const [selectedId, setSelectedId] = useState<string>(AUTO_SOURCE_ID);
    87→  const [autoPreferred, setAutoPreferred] = useState<string | null>(() => {
    88→    const saved = useMediaStore.getState().sourceId;
    89→    return PLAYER_SOURCES.some((s) => s.id === saved) ? saved : null;
    90→  });
    91→  const [healthMap, setHealthMap] = useState<Record<string, Health>>({});
    92→  const [archive, setArchive] = useState<ArchiveMatchClient | null>(null);
    93→  const [archiveState, setArchiveState] = useState<"loading" | "found" | "none">("loading");
    94→  const [videoError, setVideoError] = useState(false);
    95→  const [autoPickId, setAutoPickId] = useState<string | null>(null);
    96→  const [tried, setTried] = useState<string[]>([]);
    97→  const [banner, setBanner] = useState<string | null>(null);
    98→  const [season, setSeason] = useState(() => target?.season ?? 1);
    99→  const [episode, setEpisode] = useState(() => target?.episode ?? 1);
   100→  const [episodes, setEpisodes] = useState<Episode[]>([]);
   101→  const [epLoading, setEpLoading] = useState(() => type === "tv");
   102→  const [details, setDetails] = useState<MediaDetails | null>(null);
   103→  const [showEpisodes, setShowEpisodes] = useState(false);
   104→  const [frameState, setFrameState] = useState<FrameState>("loading");
   105→  const [reloadNonce, setReloadNonce] = useState(0);
   106→  const mounted = useMounted();
   107→  const addContinue = useMediaStore((s) => s.addContinue);
   108→  const toggleWatchlist = useMediaStore((s) => s.toggleWatchlist);
   109→  const setSourceId = useMediaStore((s) => s.setSourceId);
   110→  const watchlist = useMediaStore((s) => s.watchlist);
   111→
   112→  const nameOf = (id: string) =>
   113→    id === ARCHIVE_SOURCE_ID
   114→      ? "CineX Direct"
   115→      : PLAYER_SOURCES.find((s) => s.id === id)?.name ?? "Server";
   116→
   117→  // Healthy embed servers, fastest first (frameable ones outrank popup-ish ones;
   118→  // the user's preferred server goes first while it stays healthy)
   119→  const rankedHealthy = useMemo(() => {
   120→    const score = (s: Health) => (s.state === "up" ? 0 : s.state === "popup" ? 1 : 2);
   121→    const list = PLAYER_SOURCES.filter((s) => {
   122→      const h = healthMap[s.id];
   123→      return h && (h.state === "up" || h.state === "popup");
   124→    }).sort(
   125→      (a, b) =>
   126→        score(healthMap[a.id]) - score(healthMap[b.id]) ||
   127→        (healthMap[a.id].ms ?? 9e9) - (healthMap[b.id].ms ?? 9e9)
   128→    );
   129→    if (!autoPreferred) return list;
   130→    const prefIdx = list.findIndex((s) => s.id === autoPreferred);
   131→    if (prefIdx < 0) return list;
   132→    const pref = list.splice(prefIdx, 1)[0];
   133→    return [pref, ...list];
   134→  }, [healthMap, autoPreferred]);
   135→
   136→  // Which source is actually on screen right now
   137→  const effectiveId = (() => {
   138→    if (selectedId === AUTO_SOURCE_ID) {
   139→      if (type === "movie" && archiveState === "found" && archive && !videoError)
   140→        return ARCHIVE_SOURCE_ID;
   141→      if (autoPickId) return autoPickId;
   142→      return rankedHealthy[0]?.id ?? PLAYER_SOURCES[0].id;
   143→    }
   144→    if (selectedId === ARCHIVE_SOURCE_ID && archiveState === "found" && !videoError)
   145→      return ARCHIVE_SOURCE_ID;
   146→    return PLAYER_SOURCES.some((s) => s.id === selectedId) ? selectedId : PLAYER_SOURCES[0].id;
   147→  })();
   148→
   149→  // identity of the current embed; resetting to "loading" is done at render time
   150→  const srcKey = `${itemId ?? "none"}-${effectiveId}-${season}-${episode}-${reloadNonce}`;
   151→  const [prevSrcKey, setPrevSrcKey] = useState(srcKey);
   152→  if (prevSrcKey !== srcKey) {
   153→    setPrevSrcKey(srcKey);
   154→    setFrameState("loading");
   155→    setVideoError(false);
   156→  }
   157→
   158→  // failover: move to the next healthy server we haven't tried yet
   159→  const failover = (fromId: string, reason: string) => {
   160→    const next =
   161→      rankedHealthy.map((s) => s.id).find((id) => id !== fromId && !tried.includes(id)) ??
   162→      null;
   163→    if (next) {
   164→      setTried((t) => [...t, fromId]);
   165→      setAutoPickId(next);
   166→      setSelectedId(AUTO_SOURCE_ID);
   167→      if (autoPreferred && autoPreferred !== next) setAutoPreferred(null); // it just proved unreliable
   168→      setBanner(reason.replace("{next}", nameOf(next)));
   169→    }
   170→    return next;
   171→  };
   172→
   173→  // record progress for "Continue Watching" (external system write);
   174→  // re-records as the user navigates seasons/episodes so we resume at the latest spot
   175→  useEffect(() => {
   176→    if (!target || !mounted) return;
   177→    const t = getMediaType(target.item);
   178→    addContinue(
   179→      target.item,
   180→      t === "tv" ? season : undefined,
   181→      t === "tv" ? episode : undefined
   182→    );
   183→  }, [target, mounted, addContinue, season, episode]);
   184→
   185→  // fetch details (for season list)
   186→  useEffect(() => {
   187→    if (!target) return;
   188→    let cancelled = false;
   189→    tmdb<MediaDetails>(`${type}/${target.item.id}`)
   190→      .then((d) => !cancelled && setDetails(d))
   191→      .catch(() => {});
   192→    return () => {
   193→      cancelled = true;
   194→    };
   195→  }, [target, type]);
   196→
   197→  // fetch episodes when season changes (TV only)
   198→  useEffect(() => {
   199→    if (!target || type !== "tv") return;
   200→    let cancelled = false;
   201→    tmdb<{ episodes: Episode[] }>(`tv/${target.item.id}/season/${season}`)
   202→      .then((res) => {
   203→        if (!cancelled) {
   204→          setEpisodes(res.episodes ?? []);
   205→          setEpLoading(false);
   206→        }
   207→      })
   208→      .catch(() => {
   209→        if (!cancelled) {
   210→          setEpisodes([]);
   211→          setEpLoading(false);
   212→        }
   213→      });
   214→    return () => {
   215→      cancelled = true;
   216→    };
   217→  }, [target, type, season]);
   218→
   219→  // probe every server once per title so the strip shows live status dots + latency
   220→  useEffect(() => {
   221→    if (itemId === null) return;
   222→    let cancelled = false;
   223→    PLAYER_SOURCES.forEach((s) => {
   224→      const probe = type === "movie" ? s.movie(itemId) : s.tv(itemId, 1, 1);
   225→      fetch(`/api/embed-check?url=${encodeURIComponent(probe)}`)
   226→        .then((r) => r.json())
   227→        .then((j: { ok?: boolean; ms?: number; frameable?: boolean }) => {
   228→          if (cancelled) return;
   229→          const h: Health = !j.ok
   230→            ? { state: "down", ms: j.ms }
   231→            : j.frameable === false
   232→              ? { state: "popup", ms: j.ms }
   233→              : { state: "up", ms: j.ms };
   234→          setHealthMap((prev) => ({ ...prev, [s.id]: h }));
   235→        })
   236→        .catch(() => {
   237→          if (!cancelled) setHealthMap((prev) => ({ ...prev, [s.id]: { state: "down" } }));
   238→        });
   239→    });
   240→
   241→    // CineX Direct: look for a public-domain copy (movies only)
   242→    if (type === "movie" && target) {
   243→      const q = getTitle(target.item);
   244→      const y = getYear(target.item);
   245→      fetch(`/api/archive?title=${encodeURIComponent(q)}&year=${encodeURIComponent(y)}`)
   246→        .then((r) => r.json())
   247→        .then((j: { match: ArchiveMatchClient | null }) => {
   248→          if (cancelled) return;
   249→          setArchive(j.match);
   250→          setArchiveState(j.match ? "found" : "none");
   251→        })
   252→        .catch(() => {
   253→          if (!cancelled) setArchiveState("none");
   254→        });
   255→    }
   256→    return () => {
   257→      cancelled = true;
   258→    };
   259→  }, [itemId, type]);
   260→
   261→  // watchdog: if the current embed never loads, fail over automatically
   262→  // (the native direct player has its own error/stall handling — skip it there)
   263→  useEffect(() => {
   264→    if (!target || effectiveId === ARCHIVE_SOURCE_ID) return;
   265→    const t = setTimeout(() => {
   266→      setFrameState((s) => {
   267→        if (s !== "loading") return s;
   268→        // auto mode: hop to the next healthy server
   269→        if (selectedId === AUTO_SOURCE_ID) {
   270→          failover(effectiveId, `{name} didn't respond — switched to {next}`);
   271→        } else if (
   272→          rankedHealthy.length &&
   273→          healthMap[selectedId]?.state !== "up" &&
   274→          healthMap[selectedId]?.state !== "checking"
   275→        ) {
   276→          // manual mode on a dead server: switch to auto once per title
   277→          setTried((tt) => [...tt, effectiveId]);
   278→          setAutoPickId(rankedHealthy[0].id);
   279→          setSelectedId(AUTO_SOURCE_ID);
   280→          setBanner(`${nameOf(selectedId)} didn't respond — switched to ${nameOf(rankedHealthy[0].id)}`);
   281→        }
   282→        return s;
   283→      });
   284→    }, LOAD_TIMEOUT_MS);
   285→  }, [srcKey, frameState]);
   286→
   287→  // transient banner auto-dismiss
   288→  useEffect(() => {
   289→    if (!banner) return;
   290→    const t = setTimeout(() => setBanner(null), 7000);
   291→    return () => clearTimeout(t);
   292→  }, [banner]);
   293→
   294→  // persist the preferred server for future sessions (external store write);
   295→  // "auto" when the user hasn't expressed a preference
   296→  useEffect(() => {
   297→    setSourceId(autoPreferred ?? AUTO_SOURCE_ID);
   298→  }, [autoPreferred, setSourceId]);
   299→
   300→  // lock scroll + escape
   301→  useEffect(() => {
   302→    if (!target) return;
   303→    const original = document.body.style.overflow;
   304→    document.body.style.overflow = "hidden";
   305→    const onKey = (ev: KeyboardEvent) => {
   306→      if (ev.key === "Escape") onClose();
   307→    };
   308→    window.addEventListener("keydown", onKey);
   309→    return () => {
   310→      document.body.style.overflow = original;
   311→      window.removeEventListener("keydown", onKey);
   312→    };
   313→  }, [target, onClose]);
   314→
   315→  const seasonsList = (
   316→    details?.seasons ?? []
   317→  ).filter((s) => s.season_number > 0 && s.episode_count > 0);
   318→
   319→  const maxEpisode =
   320→    episodes.length ||
   321→    details?.seasons?.find((s) => s.season_number === season)?.episode_count ||
   322→    1;
   323→
   324→  if (!target) return null;
   325→
   326→  const item = target.item;
   327→  const title = getTitle(item);
   328→  const isDirect = effectiveId === ARCHIVE_SOURCE_ID;
   329→  const currentServer = PLAYER_SOURCES.find((s) => s.id === effectiveId);
   330→  const src = currentServer
   331→    ? type === "movie"
   332→      ? currentServer.movie(item.id)
   333→      : currentServer.tv(item.id, season, episode)
   334→    : "";
   335→
   336→  const goEpisode = (s: number, e: number) => {
   337→    setSeason(s);
   338→    setEpisode(e);
   339→    setShowEpisodes(false);
   340→  };
   341→
   342→  const nextEpisode = () => {
   343→    if (episode < maxEpisode) goEpisode(season, episode + 1);
   344→    else if (seasonsList.some((s) => s.season_number === season + 1)) goEpisode(season + 1, 1);
   345→  };
   346→
   347→  const prevEpisode = () => {
   348→    if (episode > 1) goEpisode(season, episode - 1);
   349→    else if (season > 1) goEpisode(season - 1, 1);
   350→  };
   351→
   352→  const inList = mounted
   353→    ? watchlist.some((w) => w.id === item.id && w.media_type === type)
   354→    : false;
   355→
   356→  const switchSource = (id: string) => {
   357→    if (id === effectiveId) {
   358→      // re-selecting the active source = manual retry
   359→      setFrameState("loading");
   360→      setReloadNonce((n) => n + 1);
   361→      if (id === ARCHIVE_SOURCE_ID) setVideoError(false);
   362→      return;
   363→    }
   364→    setTried((t) => [...t, effectiveId]);
   365→    if (id === ARCHIVE_SOURCE_ID) setVideoError(false);
   366→    setSelectedId(id);
   367→    if (PLAYER_SOURCES.some((s) => s.id === id)) {
   368→      setAutoPreferred(id); // remember the user's preference
   369→      setAutoPickId(id);
   370→    } else if (id === AUTO_SOURCE_ID) {
   371→      setAutoPickId(null); // recompute from live health ranking
   372→    }
   373→  };
   374→
   375→  const retryServer = () => {
   376→    setFrameState("loading");
   377→    setReloadNonce((n) => n + 1);
   378→    if (isDirect) setVideoError(false);
   379→  };
   380→
   381→  const tryNextServer = () => {
   382→    if (selectedId === AUTO_SOURCE_ID || isDirect) {
   383→      if (!failover(effectiveId, `{name} failed — switching to {next}`)) {
   384→        // nothing left to try — cycle manually through the registry
   385→        const i = PLAYER_SOURCES.findIndex((s) => s.id === effectiveId);
   386→        switchSource(PLAYER_SOURCES[(i + 1) % PLAYER_SOURCES.length].id);
   387→      }
   388→    } else {
   389→      const i = PLAYER_SOURCES.findIndex((s) => s.id === selectedId);
   390→      switchSource(PLAYER_SOURCES[(i + 1) % PLAYER_SOURCES.length].id);
   391→    }
   392→  };
   393→
   394→  return (
   395→    <div
   396→      className="fixed inset-0 z-[80] flex flex-col bg-black"
   397→      role="dialog"
   398→      aria-modal="true"
   399→      aria-label={`Playing ${title}`}
   400→    >
   401→      {/* Top bar */}
   402→      <div className="flex items-center justify-between gap-3 px-3 py-2.5 md:px-5">
   403→        <div className="flex min-w-0 items-center gap-3">
   404→          <button
   405→            type="button"
   406→            onClick={onClose}
   407→            aria-label="Close player"
   408→            className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full bg-white/10 text-white transition hover:bg-red-600"
   409→          >
   410→            <X className="h-5 w-5" />
   411→          </button>
   412→          <div className="min-w-0">
   413→            <h2 className="truncate text-sm font-bold text-white md:text-base">
   414→              {title}
   415→              {type === "tv" && (
   416→                <span className="ml-2 font-normal text-zinc-400">
   417→                  S{season}:{episode < 10 ? `0${episode}` : episode}
   418→                </span>
   419→              )}
   420→            </h2>
   421→            {type === "tv" && episodes.find((ep) => ep.episode_number === episode)?.name && (
   422→              <p className="truncate text-xs text-zinc-500">
   423→                {episodes.find((ep) => ep.episode_number === episode)?.name}
   424→              </p>
   425→            )}
   426→          </div>
   427→        </div>
   428→
   429→        <div className="flex shrink-0 items-center gap-2">
   430→          <button
   431→            type="button"
   432→            onClick={() =>
   433→              toggleWatchlist(
   434→                item,
   435→                type === "tv" ? season : undefined,
   436→                type === "tv" ? episode : undefined
   437→              )
   438→            }
   439→            className={`hidden h-9 items-center gap-1.5 rounded-full px-3 text-xs font-semibold sm:flex ${
   440→              inList
   441→                ? "bg-emerald-600/20 text-emerald-400 ring-1 ring-emerald-600/50"
   442→                : "bg-white/10 text-zinc-200 hover:bg-white/20"
   443→            }`}
   444→          >
   445→            {inList ? <Check className="h-3.5 w-3.5" /> : <Plus className="h-3.5 w-3.5" />}
   446→            {inList ? "Saved" : "Save"}
   447→          </button>
   448→          {type === "tv" && (
   449→            <button
   450→              type="button"
   451→              onClick={() => setShowEpisodes((v) => !v)}
   452→              className="flex h-9 items-center gap-1.5 rounded-full bg-white/10 px-3 text-xs font-semibold text-zinc-200 transition hover:bg-white/20 lg:hidden"
   453→            >
   454→              <List className="h-4 w-4" /> Episodes
   455→            </button>
   456→          )}
   457→          <a
   458→            href={isDirect && archive ? archive.fileUrl : src}
   459→            target="_blank"
   460→            rel="noopener noreferrer"
   461→            aria-label="Open stream in a new tab"
   462→            title="Open this server in a new tab (works even when embedding is blocked)"
   463→            className="flex h-9 items-center gap-1.5 rounded-full bg-white/10 px-3 text-xs font-semibold text-zinc-200 transition hover:bg-white/20"
   464→          >
   465→            <ExternalLink className="h-3.5 w-3.5" />
   466→            <span className="hidden sm:inline">Pop out</span>
   467→          </a>
   468→          <button
   469→            type="button"
   470→            onClick={tryNextServer}
   471→            className="flex h-9 items-center gap-1.5 rounded-full bg-white/10 px-3 text-xs font-semibold text-zinc-200 transition hover:bg-white/20"
   472→            aria-label="Switch streaming source"
   473→          >
   474→            <Server className="h-3.5 w-3.5" />
   475→            <span className="hidden sm:inline">
   476→              {isDirect ? "CineX Direct" : selectedId === AUTO_SOURCE_ID ? "Auto" : nameOf(selectedId)}
   477→            </span>
   478→            <span className="sm:hidden">Src</span>
   479→          </button>
   480→        </div>
   481→      </div>
   482→
   483→      {/* Server strip with live health dots + latency */}
   484→      <div className="no-scrollbar flex items-center gap-1.5 overflow-x-auto px-3 pb-2 md:px-5">
   485→        <span className="shrink-0 text-[10px] font-bold uppercase tracking-wider text-zinc-600">
   486→          Server
   487→        </span>
   488→
   489→        {/* Auto (best) */}
   490→        <button
   491→          type="button"
   492→          onClick={() => switchSource(AUTO_SOURCE_ID)}
   493→          title="Automatically picks the fastest healthy server and fails over when one dies"
   494→          className={`flex shrink-0 items-center gap-1.5 rounded-full px-2.5 py-1 text-[11px] font-semibold transition ${
   495→            selectedId === AUTO_SOURCE_ID
   496→              ? "bg-red-600 text-white"
   497→              : "bg-white/8 text-zinc-400 hover:bg-white/15 hover:text-zinc-200"
   498→          }`}
   499→        >
   500→          <Zap className="h-3 w-3" />
   501→          Auto{autoPickId ? ` · ${nameOf(autoPickId)}` : ""}
   502→        </button>
   503→
   504→        {/* CineX Direct — ad-free public-domain player (movies) */}
   505→        {type === "movie" && (
   506→          <button
   507→            type="button"
   508→            onClick={() => archiveState === "found" && switchSource(ARCHIVE_SOURCE_ID)}
   509→            disabled={archiveState !== "found"}
   510→            title={
   511→              archiveState === "found"
   512→                ? "Built-in player, direct public-domain stream — 100% ad-free"
   513→                : archiveState === "loading"
   514→                  ? "Checking archive.org for a public-domain copy…"
   515→                  : "No public-domain copy found for this title"
   516→            }
   517→            className={`flex shrink-0 items-center gap-1.5 rounded-full px-2.5 py-1 text-[11px] font-semibold transition ${
   518→              isDirect
   519→                ? "bg-emerald-600 text-white"
   520→                : archiveState === "found"
   521→                  ? "bg-emerald-600/15 text-emerald-300 ring-1 ring-emerald-600/40 hover:bg-emerald-600/25"
   522→                  : "bg-white/5 text-zinc-600"
   523→            } ${archiveState !== "found" ? "cursor-not-allowed" : ""}`}
   524→          >
   525→            <ShieldCheck className="h-3 w-3" />
