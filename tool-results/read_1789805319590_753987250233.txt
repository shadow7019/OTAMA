    38→    38→export async function cached<T>(key: string, ttlMs: number, fn: () => Promise<T>): Promise<T> {
    39→    39→  const hit = cache.get(key)
    40→    40→  if (hit && Date.now() - hit.ts < ttlMs) return hit.value as T
    41→    41→  const value = await fn()
    42→    42→  cache.set(key, { ts: Date.now(), value })
    43→    43→  return value
    44→    44→}

    48→    48→async function fetchText(url: string, timeoutMs = 12_000): Promise<string> {
    49→    49→  const ctrl = new AbortController()
    50→    50→  const t = setTimeout(() => ctrl.abort(), timeoutMs)
    51→    51→  try {
    52→    52→    const res = await fetch(url, {
    53→    53→      signal: ctrl.signal,
    54→    54→      cache: 'no-store',
    55→    55→      headers: { 'User-Agent': UA, Accept: '*/*' },
    56→    56→    })
    57→    57→    if (!res.ok) throw new Error(`HTTP ${res.status} for ${url}`)
    58→    58→    return await res.text()
    59→    59→  } finally {
    60→    60→    clearTimeout(t)
    61→    61→  }
    62→    62→}

    64→    64→async function fetchJson<T>(url: string, timeoutMs = 12_000): Promise<T> {
    65→    65→  const text = await fetchText(url, timeoutMs)
    66→    66→  if (text.trimStart().startsWith('<')) throw new Error(`Got HTML instead of JSON from ${url}`)
    67→    67→  return JSON.parse(text) as T
    68→    68→}

    78→    78→export async function cfGetText(url: string, timeoutMs = 12_000): Promise<string> {
    79→    79→  const maxTime = Math.ceil(timeoutMs / 1000)
    80→    80→  try {
    81→    81→    const { execFile } = await import('node:child_process')
    82→    82→    const out = await new Promise<string>((resolve, reject) => {
    83→    83→      execFile(
    84→    84→        'curl',
    85→    85→        ['-s', '-L', '--compressed', '--max-time', String(maxTime), '-A', UA, '-H', 'Accept: */*', url],
    86→    86→        { timeout: (maxTime + 2) * 1000, maxBuffer: 8 * 1024 * 1024 },
    87→    87→        (err, stdout) => (err ? reject(err) : resolve(stdout)),
    88→    88→      )
    89→    89→    })
    90→    90→    if (out) return out
    91→    91→  } catch { /* curl missing or failed -> fall back to fetch */ }
    92→    92→  return fetchText(url, timeoutMs)
    93→    93→}

   194→   194→export function humanSize(bytes?: number | string): string {
   195→   195→  const n = typeof bytes === 'string' ? parseInt(bytes, 10) : bytes
   196→   196→  if (!n || isNaN(n)) return ''
   197→   197→  const units = ['B', 'KB', 'MB', 'GB', 'TB']
   198→   198→  let i = 0
   199→   199→  let v = n
   200→   200→  while (v >= 1024 && i < units.length - 1) {
   201→   201→    v /= 1024
   202→   202→    i++
   203→   203→  }
   204→   204→  return `${v.toFixed(v >= 100 || i === 0 ? 0 : 1)} ${units[i]}`
   205→   205→}

   207→   207→const QUALITY_RE = /\b(2160p|1440p|1080p|1080i|720p|576p|480p|4k|uhd|web[ .-]?dl|web[ .-]?rip|brrip|bluray|bdrip|hdrip|dvdrip|hdtv|cam|ts)\b/i

   217→   217→/**
   218→   218→ * Codec detection from a release name. HEVC/x265 releases typically CANNOT be
   219→   219→ * decoded by browsers (Chrome only plays H.265 with hardware support), which
   220→   220→ * is the #1 cause of the "stuck on buffering" report — so OTAMA surfaces and
   221→   221→ * deprioritises them.
   222→   222→ */
   223→   223→export function detectCodecFromName(name: string): 'h264' | 'hevc' | undefined {
   224→   224→  const s = (name || '').toLowerCase()
   225→   225→  if (/\b(x\s?265|h\.?265|hevc)\b/.test(s)) return 'hevc'
   226→   226→  if (/\b(x\s?264|h\.?264|avc)\b/.test(s)) return 'h264'
   227→   227→  return undefined
   228→   228→}
   229→   229→
   230→   230→/**
   231→   231→ * Browser playability of a release name — 0 plays everywhere, higher ranks
   232→   232→ * may not play: 1 MKV/MOV (Chromium-only), 2 AVI/TS (<video> can't demux),
   233→   233→ * 3 HEVC/x265 (browsers cannot decode without hardware support).
   234→   234→ */
   235→   235→export function playabilityRank(name: string, codec?: string): 0 | 1 | 2 | 3 {
   236→   236→  const c = (codec as 'h264' | 'hevc' | undefined) || detectCodecFromName(name)
   237→   237→  if (c === 'hevc') return 3
   238→   238→  const ext = (name.match(/\.(mp4|m4v|mkv|mov|avi|ts|webm)\b/i) || [])[1]?.toLowerCase()
   239→   239→  if (ext === 'avi' || ext === 'ts') return 2 // container not supported by <video>
   240→   240→  if (ext === 'mkv' || ext === 'mov') return 1 // plays in Chromium/Chrome, not Firefox/Safari
   241→   241→  return 0
   242→   242→}
   243→   243→
   244→   244→/**
   245→   245→ * Sort torrent options so that browser-playable releases come first, then by
   246→   246→ * seed count. Keeps the popular x265-HEVC packs from hijacking the top slot.
   247→   247→ */
   248→   248→export function sortTorrentsPlayableFirst(list: TorrentOption[]): TorrentOption[] {
   249→   249→  return [...list].sort((a, b) => {
   250→   250→    const pa = playabilityRank(a.title, a.codec)
   251→   251→    const pb = playabilityRank(b.title, b.codec)
   252→   252→    if (pa !== pb) return pa - pb
   253→   253→    return (b.seeds || 0) - (a.seeds || 0)
   254→   254→  })
   255→   255→}

   302→   302→/** Find movie torrents: Torrentio (multi-site, file-exact) + TPB + YTS + 1337x + SolidTorrents. */
   303→   303→export async function findMovieTorrents(imdbId?: string, title?: string, year?: number): Promise<TorrentOption[]> {
   304→   304→  return cached(`movtorrent:${imdbId || '-'}|${title || '-'}|${year || '-'}`, 10 * 60_000, async () => {
   305→   305→    const { ytsMovieTorrents } = await import('./yts')
   306→   306→    const { leetxSearch } = await import('./leetx')
   307→   307→    const { torrentioStreams } = await import('./torrentio')
   308→   308→    const { solidSearch } = await import('./solidtorrents')
   309→   309→
   310→   310→    const tpbTask = (async () => {
   311→   311→      let rows: TpbItem[] = []
   312→   312→      if (imdbId) {
   313→   313→        try {
   314→   314→          rows = await apibaySearch(imdbId)
   315→   315→        } catch { /* fall through */ }
   316→   316→      }
   317→   317→      if (rows.length === 0 && title) {
   318→   318→        const q = year ? `${title} ${year}` : title
   319→   319→        try {
   320→   320→          rows = await apibaySearch(q)
   321→   321→        } catch { /* fall through */ }
   322→   322→      }
   323→   323→      const videoCats = new Set(['201', '202', '207', '208', '209', '299'])
   324→   324→      return rows
   325→   325→        .filter((r) => videoCats.has(r.categoryCode) || r.imdb)
   326→   326→        .map(apibayToTorrentOption)
   327→   327→    })()
   328→   328→
   329→   329→    const [torrentio, tpb, yts, leetx, solid] = await Promise.all([
   330→   330→      imdbId
   331→   331→        ? torrentioStreams('movie', imdbId).catch(() => [] as TorrentOption[])
   332→   332→        : Promise.resolve([] as TorrentOption[]),
   333→   333→      tpbTask,
   334→   334→      ytsMovieTorrents(imdbId, title, year).catch(() => [] as TorrentOption[]),
   335→   335→      title
   336→   336→        ? leetxSearch(`${title}${year ? ` ${year}` : ''}`, { category: 'movies', resolve: 8 }).catch(() => [] as TorrentOption[])
   337→   337→        : Promise.resolve([] as TorrentOption[]),
   338→   338→      title
   339→   339→        ? solidSearch(`${title}${year ? ` ${year}` : ''}`, { videoOnly: true }).catch(() => [] as TorrentOption[])
   340→   340→        : Promise.resolve([] as TorrentOption[]),
   341→   341→    ])
   342→   342→
   343→   343→    const seen = new Set<string>()
   344→   344→    const merged: TorrentOption[] = []
   345→   345→    for (const t of [...torrentio, ...tpb, ...yts, ...leetx, ...solid]) {
   346→   346→      const key = (t.source || t.hash || '').toLowerCase()
   347→   347→      if (!key || seen.has(key)) continue
   348→   348→      seen.add(key)
   349→   349→      merged.push(t)
   350→   350→    }
   351→   351→    // Browser-playable (H.264/x264, MP4/MKV) releases first, seeds second.
   352→   352→    return sortTorrentsPlayableFirst(merged).slice(0, 40)
   353→   353→  })
   354→   354→}

   362→   362→/**
   363→   363→ * Does a poster URL actually resolve to an image? Cinemeta's metahub serves a
   364→   364→ * small HTML error page (content-type text/html) for titles without artwork,
   365→   365→ * which used to render as an ugly grey "initials" placeholder card.
   366→   366→ */
   367→   367→export async function posterWorks(url: string): Promise<boolean> {
   368→   368→  if (!url) return false
   369→   369→  if (!/^https?:\/\//.test(url)) return true
   370→   370→  const hit = posterCache.get(url)
   371→   371→  if (hit !== undefined) return hit
   372→   372→  let ok = true
   373→   373→  try {
   374→   374→    const res = await fetch(url, {
   375→   375→      method: 'HEAD',
   376→   376→      signal: AbortSignal.timeout(6_000),
   377→   377→      headers: { 'User-Agent': UA },
   378→   378→    })
   379→   379→    const ct = (res.headers.get('content-type') || '').toLowerCase()
   380→   380→    if (!res.ok || !ct.startsWith('image/')) ok = false
   381→   381→  } catch {
   382→   382→    // network hiccup — give the poster the benefit of the doubt
   383→   383→    ok = true
   384→   384→  }
   385→   385→  if (posterCache.size < 5000) posterCache.set(url, ok)
   386→   386→  return ok
   387→   387→}

   440→   440→export function decodeEntities(s: string): string {
   441→   441→  return s
   442→   442→    .replace(/&#(\d+);/g, (_, d) => String.fromCharCode(parseInt(d, 10)))
   443→   443→    .replace(/&#x([0-9a-fA-F]+);/g, (_, h) => String.fromCharCode(parseInt(h, 16)))
   444→   444→    .replace(/&quot;/g, '"')
   445→   445→    .replace(/&apos;/g, "'")
   446→   446→    .replace(/&lt;/g, '<')
   447→   447→    .replace(/&gt;/g, '>')
   448→   448→    .replace(/&amp;/g, '&')
   449→   449→}

   451→   451→function cdata(s: string): string {
   452→   452→  const m = /<!\[CDATA\[([\s\S]*?)\]\]>/.exec(s)
   453→   453→  return m ? m[1] : s
   454→   454→}

   467→   467→export function parseNyaaRss(xml: string): NyaaItem[] {
   468→   468→  const items: NyaaItem[] = []
   469→   469→  const itemRe = /<item>([\s\S]*?)<\/item>/g
   470→   470→  let m: RegExpExecArray | null
   471→   471→  const tag = (block: string, name: string): string => {
   472→   472→    const re = new RegExp(`<${name}[^>]*>([\\s\\S]*?)</${name}>`, 'i')
   473→   473→    const t = re.exec(block)
   474→   474→    return t ? decodeEntities(cdata(t[1]).trim()) : ''
   475→   475→  }
   476→   476→  while ((m = itemRe.exec(xml))) {
   477→   477→    const block = m[1]
   478→   478→    const hash = tag(block, 'nyaa:infoHash').toLowerCase()
   479→   479→    const title = tag(block, 'title')
   480→   480→    if (!hash || !title) continue
   481→   481→    items.push({
   482→   482→      title,
   483→   483→      hash,
   484→   484→      seeds: parseInt(tag(block, 'nyaa:seeders'), 10) || 0,
   485→   485→      leechers: parseInt(tag(block, 'nyaa:leechers'), 10) || 0,
   486→   486→      size: tag(block, 'nyaa:size'),
   487→   487→      date: tag(block, 'pubDate'),
   488→   488→      page: tag(block, 'guid'),
   489→   489→      info: tag(block, 'description'),
   490→   490→    })
   491→   491→  }
   492→   492→  return items
   493→   493→}

   574→   574→export async function seriesDetail(imdbId: string): Promise<{ item: MetaItem & { cast?: string[]; awards?: string }; seasons: { season: number; episodes: EpisodeInfo[] }[] }> {
   575→   575→  const cine = await cineMeta('series', imdbId)
   576→   576→  let item: MetaItem & { cast?: string[]; awards?: string } = cine
   577→   577→  const seasonsMap = new Map<number, EpisodeInfo[]>()
   578→   578→
   579→   579→  // Try TVMaze for reliable episode lists
   580→   580→  try {
   581→   581→    const show = await tvmazeLookupByImdb(imdbId)
   582→   582→    if (show) {
   583→   583→      const shows = await fetchJson<TvMazeShow & { _embedded?: { episodes?: TvMazeEpisode[] } }>(
   584→   584→        `${TVMAZE}/shows/${show.id}?embed=episodes`,
   585→   585→        15_000,
   586→   586→      )
   587→   587→      item = { ...normalizeTvMaze(shows), backdrop: cine.backdrop, summary: cine.summary || normalizeTvMaze(shows).summary, cast: cine.cast, awards: cine.awards }
   588→   588→      for (const ep of shows._embedded?.episodes || []) {
   589→   589→        if (ep.number == null) continue
   590→   590→        const list = seasonsMap.get(ep.season) || []
   591→   591→        list.push({
   592→   592→          season: ep.season,
   593→   593→          episode: ep.number,
   594→   594→          title: ep.name,
   595→   595→          overview: ep.summary ? ep.summary.replace(/<[^>]+>/g, '') : undefined,
   596→   596→          airDate: ep.airdate || undefined,
   597→   597→          thumbnail: ep.image?.medium || undefined,
   598→   598→        })
   599→   599→        seasonsMap.set(ep.season, list)
   600→   600→      }
   601→   601→    }
   602→   602→  } catch {
   603→   603→    // fallback: Cinemeta videos
   604→   604→  }
   605→   605→
   606→   606→  if (seasonsMap.size === 0 && cine.videos?.length) {
   607→   607→    for (const v of cine.videos) {
   608→   608→      if (v.season == null || v.episode == null || v.season < 1) continue
   609→   609→      const list = seasonsMap.get(v.season) || []
   610→   610→      list.push({
   611→   611→        season: v.season,
   612→   612→        episode: v.episode,
   613→   613→        title: v.name || v.title,
   614→   614→        overview: v.overview,
   615→   615→        airDate: v.released,
   616→   616→        thumbnail: v.thumbnail,
   617→   617→      })
   618→   618→      seasonsMap.set(v.season, list)
   619→   619→    }
   620→   620→  }
   621→   621→
   622→   622→  const seasons = [...seasonsMap.entries()]
   623→   623→    .map(([season, episodes]) => ({ season, episodes: episodes.sort((a, b) => a.episode - b.episode) }))
   624→   624→    .sort((a, b) => a.season - b.season)
   625→   625→
   626→   626→  return { item, seasons }
   627→   627→}

   629→   629→/** Episode torrents: Torrentio (multi-site, file-exact) + EZTV + TPB + 1337x + SolidTorrents. */
   630→   630→export async function findEpisodeTorrents(
   631→   631→  imdbId?: string,
   632→   632→  title?: string,
   633→   633→  season?: number,
   634→   634→  episode?: number,
   635→   635→): Promise<TorrentOption[]> {
   636→   636→  // Torrentio first — aggregates EZTV/TPB/1337x/TGX/RARBG/KAT and returns the
   637→   637→  // exact video file index for the episode (fast + fewest dead torrents).
   638→   638→  if (imdbId && /^tt\d+$/.test(imdbId)) {
   639→   639→    try {
   640→   640→      const { torrentioStreams } = await import('./torrentio')
   641→   641→      const tor = await cached(
   642→   642→        `torrentio:ep:${imdbId}:${season ?? ''}:${episode ?? ''}`,
   643→   643→        5 * 60_000,
   644→   644→        () => torrentioStreams('series', imdbId, season, episode),
   645→   645→      )
   646→   646→      if (season && episode) {
   647→   647→        const exact = sortTorrentsPlayableFirst(tor.filter((t) => t.season == null || t.season === season)).slice(0, 15)
   648→   648→        if (exact.length) return exact
   649→   649→      } else {
   650→   650→        const sorted = sortTorrentsPlayableFirst(tor).slice(0, 24)
   651→   651→        if (sorted.length) return sorted
   652→   652→      }
   653→   653→    } catch { /* Torrentio unavailable -> EZTV */ }
   654→   654→  }
   655→   655→
   656→   656→  // whole-show EZTV lookup (cached)
   657→   657→  if (imdbId && /^\d+$/.test(imdbId.replace(/^tt/, ''))) {
   658→   658→    try {
   659→   659→      const list = await cached(`eztv:${imdbId}`, 10 * 60_000, () => eztvByImdb(imdbId.replace(/^tt/, '')))
   660→   660→      const sorted = sortTorrentsPlayableFirst(list)
   661→   661→      if (season && episode) {
   662→   662→        const exact = sorted.filter((t) => t.season === season && t.episode === episode)
   663→   663→        if (exact.length) return exact.slice(0, 12)
   664→   664→      } else if (season) {
   665→   665→        const pack = sorted.filter((t) => t.season === season && (t.episode == null || t.episode === 0))
   666→   666→        if (pack.length) return pack.slice(0, 12)
   667→   667→      } else {
   668→   668→        return sorted.slice(0, 24)
   669→   669→      }
   670→   670→    } catch { /* EZTV unavailable -> TPB */ }
   671→   671→  }
   672→   672→  if (!title) return []
   673→   673→  const pad = (n?: number, w = 2) => (n != null ? String(n).padStart(w, '0') : '')
   674→   674→  const queries: string[] = []
   675→   675→  if (season && episode) {
   676→   676→    queries.push(`${title} S${pad(season)}E${pad(episode)}`)
   677→   677→    queries.push(`${title} ${season}x${pad(episode)}`)
   678→   678→  } else if (season) {
   679→   679→    queries.push(`${title} S${pad(season)}`)
   680→   680→  } else {
   681→   681→    queries.push(title)
   682→   682→  }
   683→   683→  for (const q of queries) {
   684→   684→    try {
   685→   685→      const rows = await apibaySearch(q)
   686→   686→      const tvCats = new Set(['205', '208'])
   687→   687→      const opts = rows
   688→   688→        .filter((r) => tvCats.has(r.categoryCode) || /\bS\d{1,2}E\d{1,2}\b/i.test(r.name))
   689→   689→        .map(apibayToTorrentOption)
   690→   690→      const sorted = sortTorrentsPlayableFirst(opts).slice(0, 12)
   691→   691→      if (sorted.length) return sorted
   692→   692→    } catch { /* try next */ }
   693→   693→  }
   694→   694→  // 1337x fallback (works when TPB/EZTV are blocked)
   695→   695→  if (season && episode) {
   696→   696→    try {
   697→   697→      const { leetxSearch } = await import('./leetx')
   698→   698→      const opts = await leetxSearch(`${title} S${pad(season)}E${pad(episode)}`, { category: 'tv', resolve: 8 })
   699→   699→      if (opts.length) return opts.slice(0, 12)
   700→   700→    } catch { /* next fallback */ }
   701→   701→  }
   702→   702→  // SolidTorrents fallback (DHT index, reachable when trackers are blocked)
   703→   703→  if (season && episode) {
   704→   704→    try {
   705→   705→      const { solidSearch } = await import('./solidtorrents')
   706→   706→      const opts = await solidSearch(`${title} S${pad(season)}E${pad(episode)}`, { videoOnly: true })
   707→   707→      const sorted = sortTorrentsPlayableFirst(opts).slice(0, 12)
   708→   708→      if (sorted.length) return sorted
   709→   709→    } catch { /* give up */ }
   710→   710→  }
   711→   711→  return []
   712→   712→}
