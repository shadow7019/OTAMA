    43→/**
    44→ * A torrent "server" = one ranked source returned by /api/torrent/search
    45→ * (ThePirateBay, Torrentio/1337x, YTS, EZTV …). We download it on our own
    46→ * backend and stream it to the native <video> element — no third-party
    47→ * iframe, therefore no pop-ups, no tab redirects, no injected ads.
    48→ */
    49→interface TorrentServer {
    50→  id: string;
    51→  releaseName: string;
    52→  seeders: number;
    53→  peers: number;
    54→  sizeBytes: number;
    55→  provider: string;
    56→  quality: string;
    57→  fileIdx: number | null;
    58→  /** verified = exact title/episode match · pack = season pack (right file auto-picked) · weak = title-only match */
    59→  match: "verified" | "pack" | "weak";
    60→  score: number;
    61→}
    62→
    63→interface TorrentStatus {
    64→  found: boolean;
    65→  ready?: boolean;
    66→  progress?: number;
    67→  peers?: number;
    68→  speed?: number;
    69→}
    70→
    71→interface ArchiveMatchClient {
    72→  identifier: string;
    73→  title: string;
    74→  year: number | null;
    75→  creator: string;
    76→  description: string;
    77→  license: string;
    78→  fileUrl: string;
    79→  fileSize: number;
    80→}

    84→const NO_PEER_SWITCH_MS = 18000;
    85→/** Switch when peers exist but data never arrives. */
    86→const STALL_SWITCH_MS = 40000;
    87→
    88→const fmtSize = (b: number) =>
    89→  b >= 1e9 ? `${(b / 1e9).toFixed(1)} GB` : b >= 1e6 ? `${Math.round(b / 1e6)} MB` : b > 0 ? `${Math.round(b / 1e3)} KB` : "—";
    90→const fmtSpeed = (bps: number) =>
    91→  bps >= 1e6 ? `${(bps / 1e6).toFixed(1)} MB/s` : `${Math.round(bps / 1e3)} KB/s`;
    92→const fmtSeeders = (n: number) => (n >= 1000 ? `${(n / 1000).toFixed(1)}k` : String(n));

    94→export function PlayerModal({ target, actions, onClose }: PlayerModalProps) {
    95→  // remounted per target via key — initial state is safe to derive from props
    96→  const type = target ? getMediaType(target.item) : "movie";
    97→  const itemId = target?.item.id ?? null;
    98→
    99→  /* ---------------- servers (torrent sources) ---------------- */
   100→  const [servers, setServers] = useState<TorrentServer[]>([]);
   101→  const [serversState, setServersState] = useState<"loading" | "ready" | "error">("loading");
   102→  const [searchNonce, setSearchNonce] = useState(0); // bump = manual re-search
   103→  const [activeIdx, setActiveIdx] = useState<number | null>(null);
   104→  const [manualIdx, setManualIdx] = useState<number | "archive" | "trailer" | null>(null);
   105→  const [tried, setTried] = useState<number[]>([]);
   106→  const [exhausted, setExhausted] = useState(false);
   107→  const [videoState, setVideoState] = useState<"loading" | "playing" | "error">("loading");
   108→  const [needsTap, setNeedsTap] = useState(false);
   109→  const [status, setStatus] = useState<TorrentStatus | null>(null);
   110→  const [videoNonce, setVideoNonce] = useState(0);
   111→  const [banner, setBanner] = useState<string | null>(null);
   112→
   113→  /* ---------------- archive.org direct fallback (movies) ---------------- */
   114→  const [archive, setArchive] = useState<ArchiveMatchClient | null>(null);
   115→  const [archiveState, setArchiveState] = useState<"loading" | "found" | "none">("loading");
   116→
   117→  /* ---------------- TV navigation ---------------- */
   118→  const [season, setSeason] = useState(() => target?.season ?? 1);
   119→  const [episode, setEpisode] = useState(() => target?.episode ?? 1);
   120→  const [episodes, setEpisodes] = useState<Episode[]>([]);
   121→  const [epLoading, setEpLoading] = useState(() => type === "tv");
   122→  const [details, setDetails] = useState<MediaDetails | null>(null);
   123→  const [showEpisodes, setShowEpisodes] = useState(false);
   124→
   125→  const mounted = useMounted();
   126→  const addContinue = useMediaStore((s) => s.addContinue);
   127→  const toggleWatchlist = useMediaStore((s) => s.toggleWatchlist);
   128→  const watchlist = useMediaStore((s) => s.watchlist);
   129→
   130→  // refs so callbacks/effects always read the latest values
   131→  const serversRef = useRef(servers);
   132→  serversRef.current = servers;
   133→  const activeIdxRef = useRef(activeIdx);
   134→  activeIdxRef.current = activeIdx;
   135→  const triedRef = useRef(tried);
   136→  triedRef.current = tried;
   137→  const videoStateRef = useRef(videoState);
   138→  videoStateRef.current = videoState;
   139→  const streamStartRef = useRef(0);
   140→  const currentIdRef = useRef<string | null>(null);
   141→  const videoRef = useRef<HTMLVideoElement | null>(null);
   142→
   143→  // When the browser blocks autoplay we surface a tap-to-play overlay instead
   144→  // of a stuck black frame (video data is already buffered at that point).
   145→  const handleCanPlay = (e: React.SyntheticEvent<HTMLVideoElement>) => {
   146→    setVideoState((s) => (s === "loading" ? "playing" : s));
   147→    const v = e.currentTarget;
   148→    if (v.paused) {
   149→      v.play().then(() => setNeedsTap(false)).catch(() => setNeedsTap(true));
   150→    }
   151→  };
   152→
   153→  const activeServer = activeIdx !== null ? servers[activeIdx] ?? null : null;
   154→  const isArchive = manualIdx === "archive" && archiveState === "found" && !!archive;
   155→  const showTrailer = manualIdx === "trailer";
   156→
   157→  const trailer: Video | null = useMemo(() => (details ? trailerOf(details) : null), [details]);
   158→
   159→  const stopId = useCallback((id: string | null) => {
   160→    if (!id) return;
   161→    fetch("/api/torrent/stop", {
   162→      method: "POST",
   163→      headers: { "Content-Type": "application/json" },
   164→      body: JSON.stringify({ id }),
   165→      keepalive: true,
   166→    }).catch(() => {});
   167→  }, []);
   168→
   169→  const serverLabel = useCallback((s: TorrentServer) => `${s.quality} · ${s.provider}`, []);
   170→
   171→  /* ---------------- failover ---------------- */
   172→  const failover = useCallback(
   173→    (reason: string) => {
   174→      const cur = activeIdxRef.current;
   175→      const list = serversRef.current;
   176→      const next = list.findIndex((_, i) => i !== cur && !triedRef.current.includes(i));
   177→      const triedNext = new Set(triedRef.current);
   178→      if (cur !== null) triedNext.add(cur);
   179→      triedRef.current = [...triedNext];
   180→      setTried([...triedNext]);
   181→      const failed = cur !== null ? serversRef.current[cur] : null;
   182→      if (failed) stopId(failed.id);
   183→      if (next >= 0) {
   184→        setActiveIdx(next);
   185→        setVideoState("loading");
   186→        setStatus(null);
   187→        streamStartRef.current = Date.now();
   188→        setBanner(`${reason} — switching to server ${next + 1} (${serverLabel(list[next])})`);
   189→      } else {
   190→        setExhausted(true);
   191→        setBanner(reason);
   192→      }
   193→    },
   194→    [stopId, serverLabel]
   195→  );
   196→
   197→  /* ---------------- load servers for this title/episode ---------------- */
   198→  useEffect(() => {
   199→    if (!target) return;
   200→    let cancelled = false;
   201→    setServers([]);
   202→    setServersState("loading");
   203→    setTried([]);
   204→    triedRef.current = [];
   205→    setActiveIdx(null);
   206→    setManualIdx(null);
   207→    setExhausted(false);
   208→    setVideoState("loading");
   209→    setStatus(null);
   210→    setBanner(null);
   211→
   212→    const params = new URLSearchParams({
   213→      type,
   214→      id: String(target.item.id),
   215→      title: getTitle(target.item),
   216→      year: getYear(target.item),
   217→    });
   218→    if (type === "tv") {
   219→      params.set("season", String(season));
   220→      params.set("episode", String(episode));
   221→    }
   222→    fetch(`/api/torrent/search?${params.toString()}`)
   223→      .then((r) => r.json())
   224→      .then((j: { results?: TorrentServer[] }) => {
   225→        if (cancelled) return;
   226→        setServers(Array.isArray(j.results) ? j.results : []);
   227→        setServersState("ready");
   228→      })
   229→      .catch(() => {
   230→        if (!cancelled) setServersState("error");
   231→      });
   232→
   233→    if (type === "movie") {
   234→      setArchiveState("loading");
   235→      fetch(
   236→        `/api/archive?title=${encodeURIComponent(getTitle(target.item))}&year=${encodeURIComponent(getYear(target.item))}`
   237→      )
   238→        .then((r) => r.json())
   239→        .then((j: { match: ArchiveMatchClient | null }) => {
   240→          if (cancelled) return;
   241→          setArchive(j.match);
   242→          setArchiveState(j.match ? "found" : "none");
   243→        })
   244→        .catch(() => {
   245→          if (!cancelled) setArchiveState("none");
   246→        });
   247→    } else {
   248→      setArchive(null);
   249→      setArchiveState("none");
   250→    }
   251→    return () => {
   252→      cancelled = true;
   253→    };
   254→  }, [target, type, season, episode, searchNonce]);
   255→
   256→  /* ---------------- auto-pick the best untried server ---------------- */
   257→  useEffect(() => {
   258→    if (serversState !== "ready") return;
   259→    if (manualIdx !== null || activeIdx !== null || exhausted) return;
   260→    const idx = servers.findIndex((_, i) => !tried.includes(i));
   261→    if (idx >= 0) {
   262→      setActiveIdx(idx);
   263→      setVideoState("loading");
   264→      streamStartRef.current = Date.now();
   265→    } else {
   266→      setExhausted(true);
   267→    }
   268→  }, [serversState, servers, tried, activeIdx, manualIdx, exhausted]);
   269→
   270→  /* ---------------- stream URL ---------------- */
   271→  const streamUrl = useMemo(() => {
   272→    if (!activeServer || !target) return null;
   273→    const file = activeServer.fileIdx != null ? `&file=${activeServer.fileIdx}` : "";
   274→    // season/episode steer file selection inside season packs and year
   275→    // steers multi-year movie collections — so the exact requested content
   276→    // plays instead of the pack's biggest file
   277→    const hints =
   278→      type === "tv"
   279→        ? `&season=${season}&episode=${episode}`
   280→        : `&year=${getYear(target.item)}`;
   281→    return `/api/torrent/stream?id=${activeServer.id}&dn=${encodeURIComponent(activeServer.releaseName)}${file}${hints}`;
   282→  }, [activeServer, type, season, episode, target?.item]);
   283→
   284→  useEffect(() => {
   285→    streamStartRef.current = Date.now();
   286→    setStatus(null);
   287→    setNeedsTap(false);
   288→  }, [streamUrl, videoNonce]);
   289→
   290→  /* ---------------- live status + watchdog failover ---------------- */
   291→  useEffect(() => {
   292→    if (!streamUrl || !activeServer) {
   293→      setStatus(null);
   294→      return;
   295→    }
   296→    let cancelled = false;
   297→    const tick = async () => {
   298→      if (cancelled) return;
   299→      try {
   300→        const r = await fetch(`/api/torrent/status?id=${activeServer.id}`);
   301→        const j: TorrentStatus = await r.json();
   302→        if (cancelled) return;
   303→        setStatus(j);
   304→        if (j.found === false) {
   305→          failover("This server dropped the stream");
   306→          return;
   307→        }
   308→        if (videoStateRef.current === "playing") return;
   309→        const elapsed = Date.now() - streamStartRef.current;
   310→        if (elapsed > NO_PEER_SWITCH_MS && (j.peers ?? 0) === 0 && (j.progress ?? 0) === 0) {
   311→          failover("No peers on this server");
   312→        } else if (elapsed > STALL_SWITCH_MS && videoStateRef.current === "loading") {
   313→          // data may be flowing but the wrong pieces — this server can't start in time
   314→          failover("This server is too slow to start");
   315→        }
   316→      } catch {
   317→        /* transient — next tick retries */
   318→      }
   319→    };
   320→    tick();
   321→    const iv = setInterval(tick, 2000);
   322→    return () => {
   323→      cancelled = true;
   324→      clearInterval(iv);
   325→    };
   326→  }, [streamUrl, activeServer, failover]);
   327→
   328→  /* ---------------- stop torrents on switch / unmount ---------------- */
   329→  useEffect(() => {
   330→    const id = activeServer?.id ?? null;
   331→    if (currentIdRef.current && currentIdRef.current !== id) stopId(currentIdRef.current);
   332→    currentIdRef.current = id;
   333→  }, [activeServer?.id, stopId]);
   334→  useEffect(() => {
   335→    return () => {
   336→      if (currentIdRef.current) stopId(currentIdRef.current);
   337→    };
   338→  }, [stopId]);
   339→
   340→  /* ---------------- record progress for Continue Watching ---------------- */
   341→  useEffect(() => {
   342→    if (!target || !mounted) return;
   343→    const t = getMediaType(target.item);
   344→    addContinue(target.item, t === "tv" ? season : undefined, t === "tv" ? episode : undefined);
   345→  }, [target, mounted, addContinue, season, episode]);
   346→
   347→  /* ---------------- details (season list + trailer fallback) ---------------- */
   348→  useEffect(() => {
   349→    if (!target) return;
   350→    let cancelled = false;
   351→    tmdb<MediaDetails>(`${type}/${target.item.id}`)
   352→      .then((d) => !cancelled && setDetails(d))
   353→      .catch(() => {});
   354→    return () => {
   355→      cancelled = true;
   356→    };
   357→  }, [target, type]);
   358→
   359→  useEffect(() => {
   360→    if (!target || type !== "tv") return;
   361→    let cancelled = false;
   362→    tmdb<{ episodes: Episode[] }>(`tv/${target.item.id}/season/${season}`)
   363→      .then((res) => {
   364→        if (!cancelled) {
   365→          setEpisodes(res.episodes ?? []);
   366→          setEpLoading(false);
   367→        }
   368→      })
   369→      .catch(() => {
   370→        if (!cancelled) {
   371→          setEpisodes([]);
   372→          setEpLoading(false);
   373→        }
   374→      });
   375→    return () => {
   376→      cancelled = true;
   377→    };
   378→  }, [target, type, season]);
   379→
   380→  /* ---------------- auto-refresh episodes while open ----------------
   381→   * Daily shows (Bigg Boss…) publish a new episode every day: re-pull the
   382→   * season when the tab regains focus and every few minutes, so the latest
   383→   * episode appears on its own — no reload needed. */
   384→  const epRefreshRef = useRef(0);
   385→  useEffect(() => {
   386→    if (!target || type !== "tv") return;
   387→    const refresh = () => {
   388→      if (Date.now() - epRefreshRef.current < 90_000) return; // throttle
   389→      epRefreshRef.current = Date.now();
   390→      tmdb<{ episodes: Episode[] }>(`tv/${target.item.id}/season/${season}`, { _fresh: 1 })
   391→        .then((res) => setEpisodes(res.episodes ?? []))
   392→        .catch(() => {});
   393→    };
   394→    const onVis = () => {
   395→      if (document.visibilityState === "visible") refresh();
   396→    };
   397→    const iv = setInterval(onVis, 3 * 60_000);
   398→    document.addEventListener("visibilitychange", onVis);
   399→    return () => {
   400→      clearInterval(iv);
   401→      document.removeEventListener("visibilitychange", onVis);
   402→    };
   403→  }, [target, type, season]);
   404→
   405→  /* ---------------- banner auto-dismiss ---------------- */
   406→  useEffect(() => {
   407→    if (!banner) return;
   408→    const t = setTimeout(() => setBanner(null), 7000);
   409→    return () => clearTimeout(t);
   410→  }, [banner]);
   411→
   412→  /* ---------------- scroll lock + escape ---------------- */
   413→  useEffect(() => {
   414→    if (!target) return;
   415→    const original = document.body.style.overflow;
   416→    document.body.style.overflow = "hidden";
   417→    const onKey = (ev: KeyboardEvent) => {
   418→      if (ev.key === "Escape") onClose();
   419→    };
   420→    window.addEventListener("keydown", onKey);
   421→    return () => {
   422→      document.body.style.overflow = original;
   423→      window.removeEventListener("keydown", onKey);
   424→    };
   425→  }, [target, onClose]);
   426→
   427→  const seasonsList = (details?.seasons ?? []).filter(
   428→    (s) => s.season_number > 0 && s.episode_count > 0
   429→  );
   430→  const maxEpisode =
   431→    episodes.length ||
   432→    details?.seasons?.find((s) => s.season_number === season)?.episode_count ||
   433→    1;
   434→
   435→  /* -------- episode freshness (daily shows like Bigg Boss) -------- */
   436→  const now = new Date();
   437→  const todayYmd = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, "0")}-${String(
   438→    now.getDate()
   439→  ).padStart(2, "0")}`;
   440→  const epAired = (ep: Episode) => !!ep.air_date && ep.air_date <= todayYmd;
   441→  const latestAired = useMemo(() => {
   442→    for (let i = episodes.length - 1; i >= 0; i--) {
   443→      if (epAired(episodes[i])) return episodes[i];
   444→    }
   445→    return null;
   446→  }, [episodes]);
   447→  const currentEpisode = episodes.find((ep) => ep.episode_number === episode);
   448→
   449→  if (!target) return null;
   450→
   451→  const item = target.item;
   452→  const title = getTitle(item);
   453→
   454→  const goEpisode = (s: number, e: number) => {
   455→    setSeason(s);
   456→    setEpisode(e);
   457→    setShowEpisodes(false);
   458→  };
   459→  const nextEpisode = () => {
   460→    // never step into episodes that haven't aired yet (TMDB publishes
   461→    // placeholders days ahead for daily shows)
   462→    const cap = latestAired?.episode_number ?? maxEpisode;
   463→    if (episode < cap) goEpisode(season, episode + 1);
   464→    else if (seasonsList.some((s) => s.season_number === season + 1)) goEpisode(season + 1, 1);
   465→  };
   466→  const prevEpisode = () => {
   467→    if (episode > 1) goEpisode(season, episode - 1);
   468→    else if (season > 1) goEpisode(season - 1, 1);
   469→  };
   470→
   471→  const inList = mounted
   472→    ? watchlist.some((w) => w.id === item.id && w.media_type === type)
   473→    : false;
   474→
   475→  /* ---------------- user actions ---------------- */
   476→  const selectTorrent = (idx: number) => {
   477→    if (idx === activeIdx) {
   478→      // re-selecting the active server = manual retry
   479→      setVideoState("loading");
   480→      setVideoNonce((n) => n + 1);
   481→      return;
   482→    }
   483→    setManualIdx(idx);
   484→    setExhausted(false);
   485→    setVideoState("loading");
   486→    setActiveIdx(idx);
   487→    streamStartRef.current = Date.now();
   488→  };
   489→
   490→  const selectAuto = () => {
   491→    setManualIdx(null);
   492→    setExhausted(false);
   493→    setTried([]);
   494→    triedRef.current = [];
   495→    setActiveIdx(null); // auto-pick effect chooses the best server again
   496→  };
   497→
   498→  const selectArchive = () => {
   499→    if (archiveState !== "found" || !archive) return;
   500→    if (manualIdx === "archive") {
   501→      setVideoNonce((n) => n + 1);
   502→      return;
   503→    }
   504→    if (activeServer) stopId(activeServer.id);
   505→    setManualIdx("archive");
   506→    setExhausted(false);
   507→    setVideoState("loading");
   508→  };
   509→
   510→  const playTrailer = () => {
   511→    if (!trailer) return;
   512→    if (activeServer) stopId(activeServer.id);
   513→    setManualIdx("trailer");
   514→    setExhausted(false);
   515→  };
   516→
   517→  const retryAll = () => {
   518→    setTried([]);
   519→    triedRef.current = [];
   520→    setExhausted(false);
   521→    setManualIdx(null);
   522→    setActiveIdx(null);
   523→    setVideoState("loading");
   524→    setBanner(null);
   525→  };
   526→
   527→  const retryCurrent = () => {
   528→    setExhausted(false);
   529→    setVideoState("loading");
   530→    setVideoNonce((n) => n + 1);
   531→  };
   532→
   533→  /* ---------------- derived render state ---------------- */
   534→  const noSources =
   535→    (serversState === "ready" && servers.length === 0) || serversState === "error";
   536→  // exhausted also becomes true when zero servers exist — in that case only
   537→  // the no-source panel may show, never the "servers failed" panel too
   538→  const showFailPanel =
   539→    !noSources &&
   540→    ((exhausted && !isArchive && !showTrailer) ||
   541→      (videoState === "error" && exhausted && !isArchive && !showTrailer));
   542→  const showNoSourcePanel =
   543→    noSources && !isArchive && !showTrailer && archiveState !== "found";
   544→
   545→  const downloading = status && (status.progress ?? 0) < 0.999;
   546→  const videoKey = `${streamUrl ?? "none"}-${videoNonce}`;
   547→
   548→  return (
   549→    <div
   550→      className="fixed inset-0 z-[80] flex flex-col bg-black"
   551→      role="dialog"
   552→      aria-modal="true"
   553→      aria-label={`Playing ${title}`}
   554→    >
   555→      {/* Top bar */}
   556→      <div className="flex items-center justify-between gap-3 px-3 py-2.5 md:px-5">
   557→        <div className="flex min-w-0 items-center gap-3">
   558→          <button
   559→            type="button"
