    44→    44→/**
    45→    45→ * Audio-language / dub tags parsed from a release name. These power the
    46→    46→ * player's server categorisation so users can pick Subbed / English / Hindi /
    47→    47→ * Tamil… servers instead of digging through release names.
    48→    48→ */
    49→    49→const AUDIO_TAG_PATTERNS: [string, RegExp][] = [
    50→    50→  ["dual", /\bdual[\s._-]*(?:audio|aac|ac3)?\b|\bdualaudio\b|\bdual-audio\b/],
    51→    51→  ["multi", /\bmulti(?:ple)?[\s._-]*(?:audio|subs?|dubs?)?\b(?!.*\bsingle\b)/],
    52→    52→  ["hindi", /\bhindi\b|\bhin\b/],
    53→    53→  ["tamil", /\btamil\b|\btam\b/],
    54→    54→  ["telugu", /\btelugu\b|\btel\b/],
    55→    55→  ["malayalam", /\bmalayalam\b|\bmal\b/],
    56→    56→  ["kannada", /\bkannada\b|\bkan\b/],
    57→    57→  ["bengali", /\bbengali\b|\bbangla\b|\bben\b/],
    58→    58→  ["marathi", /\bmarathi\b/],
    59→    59→  ["punjabi", /\bpunjabi\b|\bpa\b/],
    60→    60→  ["chi", /\bmandarin\b|\bcantonese\b|\bchinese\b|\bchi\b|\bzho\b/],
    61→    61→  ["eng", /\b(?:eng|english)[\s._-]*(?:dub|audio|aac|ac3|eac3|ddp?5?|2\.0|5\.1)\b|\bdubbed\b(?!.*\bjapanese?\b)|\bfandub\b|\beng?\b(?=.*\bdubs?\b)/],
    62→    62→  ["eng", /\bdublado\b|\benglish[\s._-]*dubs?\b/],
    63→    63→  ["spa", /\b(?:latino|latino|espanol|spanish|castellano)\b/],
    64→    64→  ["por", /\b(?:portugues|portuguesa|pt[\s._-]*br)\b/],
    65→    65→  ["jap", /\bjapanese?\b|\bjpn?\b(?![^.]*\bsub)/],
    66→    66→  ["raw", /\braw\b/],
    67→    67→];
    68→    68→
    69→    69→/**
    70→    70→ * Tags for one release. Japanese-audio-with-subs (the default anime shape:
    71→    71→ * fansub groups, "Eng-Subs", "MultiSub") lands on "sub"; anything with a
    72→    72→ * dub/audio tag lands on its language; "dual"/"multi" are additive so a
    73→    73→ * "Dual-Audio Eng/Jap" release matches both language filters.
    74→    74→ */
    75→    75→export function detectAudioTags(releaseName: string, isAnime: boolean): string[] {
    76→    76→  const n = releaseName.toLowerCase();
    77→    77→  const tags = new Set<string>();
    78→    78→  for (const [tag, re] of AUDIO_TAG_PATTERNS) {
    79→    79→    if (re.test(n)) tags.add(tag);
    80→    80→  }
    81→    81→  // "Eng-Subs" / "MultiSub" mean subtitles, not a dub — remove the dub tag
    82→    82→  // the generic \beng\b pattern may have picked up from them
    83→    83→  const subsOnly = /\b(?:eng|english)[\s._-]*subs?\b|\bmultisubs?\b|\bvost(?:fr|en)?\b|\bsub ?(?:eng|es|indo)?\b/.test(n);
    84→    84→  if (subsOnly && !/\b(?:dub|dubbed|dublado|dual)\b/.test(n)) tags.delete("eng");
    85→    85→  // "Hindi Dubbed" dubs INTO Hindi — the generic \bdubbed\b tag must not add
    86→    86→  // English unless the name actually mentions it
    87→    87→  const indianTag = ["hindi", "tamil", "telugu", "malayalam", "kannada", "bengali", "marathi", "punjabi", "spa", "por", "chi"].find((t) => tags.has(t));
    88→    88→  if (tags.has("eng") && indianTag && !/\b(?:eng|english)\b/.test(n)) tags.delete("eng");
    89→    89→  if (tags.has("dual") || tags.has("multi")) {
    90→    90→    // a dual/multi-audio release carries both the original and a dub
    91→    91→    if (isAnime) tags.add("jap");
    92→    92→    if (/\b(?:eng|english)\b/.test(n) || tags.has("eng")) tags.add("eng");
    93→    93→  }
    94→    94→  // "[Multi-Dub]" on anime = several dubs, English always among them
    95→    95→  if (isAnime && /\bmulti[\s._-]*dubs?\b/.test(n)) tags.add("eng");
    96→    96→  if (tags.size === 0) {
    97→    97→    // anime fansub defaults to subbed japanese; western releases unknown
    98→    98→    if (isAnime) return ["sub", "jap"];
    99→    99→    return ["unknown"];
   100→   100→  }
   101→   101→  if (isAnime && !tags.has("eng") && !indianTag &&
   102→   102→      !tags.has("dual") && !tags.has("multi") &&
   103→   103→      !tags.has("raw")) {
   104→   104→    tags.add("sub"); // japanese audio + subs is the anime default
   105→   105→    tags.add("jap");
   106→   106→  }
   107→   107→  return [...tags];
   108→   108→}

   112→   112→function parseCount(raw: string | undefined | null): number {
   113→   113→  if (!raw) return 0;
   114→   114→  const m = raw.replace(/,/g, "").trim().match(/^([\d.]+)\s*([KM])?$/i);
   115→   115→  if (!m) return 0;
   116→   116→  const n = parseFloat(m[1]);
   117→   117→  const mult = m[2]?.toUpperCase() === "K" ? 1e3 : m[2]?.toUpperCase() === "M" ? 1e6 : 1;
   118→   118→  return Math.round((isFinite(n) ? n : 0) * mult);
   119→   119→}
   120→   120→
   121→   121→function parseSizeBytes(raw: string | undefined | null): number {
   122→   122→  if (!raw) return 0;
   123→   123→  const m = raw.replace(/,/g, "").match(/([\d.]+)\s*(B|KB|MB|GB|TB)/i);
   124→   124→  if (!m) return 0;
   125→   125→  const n = parseFloat(m[1]);
   126→   126→  const unit = m[2].toUpperCase();
   127→   127→  const mult = unit === "TB" ? 1e12 : unit === "GB" ? 1e9 : unit === "MB" ? 1e6 : unit === "KB" ? 1e3 : 1;
   128→   128→  return Math.round((isFinite(n) ? n : 0) * mult);
   129→   129→}

   131→   131→function detectQuality(name: string): string {
   132→   132→  const n = name.toLowerCase();
   133→   133→  if (/\b(2160p|4k)\b/.test(n)) return "2160p";
   134→   134→  if (/\b1080p\b/.test(n)) return "1080p";
   135→   135→  if (/\b720p\b/.test(n)) return "720p";
   136→   136→  if (/\b480p\b|\bdvdrip\b|\bdvd\b/.test(n)) return "480p";
   137→   137→  if (/\b(360p|sd)\b/.test(n)) return "SD";
   138→   138→  return "unknown";
   139→   139→}

   141→   141→async function fetchJson<T>(url: string, timeoutMs = 8000): Promise<T | null> {

   146→   146→    });

   154→   154→/* ---------- strict validation ---------- */
   155→   155→
   156→   156→/**
   157→   157→ * Full strict match for IMDb-keyed provider rows (Torrentio). The plain
   158→   158→ * sanity check above is not enough for TV: releases with NO season/episode
   159→   159→ * markers in the name all pass it, so same-franchise uploads that share the
   160→   160→ * IMDb entry ("Bleach 001~366" answering a Thousand-Year Blood War request)
   161→   161→ * were slipping in as "verified". Every TV row now runs the same matchTv
   162→   162→ * rules as the keyword providers; only a completely unmarked, clean-title
   163→   163→ * name may still fall back to "weak".
   164→   164→ */
   165→   165→function strictImdbMatch(
   166→   166→  p: ParsedRelease,
   167→   167→  type: "movie" | "tv",
   168→   168→  title: string,
   169→   169→  originalTitle: string | undefined,
   170→   170→  year: string,
   171→   171→  season: number,
   172→   172→  episode: number,
   173→   173→  ctx: Pick<BayContext, "airDate" | "showLang" | "absEpisode" | "relEpisode" | "isAnime">,
   174→   174→  pubDate?: number | null
   175→   175→): MatchClass | null {
   176→   176→  if (type === "movie") return matchMovie(p, title, originalTitle, year);
   177→   177→  const m = matchTv(p, season, episode, title, originalTitle, {
   178→   178→    airDate: ctx.airDate,
   179→   179→    showLang: ctx.showLang,
   180→   180→    absEpisode: ctx.absEpisode,
   181→   181→    relEpisode: ctx.relEpisode,
   182→   182→    pubDate,
   183→   183→  });
   184→   184→  if (m) return m;
   185→   185→  // anime multi-episode batches: shared range check with the Nyaa provider
   186→   186→  if (ctx.isAnime) {
   187→   187→    return animeBatchMatch(p, p.titleTokens.join(" "), {
   188→   188→      absEpisode: ctx.absEpisode,
   189→   189→      relEpisode: ctx.relEpisode,
   190→   190→      episode,
   191→   191→    }, title, originalTitle);
   192→   192→  }
   193→   193→  // last resort for TV: a release with no season/episode/date markers at all
   194→   194→  // and a perfectly clean title ("Show.Name.1080p.WEB-DL") stays reachable
   195→   195→  // but demoted to weak — it may be any episode of the show
   196→   196→  const noMarkers =
   197→   197→    p.season === null &&
   198→   198→    p.episode === null &&
   199→   199→    !p.isSeasonPack &&
   200→   200→    !p.seasonRange &&
   201→   201→    !p.date &&
   202→   202→    !p.titleTokens.some((t) => /^\d+$/.test(t));
   203→   203→  if (noMarkers) {
   204→   204→    const { ratio, unmatched } = bestTitleMatch(p, [title, originalTitle].filter(Boolean) as string[]);
   205→   205→    if (ratio >= 0.99 && unmatched.length === 0) return "weak";
   206→   206→  }
   207→   207→  return null;
   208→   208→}
   209→   209→
   210→   210→/**
   211→   211→ * Shared anime batch-range acceptance (used by the Nyaa provider and the
   212→   212→ * strict IMDb match). A batch "Show - 001 ~ 366" answers an episode request
   213→   213→ * when the requested number falls inside the range — with one guard: a
   214→   214→ * SEQUEL-named batch ("Thousand-Year Blood War S01E01-06") must never answer
   215→   215→ * a base-season request, because its range is the sequel's own numbering.
   216→   216→ */
   217→   217→function animeBatchMatch(
   218→   218→  p: ParsedRelease,
   219→   219→  releaseName: string,
   220→   220→  ctx: { absEpisode?: number | null; relEpisode?: number | null; episode: number },
   221→   221→  title: string,
   222→   222→  originalTitle: string | undefined
   223→   223→): MatchClass | null {
   224→   224→  const br = batchRange(releaseName);
   225→   225→  if (!br) return null;
   226→   226→  const subs = subtitleWords(p, [title, originalTitle]).length;
   227→   227→  const sequelNamed = ctx.relEpisode == null && subs >= 2;
   228→   228→  const wanted = ctx.absEpisode ?? ctx.episode;
   229→   229→  const inAbs = wanted >= br[0] && wanted <= br[1];
   230→   230→  const inRel = ctx.relEpisode != null && ctx.relEpisode >= br[0] && ctx.relEpisode <= br[1];
   231→   231→  if (inAbs && !sequelNamed) return "pack";
   232→   232→  if (inRel && subs >= 2) return "pack";
   233→   233→  return null;
   234→   234→}

   246→   246→/** IMDb-keyed — but indexers mislabel ids, so names are still cross-checked. */
   247→   247→async function providerTorrentio(
   248→   248→  type: "movie" | "tv",
   249→   249→  imdbId: string,
   250→   250→  season: number,
   251→   251→  episode: number,
   252→   252→  year: string,
   253→   253→  title: string,
   254→   254→  originalTitle: string | undefined,
   255→   255→  ctx: Pick<BayContext, "airDate" | "showLang" | "absEpisode" | "relEpisode" | "isAnime">
   256→   256→): Promise<TorrentSource[]> {
   257→   257→  // Stremio addon stream ids are single segments: `tt…:season:episode`.
   258→   258→  // (slash-separated paths 404 since the 2026 addon API tightening)
   259→   259→  const path =
   260→   260→    type === "movie"
   261→   261→      ? `stream/movie/${imdbId}.json`
   262→   262→      : `stream/series/${imdbId}:${season}:${episode}.json`;
   263→   263→  const data = await fetchJson<{ streams?: TorrentioStream[] }>(
   264→   264→    `https://torrentio.strem.fun/${path}`
   265→   265→  );
   266→   266→  const streams = data?.streams ?? [];
   267→   267→  const out: TorrentSource[] = [];
   268→   268→  for (const s of streams) {
   269→   269→    if (!s.infoHash || !/^[0-9a-f]{40}$/i.test(s.infoHash)) continue;
   270→   270→    const titleLines = (s.title ?? "").split("\n");
   271→   271→    const releaseName = (titleLines[0] || s.behaviorHints?.filename || "Torrent").trim();
   272→   272→    // indexers occasionally attach a wrong IMDb id — cross-check the name
   273→   273→    // with the full strict matcher (catches franchise uploads that share the
   274→   274→    // IMDb entry, e.g. original-series batches answering a TYBW request)
   275→   275→    const parsed = parseRelease(releaseName);
   276→   276→    const match = strictImdbMatch(parsed, type, title, originalTitle, year, season, episode, ctx);
   277→   277→    if (!match) continue;
   278→   278→    const metaLine = titleLines.find((l) => l.includes("👤")) ?? "";
   279→   279→    const seeders = parseCount(metaLine.match(/👤\s*([\d.,]+\s*[KM]?)/)?.[1]);
   280→   280→    const sizeBytes = parseSizeBytes(metaLine.match(/💾\s*([\d.,]+\s*[KMGTB]+)/i)?.[1]);
   281→   281→    const provider = metaLine.match(/⚙️\s*([^\n]+)/)?.[1]?.trim() || "Torrentio";
   282→   282→    const hintName = s.name ?? "";
   283→   283→    const quality =
   284→   284→      /4k/i.test(hintName) && !/1080p/.test(releaseName)
   285→   285→        ? "2160p"
   286→   286→        : detectQuality(releaseName);
   287→   287→    out.push({
   288→   288→      id: s.infoHash.toLowerCase(),
   289→   289→      releaseName,
   290→   290→      seeders,
   291→   291→      peers: 0,
   292→   292→      sizeBytes,
   293→   293→      provider,
   294→   294→      quality,
   295→   295→      fileIdx: typeof s.fileIdx === "number" ? s.fileIdx : null,
   296→   296→      match,
   297→   297→      audioTags: detectAudioTags(releaseName, Boolean(ctx.isAnime)),
   298→   298→      score: 0,
   299→   299→    });
   300→   300→  }
   301→   301→  return out;
   302→   302→}

   326→   326→/** Air date (YYYY-MM-DD) of one episode — short TTL so brand-new daily-show

   331→   331→export async function getEpisodeAirDate(
   332→   332→  tmdbId: number,
   333→   333→  season: number,
   334→   334→  episode: number
   335→   335→): Promise<string | null> {
   336→   336→  const key = `tv:${tmdbId}:${season}`;
   337→   337→  let hit = seasonCache.get(key);
   338→   338→  if (!hit || Date.now() - hit.ts > SEASON_TTL) {
   339→   339→    try {
   340→   340→      const data = await tmdbServer<{
   341→   341→        episodes?: { episode_number: number; air_date: string | null }[];
   342→   342→      }>(`tv/${tmdbId}/season/${season}`);
   343→   343→      const dates = new Map<number, string | null>();
   344→   344→      for (const e of data.episodes ?? []) dates.set(e.episode_number, e.air_date ?? null);
   345→   345→      hit = { dates, ts: Date.now() };
   346→   346→      seasonCache.set(key, hit);
   347→   347→    } catch {
   348→   348→      return null;
   349→   349→    }
   350→   350→  }
   351→   351→  return hit.dates.get(episode) ?? null;
   352→   352→}

   372→   372→/**
   373→   373→ * Keyword provider — answers on title words alone, so every result is
   374→   374→ * strictly validated here against the requested year (movies) or
   375→   375→ * season/episode (TV) before it may enter the candidate list.
   376→   376→ */
   377→   377→async function providerPirateBay(
   378→   378→  type: "movie" | "tv",
   379→   379→  ctx: BayContext
   380→   380→): Promise<TorrentSource[]> {
   381→   381→  const { title, originalTitle, year, season, episode } = ctx;
   382→   382→  const queries =
   383→   383→    type === "movie"
   384→   384→      ? [`${title} ${year ?? ""}`.trim(), title]
   385→   385→      : [
   386→   386→          `${title} s${pad(season)}e${pad(episode)}`,
   387→   387→          // language variant — Indian versions ("Bigg Boss Telugu") overflow the
   388→   388→          // bare-title result window on trackers
   389→   389→          ...(ctx.showLang ? [`${title} ${ctx.showLang}`] : []),
   390→   390→          `${title} ${season}x${pad(episode)}`,
   391→   391→          title, // dated daily-show names carry no SxxExx — match by name + air date
   392→   392→        ];
   393→   393→
   394→   394→  const seen = new Map<string, TorrentSource>();
   395→   395→  const collect = (rows: ApibayResult[]) => {
   396→   396→    for (const r of rows) {
   397→   397→      if (!r.info_hash || !/^[0-9a-f]{40}$/i.test(r.info_hash)) continue;
   398→   398→      const name = (r.name ?? "Torrent").trim();
   399→   399→      if (name === "No results") continue;
   400→   400→      const p = parseRelease(name);
   401→   401→      const match =
   402→   402→        type === "movie"
   403→   403→          ? matchMovie(p, title, originalTitle, year)
   404→   404→          : matchTv(p, season, episode, title, originalTitle, {
   405→   405→              airDate: ctx.airDate,
   406→   406→              showLang: ctx.showLang,
   407→   407→              absEpisode: ctx.absEpisode,
   408→   408→            });
   409→   409→      if (!match) continue; // same name, different film/episode — reject

   417→   417→        id: key,
   418→   418→        releaseName: name,

   420→   420→        peers: parseCount(r.leechers),

   424→   424→        fileIdx: null,
   425→   425→        match,
   426→   426→        countryPenalty: cPen,

   428→   428→        score: 0,
   429→   429→      });
   430→   430→    }
   431→   431→  };
   432→   432→
   433→   433→  // apibay answers tiny JSON quickly; sequential keeps request count low
   434→   434→  for (const q of queries) {
   435→   435→    if (seen.size >= 5) break;
   436→   436→    const rows = await fetchJson<ApibayResult[]>(
   437→   437→      `https://apibay.org/q.php?q=${encodeURIComponent(q)}&cat=`,
   438→   438→      7000
   439→   439→    );
   440→   440→    if (rows && Array.isArray(rows)) collect(rows);
   441→   441→  }
   442→   442→
   443→   443→  // episode-level search came up empty → try season packs for TV
   444→   444→  if (type === "tv" && seen.size === 0) {
   445→   445→    const packQueries = [`${title} s${pad(season)}`, `${title} season ${season}`];
   446→   446→    for (const q of packQueries) {

   449→   449→        7000
   450→   450→      );
   451→   451→      if (rows && Array.isArray(rows)) collect(rows);
   452→   452→      if (seen.size > 0) break;
   453→   453→    }
   454→   454→  }
   455→   455→  return [...seen.values()];

   533→   533→            absEpisode: ctx.absEpisode,
   534→   534→          });

   536→   536→    if (type === "tv" && countryPenalty(p, ctx.countries) < 0) continue;

   539→   539→      releaseName: c.name,
   540→   540→      seeders: c.seeders,

   751→   751→      `https://nyaa.si/?page=rss&f=0&c=0_0&q=${encodeURIComponent(query)}`,
   752→   752→      {

   759→   759→    );
   760→   760→    if (!res.ok) return [];
   761→   761→    const xml = await res.text();
   762→   762→    const items: NyaaItem[] = [];
   763→   763→    for (const chunk of xml.split("<item>").slice(1)) {
   764→   764→      const pick = (re: RegExp) => chunk.match(re)?.[1]?.trim() ?? "";
   765→   765→      const title = pick(/<title>([\s\S]*?)<\/title>/);
   766→   766→      const infoHash = pick(/<nyaa:infoHash>([0-9a-fA-F]{40})<\/nyaa:infoHash>/).toLowerCase();

   771→   771→      const pubMs = pubRaw ? new Date(pubRaw).getTime() : NaN;
   772→   772→      items.push({
   773→   773→        title,
   774→   774→        infoHash,
   775→   775→        seeders: parseInt(pick(/<nyaa:seeders>(\d+)<\/nyaa:seeders>/), 10) || 0,
   776→   776→        leechers: parseInt(pick(/<nyaa:leechers>(\d+)<\/nyaa:leechers>/), 10) || 0,
   777→   777→        sizeBytes: parseSizeBytes(sizeRaw),
   778→   778→        pubDate: Number.isFinite(pubMs) ? pubMs : null,
   779→   779→      });
   780→   780→    }
   781→   781→    return items;
   782→   782→  } catch {
   783→   783→    return [];

   789→   789→export function batchRange(name: string): [number, number] | null {
   790→   790→  const m = name.match(/(\d{1,4})(?:\.\d)?\s*(?:[-–—~]|to)\s*E?(\d{1,4})\b/i);
   791→   791→  if (!m) return null;
   792→   792→  const a = Number(m[1]);
   793→   793→  const b = Number(m[2]);
   794→   794→  return b > a && b - a <= 400 ? [a, b] : null;
   795→   795→}

   818→   818→        ];

   822→   822→    if (seen.size >= 6) break;

   827→   827→      let match: MatchClass | null;
   828→   828→      if (type === "movie") {
   829→   829→        match = matchMovie(p, title, originalTitle, ctx.year);
   830→   830→      } else {
   831→   831→        match = matchTv(p, season, episode, title, originalTitle, {
   832→   832→          airDate: ctx.airDate,
   833→   833→          absEpisode,
   834→   834→          relEpisode: relWanted,
   835→   835→          pubDate: r.pubDate,
   836→   836→        });
   837→   837→        if (!match) {
   838→   838→          // multi-episode batch: keep it as a pack — the stream route picks
   839→   839→          // the right file inside the torrent by episode number. The relative
   840→   840→          // number is only accepted inside a SUBTITLED release (>= 2 unconsumed
   841→   841→          // title words — "Sennen Kessen-hen 1-52 Batch"), so the base show's
   842→   842→          // batches ("Bleach 001~366", "BLEACH Kai 1~366") can never answer
   843→   843→          // a sequel-season request, and sequel-named batches can never answer
   844→   844→          // a base-season request.
   845→   845→          match = animeBatchMatch(p, r.title, {
   846→   846→            absEpisode,
   847→   847→            relEpisode: relWanted,
   848→   848→            episode,
   849→   849→          }, title, originalTitle);
   850→   850→        }
   851→   851→      }
   852→   852→      if (!match) continue;
   853→   853→      seen.set(r.infoHash, {
   854→   854→        id: r.infoHash,
   855→   855→        releaseName: r.title,
   856→   856→        seeders: r.seeders,
   857→   857→        peers: r.leechers,
   858→   858→        sizeBytes: r.sizeBytes,
   859→   859→        provider: "Nyaa",
   860→   860→        quality: detectQuality(r.title),
   861→   861→        fileIdx: null,
   862→   862→        match,
   863→   863→        audioTags: detectAudioTags(r.title, true),
   864→   864→        score: 0,
   865→   865→      });
   866→   866→    }
   867→   867→  }
   868→   868→  return [...seen.values()];
   869→   869→}

   871→   871→/* ---------- scoring ---------- */
   872→   872→
   873→   873→export function scoreTorrent(t: TorrentSource): number {
   874→   874→  let s = Math.min(Math.log10(t.seeders + 1) * 4, 13);
   875→   875→  if (t.quality === "1080p") s += 5;
   876→   876→  else if (t.quality === "720p") s += 4;
   877→   877→  else if (t.quality === "2160p") s += 2;
   878→   878→  else if (t.quality === "480p") s += 1;
   879→   879→  const n = t.releaseName.toLowerCase();
   880→   880→  if (/x265|h\.?265|hevc/.test(n)) s -= 4; // HEVC support is spotty in browsers
   881→   881→  if (/av1/.test(n)) s += 1;
   882→   882→  if (/x264|h\.?264|xvid/.test(n)) s += 2;
   883→   883→  if (/\b(cam|ts|telesync|screener|dvdscr|hdts|\btc\b)\b/.test(n)) s -= 12;
   884→   884→  if (/\b3d\b/.test(n)) s -= 6;
   885→   885→  if (/\.mp4\b/.test(n) || /yts/i.test(t.releaseName)) s += 1.5;
   886→   886→  if (t.sizeBytes > 0 && t.sizeBytes < 120 * 1024 * 1024) s -= 2;
   887→   887→  if (t.sizeBytes > 25 * 1024 * 1024 * 1024) s -= 3;
   888→   888→  return Math.round(s * 10) / 10;
   889→   889→}
   890→   890→
   891→   891→/** Bonus/penalty applied for how tightly a release matches the request. */
   892→   892→const MATCH_RANK: Record<MatchClass, number> = {
   893→   893→  verified: 8,
   894→   894→  pack: 2.5,
   895→   895→  weak: -6,
   896→   896→};
   897→   897→
   898→   898→function merge(sources: TorrentSource[]): TorrentSource[] {
   899→   899→  const best = new Map<string, TorrentSource>();
   900→   900→  for (const t of sources) {
   901→   901→    const prev = best.get(t.id);
   902→   902→    if (!prev || t.seeders > prev.seeders) best.set(t.id, t);
   903→   903→  }
   904→   904→  const list = [...best.values()];
   905→   905→  for (const t of list) {
   906→   906→    t.score =
   907→   907→      Math.round((scoreTorrent(t) + MATCH_RANK[t.match] + (t.countryPenalty ?? 0)) * 10) / 10;
   908→   908→  }
   909→   909→  // only fall back to unverified (no-year) releases when nothing better exists
   910→   910→  const confident = list.filter((t) => t.match !== "weak");
   911→   911→  const final = confident.length ? confident : list;
   912→   912→  return final.sort((a, b) => b.score - a.score || b.seeders - a.seeders).slice(0, 14);
   913→   913→}

   939→   939→export async function searchTorrentSources(args: SearchArgs): Promise<SearchResult> {
   940→   940→  const { type, tmdbId, title, season, episode } = args;
   941→   941→  const [imdbId, meta] = await Promise.all([getImdbId(type, tmdbId), getTitleMeta(type, tmdbId)]);
   942→   942→  // TMDB's own year is canonical (guards against stale/rounded client data)
   943→   943→  const year = meta.year || args.year || "";
   944→   944→  // Indian daily shows (Bigg Boss…) name episodes by broadcast date and tag
   945→   945→  // language versions — both are needed to match them without mixing content
   946→   946→  const airDate = type === "tv" ? await getEpisodeAirDate(tmdbId, season, episode) : null;
   947→   947→  const showLang = meta.originalLanguage ? LANG_WORD[meta.originalLanguage] ?? null : null;
   948→   948→  // "Bigg Boss Tamil" uploads are named "Bigg Boss (2026) S09E101 [Tamil …]":
   949→   949→  // the language moves into the tag area, so match against the base title and
   950→   950→  // let the release's language tag pick the right version (mismatch = reject)
   951→   951→  const matchTitle =
   952→   952→    type === "tv" && showLang
   953→   953→      ? title.replace(new RegExp(`\\s+${showLang}$`, "i"), "").trim() || title
   954→   954→      : title;
   955→   955→
   956→   956→  // Japanese titles are anime: fansub indexers (Nyaa) get queried and the
   957→   957→  // broadcast-order episode number is offered to the matchers. Sequel seasons
   958→   958→  // folded into the base show ("Bleach" S02 = TYBW) additionally carry the
   959→   959→  // fansub-relative episode number.
   960→   960→  const isAnime = (meta.originalLanguage ?? "") === "ja";
   961→   961→  const absEpisode = type === "tv" && isAnime ? await getAbsoluteEpisode(tmdbId, season, episode) : null;
   962→   962→  const relEpisode = type === "tv" && isAnime && season >= 2 ? episode : null;
   963→   963→
   964→   964→  const jobs: Promise<TorrentSource[]>[] = [];
   965→   965→  if (matchTitle) {
   966→   966→    const bayCtx: BayContext = {
   967→   967→      title: matchTitle,
   968→   968→      originalTitle: meta.originalTitle ?? undefined,
   969→   969→      year: year || undefined,
   970→   970→      countries: meta.countries,
   971→   971→      season,
   972→   972→      episode,
   973→   973→      airDate,
   974→   974→      showLang,
   975→   975→      absEpisode,
   976→   976→      relEpisode,
   977→   977→      isAnime,
   978→   978→    };
   979→   979→    jobs.push(providerPirateBay(type, bayCtx));
   980→   980→    jobs.push(providerBitSearch(type, bayCtx));
   981→   981→    jobs.push(providerTorrentsCsv(type, bayCtx));
   982→   982→    if (isAnime) jobs.push(providerNyaa(type, bayCtx));
   983→   983→  }
   984→   984→  if (imdbId) {
   985→   985→    jobs.push(
   986→   986→      providerTorrentio(type, imdbId, season, episode, year, matchTitle, meta.originalTitle ?? undefined, {
   987→   987→        airDate,
   988→   988→        showLang,
   989→   989→        absEpisode,
   990→   990→        relEpisode,
   991→   991→        isAnime,
   992→   992→      })
   993→   993→    );
   994→   994→    if (type === "movie") jobs.push(providerYts(imdbId, year));
   995→   995→    else jobs.push(providerEztv(imdbId, season, episode));
   996→   996→  }
   997→   997→
   998→   998→  const settled = await Promise.allSettled(jobs);
   999→   999→  const all: TorrentSource[] = [];
  1000→  1000→  for (const r of settled) if (r.status === "fulfilled") all.push(...r.value);
  1001→  1001→  return {
  1002→  1002→    results: merge(all),
  1003→  1003→    hints: { absEpisode: absEpisode ?? null, relEpisode: relEpisode ?? null },
  1004→  1004→  };

  1011→  1011→export async function quickAvailability(
  1012→  1012→  type: "movie" | "tv",

  1018→  1018→  const hit = availCache.get(key);

  1020→  1020→  const data = await fetchJson<{ streams?: TorrentioStream[] }>(

  1024→  1024→    8000

  1029→  1029→}
