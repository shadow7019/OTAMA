    36→    36→interface PlayerModalProps {

    48→    48→

    65→    65→  up: "bg-emerald-400",
    66→    66→  popup: "bg-amber-400",
    67→    67→  down: "bg-red-500",
    68→    68→};
    69→    69→
    70→    70→const HEALTH_TIP: Record<HealthState, string> = {
    71→    71→  checking: "Checking server…",
    72→    72→  up: "Server is up",
    73→    73→  popup: "Server is up but may block embedding — use Pop out",
    74→    74→  down: "Server not responding right now",
    75→    75→};

    77→    77→export function PlayerModal({ target, actions, onClose }: PlayerModalProps) {
    78→    78→  // remounted per target via key — initial state is safe to derive from props
    79→    79→  const type = target ? getMediaType(target.item) : "movie";
    80→    80→  const itemId = target?.item.id ?? null;
    81→    81→
    82→    82→  // Auto mode is the default every session: the player health-checks all servers
    83→    83→  // and picks the fastest healthy one (ad-free direct stream first when available).
    84→    84→  // The persisted manual choice is honoured as auto's preferred-first hint, not as
    85→    85→  // a forced pick — so a dead saved server can never break playback again.
    86→    86→  const [selectedId, setSelectedId] = useState<string>(AUTO_SOURCE_ID);
    87→    87→  const [autoPreferred, setAutoPreferred] = useState<string | null>(() => {
    88→    88→    const saved = useMediaStore.getState().sourceId;
    89→    89→    return PLAYER_SOURCES.some((s) => s.id === saved) ? saved : null;
    90→    90→  });
    91→    91→  const [healthMap, setHealthMap] = useState<Record<string, Health>>({});
    92→    92→  const [archive, setArchive] = useState<ArchiveMatchClient | null>(null);
    93→    93→  const [archiveState, setArchiveState] = useState<"loading" | "found" | "none">("loading");
    94→    94→  const [videoError, setVideoError] = useState(false);
    95→    95→  const [autoPickId, setAutoPickId] = useState<string | null>(null);
    96→    96→  const [tried, setTried] = useState<string[]>([]);
    97→    97→  const [banner, setBanner] = useState<string | null>(null);
    98→    98→  const [season, setSeason] = useState(() => target?.season ?? 1);
    99→    99→  const [episode, setEpisode] = useState(() => target?.episode ?? 1);
   100→   100→  const [episodes, setEpisodes] = useState<Episode[]>([]);
   101→   101→  const [epLoading, setEpLoading] = useState(() => type === "tv");
   102→   102→  const [details, setDetails] = useState<MediaDetails | null>(null);
   103→   103→  const [showEpisodes, setShowEpisodes] = useState(false);
   104→   104→  const [frameState, setFrameState] = useState<FrameState>("loading");
   105→   105→  const [reloadNonce, setReloadNonce] = useState(0);
   106→   106→  const mounted = useMounted();
   107→   107→  const addContinue = useMediaStore((s) => s.addContinue);
   108→   108→  const toggleWatchlist = useMediaStore((s) => s.toggleWatchlist);
   109→   109→  const setSourceId = useMediaStore((s) => s.setSourceId);
   110→   110→  const watchlist = useMediaStore((s) => s.watchlist);
   111→   111→
   112→   112→  const nameOf = (id: string) =>
   113→   113→    id === ARCHIVE_SOURCE_ID
   114→   114→      ? "CineX Direct"
   115→   115→      : PLAYER_SOURCES.find((s) => s.id === id)?.name ?? "Server";
   116→   116→
   117→   117→  // Healthy embed servers, fastest first (frameable ones outrank popup-ish ones;
   118→   118→  // the user's preferred server goes first while it stays healthy)
   119→   119→  const rankedHealthy = useMemo(() => {
   120→   120→    const score = (s: Health) => (s.state === "up" ? 0 : s.state === "popup" ? 1 : 2);
   121→   121→    const list = PLAYER_SOURCES.filter((s) => {
   122→   122→      const h = healthMap[s.id];
   123→   123→      return h && (h.state === "up" || h.state === "popup");
   124→   124→    }).sort(
   125→   125→      (a, b) =>
   126→   126→        score(healthMap[a.id]) - score(healthMap[b.id]) ||
   127→   127→        (healthMap[a.id].ms ?? 9e9) - (healthMap[b.id].ms ?? 9e9)
   128→   128→    );
   129→   129→    if (!autoPreferred) return list;
   130→   130→    const prefIdx = list.findIndex((s) => s.id === autoPreferred);
   131→   131→    if (prefIdx < 0) return list;
   132→   132→    const pref = list.splice(prefIdx, 1)[0];
   133→   133→    return [pref, ...list];
   134→   134→  }, [healthMap, autoPreferred]);
   135→   135→
   136→   136→  // Which source is actually on screen right now
   137→   137→  const effectiveId = (() => {
   138→   138→    if (selectedId === AUTO_SOURCE_ID) {
   139→   139→      if (type === "movie" && archiveState === "found" && archive && !videoError)
   140→   140→        return ARCHIVE_SOURCE_ID;
   141→   141→      if (autoPickId) return autoPickId;
   142→   142→      return rankedHealthy[0]?.id ?? PLAYER_SOURCES[0].id;
   143→   143→    }
   144→   144→    if (selectedId === ARCHIVE_SOURCE_ID && archiveState === "found" && !videoError)
   145→   145→      return ARCHIVE_SOURCE_ID;
   146→   146→    return PLAYER_SOURCES.some((s) => s.id === selectedId) ? selectedId : PLAYER_SOURCES[0].id;
   147→   147→  })();
   148→   148→
   149→   149→  // identity of the current embed; resetting to "loading" is done at render time
   150→   150→  const srcKey = `${itemId ?? "none"}-${effectiveId}-${season}-${episode}-${reloadNonce}`;
   151→   151→  const [prevSrcKey, setPrevSrcKey] = useState(srcKey);
   152→   152→  if (prevSrcKey !== srcKey) {
   153→   153→    setPrevSrcKey(srcKey);
   154→   154→    setFrameState("loading");
   155→   155→    setVideoError(false);
   156→   156→  }
   157→   157→
   158→   158→  // failover: move to the next healthy server we haven't tried yet
   159→   159→  const failover = (fromId: string, reason: string) => {
   160→   160→    const next =
   161→   161→      rankedHealthy.map((s) => s.id).find((id) => id !== fromId && !tried.includes(id)) ??
   162→   162→      null;
   163→   163→    if (next) {
   164→   164→      setTried((t) => [...t, fromId]);
   165→   165→      setAutoPickId(next);
   166→   166→      setSelectedId(AUTO_SOURCE_ID);
   167→   167→      if (autoPreferred && autoPreferred !== next) setAutoPreferred(null); // it just proved unreliable
   168→   168→      setBanner(reason.replace("{next}", nameOf(next)));
   169→   169→    }
   170→   170→    return next;
   171→   171→  };
   172→   172→
   173→   173→  // record progress for "Continue Watching" (external system write);
   174→   174→  // re-records as the user navigates seasons/episodes so we resume at the latest spot
   175→   175→  useEffect(() => {
   176→   176→    if (!target || !mounted) return;
   177→   177→    const t = getMediaType(target.item);
   178→   178→    addContinue(
   179→   179→      target.item,
   180→   180→      t === "tv" ? season : undefined,
   181→   181→      t === "tv" ? episode : undefined
   182→   182→    );
   183→   183→  }, [target, mounted, addContinue, season, episode]);
   184→   184→
   185→   185→  // fetch details (for season list)
   186→   186→  useEffect(() => {
   187→   187→    if (!target) return;
   188→   188→    let cancelled = false;
   189→   189→    tmdb<MediaDetails>(`${type}/${target.item.id}`)
   190→   190→      .then((d) => !cancelled && setDetails(d))
   191→   191→      .catch(() => {});
   192→   192→    return () => {
   193→   193→      cancelled = true;
   194→   194→    };
   195→   195→  }, [target, type]);
   196→   196→
   197→   197→  // fetch episodes when season changes (TV only)
   198→   198→  useEffect(() => {
   199→   199→    if (!target || type !== "tv") return;
   200→   200→    let cancelled = false;
   201→   201→    tmdb<{ episodes: Episode[] }>(`tv/${target.item.id}/season/${season}`)
   202→   202→      .then((res) => {
   203→   203→        if (!cancelled) {
   204→   204→          setEpisodes(res.episodes ?? []);
   205→   205→          setEpLoading(false);
   206→   206→        }
   207→   207→      })
   208→   208→      .catch(() => {
   209→   209→        if (!cancelled) {
   210→   210→          setEpisodes([]);
   211→   211→          setEpLoading(false);
   212→   212→        }
   213→   213→      });
   214→   214→    return () => {
   215→   215→      cancelled = true;
   216→   216→    };
   217→   217→  }, [target, type, season]);
   218→   218→
   219→   219→  // probe every server once per title so the strip shows live status dots + latency
   220→   220→  useEffect(() => {
   221→   221→    if (itemId === null) return;
   222→   222→    let cancelled = false;
   223→   223→    PLAYER_SOURCES.forEach((s) => {
   224→   224→      const probe = type === "movie" ? s.movie(itemId) : s.tv(itemId, 1, 1);
   225→   225→      fetch(`/api/embed-check?url=${encodeURIComponent(probe)}`)
   226→   226→        .then((r) => r.json())
   227→   227→        .then((j: { ok?: boolean; ms?: number; frameable?: boolean }) => {
   228→   228→          if (cancelled) return;
   229→   229→          const h: Health = !j.ok
   230→   230→            ? { state: "down", ms: j.ms }
   231→   231→            : j.frameable === false
   232→   232→              ? { state: "popup", ms: j.ms }
   233→   233→              : { state: "up", ms: j.ms };
   234→   234→          setHealthMap((prev) => ({ ...prev, [s.id]: h }));
   235→   235→        })
   236→   236→        .catch(() => {
   237→   237→          if (!cancelled) setHealthMap((prev) => ({ ...prev, [s.id]: { state: "down" } }));
   238→   238→        });
   239→   239→    });
   240→   240→
   241→   241→    // CineX Direct: look for a public-domain copy (movies only)
   242→   242→    if (type === "movie" && target) {
   243→   243→      const q = getTitle(target.item);
   244→   244→      const y = getYear(target.item);
   245→   245→      fetch(`/api/archive?title=${encodeURIComponent(q)}&year=${encodeURIComponent(y)}`)
   246→   246→        .then((r) => r.json())
   247→   247→        .then((j: { match: ArchiveMatchClient | null }) => {
   248→   248→          if (cancelled) return;
   249→   249→          setArchive(j.match);
   250→   250→          setArchiveState(j.match ? "found" : "none");
   251→   251→        })
   252→   252→        .catch(() => {
   253→   253→          if (!cancelled) setArchiveState("none");
   254→   254→        });
   255→   255→    }
   256→   256→    return () => {
   257→   257→      cancelled = true;
   258→   258→    };
   259→   259→  }, [itemId, type]);
   260→   260→
   261→   261→  // watchdog: if the current embed never loads, fail over automatically
   262→   262→  // (the native direct player has its own error/stall handling — skip it there)
   263→   263→  useEffect(() => {
   264→   264→    if (!target || effectiveId === ARCHIVE_SOURCE_ID) return;
   265→   265→    const t = setTimeout(() => {
   266→   266→      setFrameState((s) => {
   267→   267→        if (s !== "loading") return s;
   268→   268→        // auto mode: hop to the next healthy server
   269→   269→        if (selectedId === AUTO_SOURCE_ID) {
   270→   270→          failover(effectiveId, `{name} didn't respond — switched to {next}`);
   271→   271→        } else if (
   272→   272→          rankedHealthy.length &&
   273→   273→          healthMap[selectedId]?.state !== "up" &&
   274→   274→          healthMap[selectedId]?.state !== "checking"
   275→   275→        ) {
   276→   276→          // manual mode on a dead server: switch to auto once per title
   277→   277→          setTried((tt) => [...tt, effectiveId]);
   278→   278→          setAutoPickId(rankedHealthy[0].id);
   279→   279→          setSelectedId(AUTO_SOURCE_ID);
   280→   280→          setBanner(`${nameOf(selectedId)} didn't respond — switched to ${nameOf(rankedHealthy[0].id)}`);
   281→   281→        }
   282→   282→        return s;
   283→   283→      });
   284→   284→    }, LOAD_TIMEOUT_MS);
   285→   285→  }, [srcKey, frameState]);
   286→   286→
   287→   287→  // transient banner auto-dismiss
   288→   288→  useEffect(() => {
   289→   289→    if (!banner) return;
   290→   290→    const t = setTimeout(() => setBanner(null), 7000);
   291→   291→    return () => clearTimeout(t);
   292→   292→  }, [banner]);
   293→   293→
   294→   294→  // persist the preferred server for future sessions (external store write);
   295→   295→  // "auto" when the user hasn't expressed a preference
   296→   296→  useEffect(() => {
   297→   297→    setSourceId(autoPreferred ?? AUTO_SOURCE_ID);
   298→   298→  }, [autoPreferred, setSourceId]);
   299→   299→
   300→   300→  // lock scroll + escape
   301→   301→  useEffect(() => {
   302→   302→    if (!target) return;
   303→   303→    const original = document.body.style.overflow;
   304→   304→    document.body.style.overflow = "hidden";
   305→   305→    const onKey = (ev: KeyboardEvent) => {
   306→   306→      if (ev.key === "Escape") onClose();
   307→   307→    };
   308→   308→    window.addEventListener("keydown", onKey);
   309→   309→    return () => {
   310→   310→      document.body.style.overflow = original;
   311→   311→      window.removeEventListener("keydown", onKey);
   312→   312→    };
   313→   313→  }, [target, onClose]);
   314→   314→
   315→   315→  const seasonsList = (
   316→   316→    details?.seasons ?? []
   317→   317→  ).filter((s) => s.season_number > 0 && s.episode_count > 0);
   318→   318→
   319→   319→  const maxEpisode =
   320→   320→    episodes.length ||
   321→   321→    details?.seasons?.find((s) => s.season_number === season)?.episode_count ||
   322→   322→    1;
   323→   323→
   324→   324→  if (!target) return null;
   325→   325→
   326→   326→  const item = target.item;
   327→   327→  const title = getTitle(item);
   328→   328→  const isDirect = effectiveId === ARCHIVE_SOURCE_ID;
   329→   329→  const currentServer = PLAYER_SOURCES.find((s) => s.id === effectiveId);
   330→   330→  const src = currentServer
   331→   331→    ? type === "movie"
   332→   332→      ? currentServer.movie(item.id)
   333→   333→      : currentServer.tv(item.id, season, episode)
   334→   334→    : "";
   335→   335→
   336→   336→  const goEpisode = (s: number, e: number) => {
   337→   337→    setSeason(s);
   338→   338→    setEpisode(e);
   339→   339→    setShowEpisodes(false);
   340→   340→  };
   341→   341→
   342→   342→  const nextEpisode = () => {
   343→   343→    if (episode < maxEpisode) goEpisode(season, episode + 1);
   344→   344→    else if (seasonsList.some((s) => s.season_number === season + 1)) goEpisode(season + 1, 1);
   345→   345→  };
   346→   346→
   347→   347→  const prevEpisode = () => {
   348→   348→    if (episode > 1) goEpisode(season, episode - 1);
   349→   349→    else if (season > 1) goEpisode(season - 1, 1);
   350→   350→  };
   351→   351→
   352→   352→  const inList = mounted
   353→   353→    ? watchlist.some((w) => w.id === item.id && w.media_type === type)
   354→   354→    : false;
   355→   355→
   356→   356→  const switchSource = (id: string) => {
   357→   357→    if (id === effectiveId) {
   358→   358→      // re-selecting the active source = manual retry
   359→   359→      setFrameState("loading");
   360→   360→      setReloadNonce((n) => n + 1);
   361→   361→      if (id === ARCHIVE_SOURCE_ID) setVideoError(false);
   362→   362→      return;
   363→   363→    }
   364→   364→    setTried((t) => [...t, effectiveId]);
   365→   365→    if (id === ARCHIVE_SOURCE_ID) setVideoError(false);
   366→   366→    setSelectedId(id);
   367→   367→    if (PLAYER_SOURCES.some((s) => s.id === id)) {
   368→   368→      setAutoPreferred(id); // remember the user's preference
   369→   369→      setAutoPickId(id);
   370→   370→    } else if (id === AUTO_SOURCE_ID) {
   371→   371→      setAutoPickId(null); // recompute from live health ranking
   372→   372→    }
   373→   373→  };
   374→   374→
   375→   375→  const retryServer = () => {
   376→   376→    setFrameState("loading");
   377→   377→    setReloadNonce((n) => n + 1);
   378→   378→    if (isDirect) setVideoError(false);
   379→   379→  };
   380→   380→
   381→   381→  const tryNextServer = () => {
   382→   382→    if (selectedId === AUTO_SOURCE_ID || isDirect) {
   383→   383→      if (!failover(effectiveId, `{name} failed — switching to {next}`)) {
   384→   384→        // nothing left to try — cycle manually through the registry
   385→   385→        const i = PLAYER_SOURCES.findIndex((s) => s.id === effectiveId);
   386→   386→        switchSource(PLAYER_SOURCES[(i + 1) % PLAYER_SOURCES.length].id);
   387→   387→      }
   388→   388→    } else {
   389→   389→      const i = PLAYER_SOURCES.findIndex((s) => s.id === selectedId);
   390→   390→      switchSource(PLAYER_SOURCES[(i + 1) % PLAYER_SOURCES.length].id);
   391→   391→    }
   392→   392→  };
   393→   393→
   394→   394→  return (
   395→   395→    <div
   396→   396→      className="fixed inset-0 z-[80] flex flex-col bg-black"
   397→   397→      role="dialog"
   398→   398→      aria-modal="true"
   399→   399→      aria-label={`Playing ${title}`}
   400→   400→    >
   401→   401→      {/* Top bar */}
   402→   402→      <div className="flex items-center justify-between gap-3 px-3 py-2.5 md:px-5">
   403→   403→        <div className="flex min-w-0 items-center gap-3">
   404→   404→          <button
   405→   405→            type="button"
   406→   406→            onClick={onClose}
   407→   407→            aria-label="Close player"
   408→   408→            className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full bg-white/10 text-white transition hover:bg-red-600"
   409→   409→          >
   410→   410→            <X className="h-5 w-5" />
   411→   411→          </button>
   412→   412→          <div className="min-w-0">
   413→   413→            <h2 className="truncate text-sm font-bold text-white md:text-base">
   414→   414→              {title}
   415→   415→              {type === "tv" && (
   416→   416→                <span className="ml-2 font-normal text-zinc-400">
   417→   417→                  S{season}:{episode < 10 ? `0${episode}` : episode}
   418→   418→                </span>
   419→   419→              )}
   420→   420→            </h2>
   421→   421→            {type === "tv" && episodes.find((ep) => ep.episode_number === episode)?.name && (
   422→   422→              <p className="truncate text-xs text-zinc-500">
   423→   423→                {episodes.find((ep) => ep.episode_number === episode)?.name}
   424→   424→              </p>
   425→   425→            )}
   426→   426→          </div>
   427→   427→        </div>
   428→   428→
   429→   429→        <div className="flex shrink-0 items-center gap-2">
   430→   430→          <button
   431→   431→            type="button"
   432→   432→            onClick={() =>
   433→   433→              toggleWatchlist(
   434→   434→                item,
   435→   435→                type === "tv" ? season : undefined,
   436→   436→                type === "tv" ? episode : undefined
   437→   437→              )
   438→   438→            }
   439→   439→            className={`hidden h-9 items-center gap-1.5 rounded-full px-3 text-xs font-semibold sm:flex ${
   440→   440→              inList
   441→   441→                ? "bg-emerald-600/20 text-emerald-400 ring-1 ring-emerald-600/50"
   442→   442→                : "bg-white/10 text-zinc-200 hover:bg-white/20"
   443→   443→            }`}
   444→   444→          >
   445→   445→            {inList ? <Check className="h-3.5 w-3.5" /> : <Plus className="h-3.5 w-3.5" />}
   446→   446→            {inList ? "Saved" : "Save"}
   447→   447→          </button>
   448→   448→          {type === "tv" && (
