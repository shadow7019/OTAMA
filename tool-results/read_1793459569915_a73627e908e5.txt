    37→interface PlayerModalProps {

    43→/**
    44→ * A torrent "server" = one ranked source returned by /api/torrent/search
    45→ * (ThePirateBay, Torrentio/1337x, YTS, EZTV …). We download it on our own
    46→ * backend and stream it to the native <video> element — no third-party
    47→ * iframe, therefore no pop-ups, no tab redirects, no injected ads.
    48→ */
    49→interface TorrentServer {
    50→  id: string;
    51→  releaseName: string;
    52→  seeders: number;
    53→  peers: number;
    54→  sizeBytes: number;
    55→  provider: string;
    56→  quality: string;
    57→  fileIdx: number | null;
    58→  /** verified = exact title/episode match · pack = season pack (right file auto-picked) · weak = title-only match */
    59→  match: "verified" | "pack" | "weak";
    60→  score: number;
    61→}
    62→
    63→interface TorrentStatus {
    64→  found: boolean;
    65→  ready?: boolean;
    66→  progress?: number;
    67→  peers?: number;
    68→  speed?: number;
    69→}
    70→
    71→interface ArchiveMatchClient {
    72→  identifier: string;
    73→  title: string;
    74→  year: number | null;
    75→  creator: string;
    76→  description: string;
    77→  license: string;
    78→  fileUrl: string;
    79→  fileSize: number;
    80→  kind: "movie" | "tv";
    81→}

    85→const NO_PEER_SWITCH_MS = 18000;
    86→/** Switch when peers exist but data never arrives. */
    87→const STALL_SWITCH_MS = 40000;
    88→
    89→const fmtSize = (b: number) =>
    90→  b >= 1e9 ? `${(b / 1e9).toFixed(1)} GB` : b >= 1e6 ? `${Math.round(b / 1e6)} MB` : b > 0 ? `${Math.round(b / 1e3)} KB` : "—";
    91→const fmtSpeed = (bps: number) =>
    92→  bps >= 1e6 ? `${(bps / 1e6).toFixed(1)} MB/s` : `${Math.round(bps / 1e3)} KB/s`;
    93→const fmtSeeders = (n: number) => (n >= 1000 ? `${(n / 1000).toFixed(1)}k` : String(n));

    95→export function PlayerModal({ target, actions, onClose }: PlayerModalProps) {
    96→  // remounted per target via key — initial state is safe to derive from props
    97→  const type = target ? getMediaType(target.item) : "movie";
    98→  const itemId = target?.item.id ?? null;
    99→
   100→  /* ---------------- servers (torrent sources) ---------------- */
   101→  const [servers, setServers] = useState<TorrentServer[]>([]);
   102→  const [serversState, setServersState] = useState<"loading" | "ready" | "error">("loading");
   103→  const [searchNonce, setSearchNonce] = useState(0); // bump = manual re-search
   104→  const [activeIdx, setActiveIdx] = useState<number | null>(null);
   105→  const [manualIdx, setManualIdx] = useState<number | "archive" | "trailer" | null>(null);
   106→  const [tried, setTried] = useState<number[]>([]);
   107→  const [exhausted, setExhausted] = useState(false);
   108→  const [videoState, setVideoState] = useState<"loading" | "playing" | "error">("loading");
   109→  const [needsTap, setNeedsTap] = useState(false);
   110→  const [status, setStatus] = useState<TorrentStatus | null>(null);
   111→  const [videoNonce, setVideoNonce] = useState(0);
   112→  const [banner, setBanner] = useState<string | null>(null);
   113→  /** automatic re-searches done for the CURRENT episode (reset on ep change) */
   114→  const [autoRetries, setAutoRetries] = useState(0);
   115→
   116→  /* ---------------- archive.org direct fallback (movies) ---------------- */
   117→  const [archive, setArchive] = useState<ArchiveMatchClient | null>(null);
   118→  const [archiveState, setArchiveState] = useState<"loading" | "found" | "none">("loading");
   119→
   120→  /* ---------------- TV navigation ---------------- */
   121→  const [season, setSeason] = useState(() => target?.season ?? 1);
   122→  const [episode, setEpisode] = useState(() => target?.episode ?? 1);
   123→  const [episodes, setEpisodes] = useState<Episode[]>([]);
   124→  const [epLoading, setEpLoading] = useState(() => type === "tv");
   125→  const [details, setDetails] = useState<MediaDetails | null>(null);
   126→  const [showEpisodes, setShowEpisodes] = useState(false);
   127→
   128→  const mounted = useMounted();
   129→  const addContinue = useMediaStore((s) => s.addContinue);
   130→  const toggleWatchlist = useMediaStore((s) => s.toggleWatchlist);
   131→  const watchlist = useMediaStore((s) => s.watchlist);
   132→
   133→  // refs so callbacks/effects always read the latest values
   134→  const serversRef = useRef(servers);
   135→  serversRef.current = servers;
   136→  const activeIdxRef = useRef(activeIdx);
   137→  activeIdxRef.current = activeIdx;
   138→  const triedRef = useRef(tried);
   139→  triedRef.current = tried;
   140→  const videoStateRef = useRef(videoState);
   141→  videoStateRef.current = videoState;
   142→  const streamStartRef = useRef(0);
   143→  const currentIdRef = useRef<string | null>(null);
   144→  const videoRef = useRef<HTMLVideoElement | null>(null);
   145→
   146→  // When the browser blocks autoplay we surface a tap-to-play overlay instead
   147→  // of a stuck black frame (video data is already buffered at that point).
   148→  const handleCanPlay = (e: React.SyntheticEvent<HTMLVideoElement>) => {
   149→    setVideoState((s) => (s === "loading" ? "playing" : s));
   150→    const v = e.currentTarget;
   151→    if (v.paused) {
   152→      v.play().then(() => setNeedsTap(false)).catch(() => setNeedsTap(true));
   153→    }
   154→  };
   155→
   156→  const activeServer = activeIdx !== null ? servers[activeIdx] ?? null : null;
   157→  const isArchive = manualIdx === "archive" && archiveState === "found" && !!archive;
   158→  const showTrailer = manualIdx === "trailer";
   159→
   160→  const trailer: Video | null = useMemo(() => (details ? trailerOf(details) : null), [details]);
   161→
   162→  const stopId = useCallback((id: string | null) => {
   163→    if (!id) return;
   164→    fetch("/api/torrent/stop", {
   165→      method: "POST",
   166→      headers: { "Content-Type": "application/json" },
   167→      body: JSON.stringify({ id }),
   168→      keepalive: true,
   169→    }).catch(() => {});
   170→  }, []);
   171→
   172→  const serverLabel = useCallback((s: TorrentServer) => `${s.quality} · ${s.provider}`, []);
   173→
   174→  /* ---------------- failover ---------------- */
   175→  const failover = useCallback(
   176→    (reason: string) => {
   177→      const cur = activeIdxRef.current;
   178→      const list = serversRef.current;
   179→      const next = list.findIndex((_, i) => i !== cur && !triedRef.current.includes(i));
   180→      const triedNext = new Set(triedRef.current);
   181→      if (cur !== null) triedNext.add(cur);
   182→      triedRef.current = [...triedNext];
   183→      setTried([...triedNext]);
   184→      const failed = cur !== null ? serversRef.current[cur] : null;
   185→      if (failed) stopId(failed.id);
   186→      if (next >= 0) {
   187→        setActiveIdx(next);
   188→        setVideoState("loading");
   189→        setStatus(null);
   190→        streamStartRef.current = Date.now();
   191→        setBanner(`${reason} — switching to server ${next + 1} (${serverLabel(list[next])})`);
   192→      } else {
   193→        setExhausted(true);
   194→        setBanner(reason);
   195→      }
   196→    },
   197→    [stopId, serverLabel]
   198→  );
   199→
   200→  /* ---------------- load servers for this title/episode ---------------- */
   201→  useEffect(() => {
   202→    if (!target) return;
   203→    let cancelled = false;
   204→    setServers([]);
   205→    setServersState("loading");
   206→    setTried([]);
   207→    triedRef.current = [];
   208→    setActiveIdx(null);
   209→    setManualIdx(null);
   210→    setExhausted(false);
   211→    setVideoState("loading");
   212→    setStatus(null);
   213→    setBanner(null);
   214→
   215→    const params = new URLSearchParams({
   216→      type,
   217→      id: String(target.item.id),
   218→      title: getTitle(target.item),
   219→      year: getYear(target.item),
   220→    });
   221→    if (type === "tv") {
   222→      params.set("season", String(season));
   223→      params.set("episode", String(episode));
   224→    }
   225→    fetch(`/api/torrent/search?${params.toString()}`)
   226→      .then((r) => r.json())
   227→      .then((j: { results?: TorrentServer[] }) => {
   228→        if (cancelled) return;
   229→        setServers(Array.isArray(j.results) ? j.results : []);
   230→        setServersState("ready");
   231→      })
   232→      .catch(() => {
   233→        if (!cancelled) setServersState("error");
   234→      });
   235→
   236→    // Otama Direct: movies → public-domain copy; TV episodes → episode lookup
   237→    // (Indian daily shows surface on archive.org hours after broadcast, long
   238→    // before the public trackers list them)
   239→    setArchiveState("loading");
   240→    const archiveUrl =
   241→      type === "movie"
   242→        ? `/api/archive?title=${encodeURIComponent(getTitle(target.item))}&year=${encodeURIComponent(getYear(target.item))}`
   243→        : `/api/archive?type=tv&id=${target.item.id}&season=${season}&episode=${episode}&title=${encodeURIComponent(getTitle(target.item))}`;
   244→    fetch(archiveUrl)
   245→      .then((r) => r.json())
   246→      .then((j: { match: ArchiveMatchClient | null }) => {
   247→        if (cancelled) return;
   248→        setArchive(j.match);
   249→        setArchiveState(j.match ? "found" : "none");
   250→      })
   251→      .catch(() => {
   252→        if (!cancelled) setArchiveState("none");
   253→      });
   254→    return () => {
   255→      cancelled = true;
   256→    };
   257→  }, [target, type, season, episode, searchNonce]);
   258→  // automatic retries are per-episode: a manual "Search again" must not reset them
   259→  useEffect(() => {
   260→    setAutoRetries(0);
   261→  }, [target, type, season, episode]);
   262→
   263→  /* ---------------- auto-pick the best untried server ---------------- */
   264→  useEffect(() => {
   265→    if (serversState !== "ready") return;
   266→    if (manualIdx !== null || activeIdx !== null || exhausted) return;
   267→    const idx = servers.findIndex((_, i) => !tried.includes(i));
   268→    if (idx >= 0) {
   269→      setActiveIdx(idx);
   270→      setVideoState("loading");
   271→      streamStartRef.current = Date.now();
   272→    } else {
   273→      setExhausted(true);
   274→    }
   275→  }, [serversState, servers, tried, activeIdx, manualIdx, exhausted]);
   276→
   277→  /* ---------------- Otama Direct for episodes with no torrent sources ----------------
   278→   * Daily-show episodes (Bigg Boss Hindi…) are usually absent from public
   279→   * trackers but uploaded to archive.org within hours of broadcast. When the
   280→   * torrent scan comes up empty and a direct episode stream exists, play it. */
   281→  useEffect(() => {
   282→    if (type !== "tv" || serversState !== "ready") return;
   283→    if (servers.length > 0 || manualIdx !== null || exhausted) return;
   284→    if (archiveState !== "found" || !archive) return;
   285→    setManualIdx("archive");
   286→    setVideoState("loading");
   287→  }, [type, serversState, servers.length, manualIdx, exhausted, archiveState, archive]);
   288→
   289→  /* ---------------- stream URL ---------------- */
   290→  const streamUrl = useMemo(() => {
   291→    if (!activeServer || !target) return null;
   292→    const file = activeServer.fileIdx != null ? `&file=${activeServer.fileIdx}` : "";
   293→    // season/episode steer file selection inside season packs and year
   294→    // steers multi-year movie collections — so the exact requested content
   295→    // plays instead of the pack's biggest file
   296→    const hints =
   297→      type === "tv"
   298→        ? `&season=${season}&episode=${episode}`
   299→        : `&year=${getYear(target.item)}`;
   300→    return `/api/torrent/stream?id=${activeServer.id}&dn=${encodeURIComponent(activeServer.releaseName)}${file}${hints}`;
   301→  }, [activeServer, type, season, episode, target?.item]);
   302→
   303→  useEffect(() => {
   304→    streamStartRef.current = Date.now();
   305→    setStatus(null);
   306→    setNeedsTap(false);
   307→  }, [streamUrl, videoNonce]);
   308→
   309→  /* ---------------- live status + watchdog failover ---------------- */
   310→  useEffect(() => {
   311→    if (!streamUrl || !activeServer) {
   312→      setStatus(null);
   313→      return;
   314→    }
   315→    let cancelled = false;
   316→    const tick = async () => {
   317→      if (cancelled) return;
   318→      try {
   319→        const r = await fetch(`/api/torrent/status?id=${activeServer.id}`);
   320→        const j: TorrentStatus = await r.json();
   321→        if (cancelled) return;
   322→        setStatus(j);
   323→        if (j.found === false) {
   324→          failover("This server dropped the stream");
   325→          return;
   326→        }
   327→        if (videoStateRef.current === "playing") return;
   328→        const elapsed = Date.now() - streamStartRef.current;
   329→        if (elapsed > NO_PEER_SWITCH_MS && (j.peers ?? 0) === 0 && (j.progress ?? 0) === 0) {
   330→          failover("No peers on this server");
   331→        } else if (elapsed > STALL_SWITCH_MS && videoStateRef.current === "loading") {
   332→          // data may be flowing but the wrong pieces — this server can't start in time
   333→          failover("This server is too slow to start");
   334→        }
   335→      } catch {
   336→        /* transient — next tick retries */
   337→      }
   338→    };
   339→    tick();
   340→    const iv = setInterval(tick, 2000);
   341→    return () => {
   342→      cancelled = true;
   343→      clearInterval(iv);
   344→    };
   345→  }, [streamUrl, activeServer, failover]);
   346→
   347→  /* ---------------- stop torrents on switch / unmount ---------------- */
   348→  useEffect(() => {
   349→    const id = activeServer?.id ?? null;
   350→    if (currentIdRef.current && currentIdRef.current !== id) stopId(currentIdRef.current);
   351→    currentIdRef.current = id;
   352→  }, [activeServer?.id, stopId]);
   353→  useEffect(() => {
   354→    return () => {
   355→      if (currentIdRef.current) stopId(currentIdRef.current);
   356→    };
   357→  }, [stopId]);
   358→
   359→  /* ---------------- record progress for Continue Watching ---------------- */
   360→  useEffect(() => {
   361→    if (!target || !mounted) return;
   362→    const t = getMediaType(target.item);
   363→    addContinue(target.item, t === "tv" ? season : undefined, t === "tv" ? episode : undefined);
   364→  }, [target, mounted, addContinue, season, episode]);
   365→
   366→  /* ---------------- details (season list + trailer fallback) ---------------- */
   367→  useEffect(() => {
   368→    if (!target) return;
   369→    let cancelled = false;
   370→    tmdb<MediaDetails>(`${type}/${target.item.id}`)
   371→      .then((d) => !cancelled && setDetails(d))
   372→      .catch(() => {});
   373→    return () => {
   374→      cancelled = true;
   375→    };
   376→  }, [target, type]);
   377→
   378→  useEffect(() => {
   379→    if (!target || type !== "tv") return;
   380→    let cancelled = false;
   381→    tmdb<{ episodes: Episode[] }>(`tv/${target.item.id}/season/${season}`)
   382→      .then((res) => {
   383→        if (!cancelled) {
   384→          setEpisodes(res.episodes ?? []);
   385→          setEpLoading(false);
   386→        }
   387→      })
   388→      .catch(() => {
   389→        if (!cancelled) {
   390→          setEpisodes([]);
   391→          setEpLoading(false);
   392→        }
   393→      });
   394→    return () => {
   395→      cancelled = true;
   396→    };
   397→  }, [target, type, season]);
   398→
   399→  /* ---------------- auto-refresh episodes while open ----------------
   400→   * Daily shows (Bigg Boss…) publish a new episode every day: re-pull the
   401→   * season when the tab regains focus and every few minutes, so the latest
   402→   * episode appears on its own — no reload needed. */
   403→  const epRefreshRef = useRef(0);
   404→  useEffect(() => {
   405→    if (!target || type !== "tv") return;
   406→    const refresh = () => {
   407→      if (Date.now() - epRefreshRef.current < 90_000) return; // throttle
   408→      epRefreshRef.current = Date.now();
   409→      tmdb<{ episodes: Episode[] }>(`tv/${target.item.id}/season/${season}`, { _fresh: 1 })
   410→        .then((res) => setEpisodes(res.episodes ?? []))
   411→        .catch(() => {});
   412→    };
   413→    const onVis = () => {
   414→      if (document.visibilityState === "visible") refresh();
   415→    };
   416→    const iv = setInterval(onVis, 3 * 60_000);
   417→    document.addEventListener("visibilitychange", onVis);
   418→    return () => {
   419→      clearInterval(iv);
   420→      document.removeEventListener("visibilitychange", onVis);
   421→    };
   422→  }, [target, type, season]);
   423→
   424→  /* ---------------- banner auto-dismiss ---------------- */
   425→  useEffect(() => {
   426→    if (!banner) return;
   427→    const t = setTimeout(() => setBanner(null), 7000);
   428→    return () => clearTimeout(t);
   429→  }, [banner]);
   430→
   431→  /* ---------------- scroll lock + escape ---------------- */
   432→  useEffect(() => {
   433→    if (!target) return;
   434→    const original = document.body.style.overflow;
   435→    document.body.style.overflow = "hidden";
   436→    const onKey = (ev: KeyboardEvent) => {
   437→      if (ev.key === "Escape") onClose();
   438→    };
   439→    window.addEventListener("keydown", onKey);
   440→    return () => {
   441→      document.body.style.overflow = original;
   442→      window.removeEventListener("keydown", onKey);
   443→    };
   444→  }, [target, onClose]);
   445→
   446→  const seasonsList = (details?.seasons ?? []).filter(
   447→    (s) => s.season_number > 0 && s.episode_count > 0
   448→  );
   449→  const maxEpisode =
   450→    episodes.length ||
   451→    details?.seasons?.find((s) => s.season_number === season)?.episode_count ||
   452→    1;
   453→
   454→  /* -------- episode freshness (daily shows like Bigg Boss) -------- */
   455→  const now = new Date();
   456→  const todayYmd = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, "0")}-${String(
   457→    now.getDate()
   458→  ).padStart(2, "0")}`;
   459→  const epAired = (ep: Episode) => !!ep.air_date && ep.air_date <= todayYmd;
   460→  const latestAired = useMemo(() => {
   461→    for (let i = episodes.length - 1; i >= 0; i--) {
   462→      if (epAired(episodes[i])) return episodes[i];
   463→    }
   464→    return null;
   465→  }, [episodes]);
   466→  const currentEpisode = episodes.find((ep) => ep.episode_number === episode);
   467→
   468→  /* ---------------- auto-research when an episode has no sources yet ----------------
   469→   * Brand-new daily-show episodes (Bigg Boss…) are often indexed only hours
   470→   * after broadcast. When the first scan comes up empty and the episode has
   471→   * already aired, re-scan automatically every 90s (max 5 tries) so the
   472→   * torrent appears the moment indexers list it — no manual refresh needed. */
   473→  const noSourcesYet = serversState === "ready" && servers.length === 0;
   474→  const episodeAired =
   475→    type !== "tv" || !currentEpisode?.air_date || currentEpisode.air_date <= todayYmd;
   476→  useEffect(() => {
   477→    if (!noSourcesYet || !episodeAired || autoRetries >= 5) return;
   478→    // a direct episode stream that is already playing must not be interrupted
   479→    if (manualIdx === "archive" && archiveState === "found") return;
   480→    const t = setTimeout(() => {
   481→      setAutoRetries((n) => n + 1);
   482→      setSearchNonce((n) => n + 1);
   483→    }, 90_000);
   484→    return () => clearTimeout(t);
   485→  }, [noSourcesYet, episodeAired, autoRetries, searchNonce, manualIdx, archiveState]);
   486→
   487→  if (!target) return null;
   488→
   489→  const item = target.item;
   490→  const title = getTitle(item);
   491→
   492→  const goEpisode = (s: number, e: number) => {
   493→    setSeason(s);
   494→    setEpisode(e);
   495→    setShowEpisodes(false);
   496→  };
   497→  const nextEpisode = () => {
   498→    // never step into episodes that haven't aired yet (TMDB publishes
   499→    // placeholders days ahead for daily shows)
   500→    const cap = latestAired?.episode_number ?? maxEpisode;
   501→    if (episode < cap) goEpisode(season, episode + 1);
   502→    else if (seasonsList.some((s) => s.season_number === season + 1)) goEpisode(season + 1, 1);
   503→  };
   504→  const prevEpisode = () => {
   505→    if (episode > 1) goEpisode(season, episode - 1);
   506→    else if (season > 1) goEpisode(season - 1, 1);
   507→  };
   508→
   509→  const inList = mounted
   510→    ? watchlist.some((w) => w.id === item.id && w.media_type === type)
   511→    : false;
   512→
   513→  /* ---------------- user actions ---------------- */
   514→  const selectTorrent = (idx: number) => {
   515→    if (idx === activeIdx) {
   516→      // re-selecting the active server = manual retry
   517→      setVideoState("loading");
   518→      setVideoNonce((n) => n + 1);
   519→      return;
   520→    }
   521→    setManualIdx(idx);
   522→    setExhausted(false);
   523→    setVideoState("loading");
   524→    setActiveIdx(idx);
   525→    streamStartRef.current = Date.now();
   526→  };
   527→
   528→  const selectAuto = () => {
   529→    setManualIdx(null);
   530→    setExhausted(false);
   531→    setTried([]);
   532→    triedRef.current = [];
   533→    setActiveIdx(null); // auto-pick effect chooses the best server again
   534→  };
   535→
   536→  const selectArchive = () => {
   537→    if (archiveState !== "found" || !archive) return;
   538→    if (manualIdx === "archive") {
   539→      setVideoNonce((n) => n + 1);
