    42→    42→export async function cached<T>(key: string, ttlMs: number, fn: () => Promise<T>): Promise<T> {
    43→    43→  const hit = cache.get(key)
    44→    44→  if (hit && Date.now() - hit.ts < ttlMs) return hit.value as T
    45→    45→  const value = await fn()
    46→    46→  cache.set(key, { ts: Date.now(), value })
    47→    47→  return value
    48→    48→}

    52→    52→async function fetchText(url: string, timeoutMs = 12_000): Promise<string> {
    53→    53→  const ctrl = new AbortController()
    54→    54→  const t = setTimeout(() => ctrl.abort(), timeoutMs)
    55→    55→  try {
    56→    56→    const res = await fetch(url, {
    57→    57→      signal: ctrl.signal,
    58→    58→      cache: 'no-store',
    59→    59→      headers: { 'User-Agent': UA, Accept: '*/*' },
    60→    60→    })
    61→    61→    if (!res.ok) throw new Error(`HTTP ${res.status} for ${url}`)
    62→    62→    return await res.text()
    63→    63→  } finally {
    64→    64→    clearTimeout(t)
    65→    65→  }
    66→    66→}

    68→    68→async function fetchJson<T>(url: string, timeoutMs = 12_000): Promise<T> {
    69→    69→  const text = await fetchText(url, timeoutMs)
    70→    70→  if (text.trimStart().startsWith('<')) throw new Error(`Got HTML instead of JSON from ${url}`)
    71→    71→  return JSON.parse(text) as T
    72→    72→}

    76→    76→/**
    77→    77→ * Some trackers (apibay = ThePirateBay API, EZTV) sit behind Cloudflare and
    78→    78→ * fingerprint TLS: Node's fetch/https stacks get 403-challenged while curl
    79→    79→ * passes. Use curl as the primary transport for those hosts with a fetch
    80→    80→ * fallback for environments without curl.
    81→    81→ */
    82→    82→export async function cfGetText(url: string, timeoutMs = 12_000): Promise<string> {
    83→    83→  const maxTime = Math.ceil(timeoutMs / 1000)
    84→    84→  try {
    85→    85→    const { execFile } = await import('node:child_process')
    86→    86→    const curlOnce = () =>
    87→    87→      new Promise<string>((resolve, reject) => {
    88→    88→        execFile(
    89→    89→          'curl',
    90→    90→          ['-s', '-L', '--compressed', '--max-time', String(maxTime), '-A', UA, '-H', 'Accept: */*', url],
    91→    91→          { timeout: (maxTime + 2) * 1000, maxBuffer: 8 * 1024 * 1024 },
    92→    92→          (err, stdout) => (err ? reject(err) : resolve(stdout)),
    93→    93→        )
    94→    94→      })
    95→    95→    let out = await curlOnce()
    96→    96→    // apibay & friends occasionally answer with a transient EMPTY body while
    97→    97→    // curl itself succeeds — one quick retry beats falling back to the
    98→    98→    // TLS-fingerprinted node fetch that Cloudflare 403-challenges.
    99→    99→    if (!out.trim()) out = await curlOnce().catch(() => '')
   100→   100→    if (out) return out
   101→   101→  } catch { /* curl missing or failed -> fall back to fetch */ }
   102→   102→  return fetchText(url, timeoutMs)
   103→   103→}

   108→   108→  return JSON.parse(text) as T
   109→   109→}

   204→   204→export function humanSize(bytes?: number | string): string {
   205→   205→  const n = typeof bytes === 'string' ? parseInt(bytes, 10) : bytes
   206→   206→  if (!n || isNaN(n)) return ''
   207→   207→  const units = ['B', 'KB', 'MB', 'GB', 'TB']
   208→   208→  let i = 0
   209→   209→  let v = n
   210→   210→  while (v >= 1024 && i < units.length - 1) {
   211→   211→    v /= 1024
   212→   212→    i++
   213→   213→  }
   214→   214→  return `${v.toFixed(v >= 100 || i === 0 ? 0 : 1)} ${units[i]}`
   215→   215→}

   217→   217→const QUALITY_RE = /\b(2160p|1440p|1080p|1080i|720p|576p|480p|4k|uhd|web[ .-]?dl|web[ .-]?rip|brrip|bluray|bdrip|hdrip|dvdrip|hdtv|cam|ts)\b/i

   227→   227→/**
   228→   228→ * Codec detection from a release name. HEVC/x265 releases typically CANNOT be
   229→   229→ * decoded by browsers (Chrome only plays H.265 with hardware support), which
   230→   230→ * is the #1 cause of the "stuck on buffering" report — so OTAMA surfaces and
   231→   231→ * deprioritises them.
   232→   232→ */
   233→   233→export function detectCodecFromName(name: string): 'h264' | 'hevc' | undefined {
   234→   234→  const s = (name || '').toLowerCase()
   235→   235→  if (/\b(x\s?265|h\.?265|hevc)\b/.test(s)) return 'hevc'
   236→   236→  if (/\b(x\s?264|h\.?264|avc)\b/.test(s)) return 'h264'
   237→   237→  return undefined
   238→   238→}
   239→   239→
   240→   240→/**
   241→   241→ * Browser playability of a release name — 0 plays everywhere, higher ranks
   242→   242→ * may not play: 1 MKV/MOV (Chromium-only), 2 AVI/TS (<video> can't demux),
   243→   243→ * 3 HEVC/x265 (browsers cannot decode without hardware support).
   244→   244→ */
   245→   245→export function playabilityRank(name: string, codec?: string): 0 | 1 | 2 | 3 {
   246→   246→  const c = (codec as 'h264' | 'hevc' | undefined) || detectCodecFromName(name)
   247→   247→  if (c === 'hevc') return 3
   248→   248→  const ext = (name.match(/\.(mp4|m4v|mkv|mov|avi|ts|webm)\b/i) || [])[1]?.toLowerCase()
   249→   249→  if (ext === 'avi' || ext === 'ts') return 2 // container not supported by <video>
   250→   250→  if (ext === 'mkv' || ext === 'mov') return 1 // plays in Chromium/Chrome, not Firefox/Safari
   251→   251→  return 0
   252→   252→}
   253→   253→
   254→   254→/**
   255→   255→ * Sort torrent options so that browser-playable releases come first, then by
   256→   256→ * seed count. Keeps the popular x265-HEVC packs from hijacking the top slot.
   257→   257→ */
   258→   258→export function sortTorrentsPlayableFirst(list: TorrentOption[]): TorrentOption[] {
   259→   259→  return [...list].sort((a, b) => {
   260→   260→    const pa = playabilityRank(a.title, a.codec)
   261→   261→    const pb = playabilityRank(b.title, b.codec)
   262→   262→    if (pa !== pb) return pa - pb
   263→   263→    return (b.seeds || 0) - (a.seeds || 0)
   264→   264→  })
   265→   265→}

   267→   267→export function normalizeApibayRow(row: ApibayRow): TpbItem | null {
   268→   268→  // apibay returns a synthetic "No results returned" row (id 0, all-zero hash)

   272→   272→  const cat = row.category || '0'

   274→   274→    id: row.id || row.info_hash,

   299→   299→export const TPB_BROWSE_CATS = ['201', '205', '207', '208', '209']

   301→   301→/**
   302→   302→ * Browse a Pirate Bay category without a query (apibay `q.php?q=category:<id>`).
   303→   303→ * Returns the newest uploads in that category, ordered by seed count so the
   304→   304→ * browse page always opens with live, well-seeded torrents. Retries transient
   305→   305→ * empty/403 answers; throws only after all attempts fail (nothing is cached

   308→   308→export async function apibayBrowse(cat: string): Promise<TpbItem[]> {
   309→   309→  const id = TPB_BROWSE_CATS.includes(cat) ? cat : '201'
   310→   310→  return cached(`apibay-browse:${id}`, 3 * 60_000, async () => {
   311→   311→    const url = `${APIBAY}/q.php?q=${encodeURIComponent(`category:${id}`)}&cat=`
   312→   312→    let rows: TpbItem[] = []
   313→   313→    for (let attempt = 0; attempt < 3 && rows.length === 0; attempt++) {
   314→   314→      if (attempt > 0) await new Promise((r) => setTimeout(r, 700 * attempt))
   315→   315→      try {
   316→   316→        const raw = await cfGetJson<ApibayRow[]>(url, 12_000)
   317→   317→        if (Array.isArray(raw)) rows = raw.map(normalizeApibayRow).filter((r): r is TpbItem => !!r)
   318→   318→      } catch { /* retry */ }
   319→   319→    }
   320→   320→    if (rows.length === 0) throw new ProviderError('ThePirateBay did not answer (Cloudflare) — try again shortly')
   321→   321→    return [...rows].sort((a, b) => b.seeds - a.seeds)
   322→   322→  })
   323→   323→}

   342→   342→/**
   343→   343→ * Race a provider against a soft deadline — best-effort sources (TGx mirror
   344→   344→ * walks can take a minute when every mirror is CF-blocked) must not stall the
   345→   345→ * whole detail panel that is waiting on Promise.all.
   346→   346→ */
   347→   347→function withDeadline<T>(p: Promise<T>, ms: number, fallback: T): Promise<T> {
   348→   348→  return Promise.race([
   349→   349→    p.catch(() => fallback),
   350→   350→    new Promise<T>((resolve) => setTimeout(() => resolve(fallback), ms)),
   351→   351→  ])
   352→   352→}

   387→   387→    })()
   388→   388→
   389→   389→    const q = title ? `${title}${year ? ` ${year}` : ''}` : ''
   390→   390→    const [torrentio, tpb, yts, leetx, solid, rarbg, lime, td, tgx] = await Promise.all([

   393→   393→        : Promise.resolve([] as TorrentOption[]),
   394→   394→      tpbTask,
   395→   395→      ytsMovieTorrents(imdbId, title, year).catch(() => [] as TorrentOption[]),
   396→   396→      q

   404→   404→        : Promise.resolve([] as TorrentOption[]),
   405→   405→      q
   406→   406→        ? limeSearch(q, { category: 'movies' }).catch(() => [] as TorrentOption[])
   407→   407→        : Promise.resolve([] as TorrentOption[]),
   408→   408→      q
   409→   409→        ? torrentDownloadsSearch(q, { resolve: 8 }).catch(() => [] as TorrentOption[])
   410→   410→        : Promise.resolve([] as TorrentOption[]),
   411→   411→      q
   412→   412→        ? withDeadline(tgxSearch(q, { resolve: 6 }).catch(() => [] as TorrentOption[]), 16_000, [] as TorrentOption[])
   413→   413→        : Promise.resolve([] as TorrentOption[]),
   414→   414→    ])
   415→   415→
   416→   416→    const seen = new Set<string>()
   417→   417→    const merged: TorrentOption[] = []
   418→   418→    for (const t of [...torrentio, ...yts, ...tpb, ...rarbg, ...lime, ...leetx, ...solid, ...td, ...tgx]) {
   419→   419→      const key = (t.source || t.hash || '').toLowerCase()
   420→   420→      if (!key || seen.has(key)) continue
   421→   421→      seen.add(key)
   422→   422→      merged.push(t)
   423→   423→    }
   424→   424→    // Browser-playable (H.264/x264, MP4/MKV) releases first, seeds second.
   425→   425→    return sortTorrentsPlayableFirst(merged).slice(0, 40)
   426→   426→  })
   427→   427→}

   433→   433→posterCheckCache.__otamaPosterCache = posterCache

   435→   435→/**
   436→   436→ * Does a poster URL actually resolve to an image? Cinemeta's metahub serves a
   437→   437→ * small HTML error page (content-type text/html) for titles without artwork,
   438→   438→ * which used to render as an ugly grey "initials" placeholder card.
   439→   439→ */
   440→   440→export async function posterWorks(url: string): Promise<boolean> {
   441→   441→  if (!url) return false
   442→   442→  if (!/^https?:\/\//.test(url)) return true
   443→   443→  const hit = posterCache.get(url)
   444→   444→  if (hit !== undefined) return hit
   445→   445→  let ok = true
   446→   446→  try {
   447→   447→    const res = await fetch(url, {
   448→   448→      method: 'HEAD',
   449→   449→      signal: AbortSignal.timeout(6_000),
   450→   450→      headers: { 'User-Agent': UA },
   451→   451→    })
   452→   452→    const ct = (res.headers.get('content-type') || '').toLowerCase()
   453→   453→    if (!res.ok || !ct.startsWith('image/')) ok = false
   454→   454→  } catch {
   455→   455→    // network hiccup — give the poster the benefit of the doubt
   456→   456→    ok = true
   457→   457→  }
   458→   458→  if (posterCache.size < 5000) posterCache.set(url, ok)
   459→   459→  return ok
   460→   460→}

   513→   513→export function decodeEntities(s: string): string {
   514→   514→  return s
   515→   515→    .replace(/&#(\d+);/g, (_, d) => String.fromCharCode(parseInt(d, 10)))
   516→   516→    .replace(/&#x([0-9a-fA-F]+);/g, (_, h) => String.fromCharCode(parseInt(h, 16)))
   517→   517→    .replace(/&([a-zA-Z]+);/g, (_, n: string) => NAMED_ENTITIES[n] || `&${n};`)
   518→   518→}

   535→   535→function cdata(s: string): string {
   536→   536→  const m = /<!\[CDATA\[([\s\S]*?)\]\]>/.exec(s)
   537→   537→  return m ? m[1] : s

   551→   551→export function parseNyaaRss(xml: string): NyaaItem[] {
   552→   552→  const items: NyaaItem[] = []
   553→   553→  const itemRe = /<item>([\s\S]*?)<\/item>/g
   554→   554→  let m: RegExpExecArray | null
   555→   555→  const tag = (block: string, name: string): string => {
   556→   556→    const re = new RegExp(`<${name}[^>]*>([\\s\\S]*?)</${name}>`, 'i')
   557→   557→    const t = re.exec(block)
   558→   558→    return t ? decodeEntities(cdata(t[1]).trim()) : ''
   559→   559→  }
   560→   560→  while ((m = itemRe.exec(xml))) {
   561→   561→    const block = m[1]
   562→   562→    const hash = tag(block, 'nyaa:infoHash').toLowerCase()
   563→   563→    const title = tag(block, 'title')
   564→   564→    if (!hash || !title) continue
   565→   565→    items.push({
   566→   566→      title,
   567→   567→      hash,
   568→   568→      seeds: parseInt(tag(block, 'nyaa:seeders'), 10) || 0,
   569→   569→      leechers: parseInt(tag(block, 'nyaa:leechers'), 10) || 0,
   570→   570→      size: tag(block, 'nyaa:size'),
   571→   571→      date: tag(block, 'pubDate'),
   572→   572→      page: tag(block, 'guid'),
   573→   573→      info: tag(block, 'description'),
   574→   574→    })
   575→   575→  }
   576→   576→  return items
   577→   577→}

   710→   710→  return { item, seasons }

   713→   713→/** Episode torrents: Torrentio (multi-site, file-exact) + EZTV + TPB + 1337x + SolidTorrents. */
   714→   714→export async function findEpisodeTorrents(
   715→   715→  imdbId?: string,
   716→   716→  title?: string,
   717→   717→  season?: number,
   718→   718→  episode?: number,
   719→   719→): Promise<TorrentOption[]> {
   720→   720→  // Torrentio first — aggregates EZTV/TPB/1337x/TGX/RARBG/KAT and returns the
   721→   721→  // exact video file index for the episode (fast + fewest dead torrents).
   722→   722→  if (imdbId && /^tt\d+$/.test(imdbId)) {
   723→   723→    try {
   724→   724→      const { torrentioStreams } = await import('./torrentio')
   725→   725→      const tor = await cached(
   726→   726→        `torrentio:ep:${imdbId}:${season ?? ''}:${episode ?? ''}`,
   727→   727→        5 * 60_000,
   728→   728→        () => torrentioStreams('series', imdbId, season, episode),
   729→   729→      )
   730→   730→      if (season && episode) {
   731→   731→        const exact = sortTorrentsPlayableFirst(tor.filter((t) => t.season == null || t.season === season)).slice(0, 15)
   732→   732→        if (exact.length) return exact
   733→   733→      } else {
   734→   734→        const sorted = sortTorrentsPlayableFirst(tor).slice(0, 24)
   735→   735→        if (sorted.length) return sorted
   736→   736→      }
   737→   737→    } catch { /* Torrentio unavailable -> EZTV */ }
   738→   738→  }
   739→   739→
   740→   740→  // whole-show EZTV lookup (cached)
   741→   741→  if (imdbId && /^\d+$/.test(imdbId.replace(/^tt/, ''))) {
   742→   742→    try {
   743→   743→      const list = await cached(`eztv:${imdbId}`, 10 * 60_000, () => eztvByImdb(imdbId.replace(/^tt/, '')))
   744→   744→      const sorted = sortTorrentsPlayableFirst(list)
   745→   745→      if (season && episode) {
   746→   746→        const exact = sorted.filter((t) => t.season === season && t.episode === episode)
   747→   747→        if (exact.length) return exact.slice(0, 12)
   748→   748→      } else if (season) {
   749→   749→        const pack = sorted.filter((t) => t.season === season && (t.episode == null || t.episode === 0))
   750→   750→        if (pack.length) return pack.slice(0, 12)
   751→   751→      } else {
   752→   752→        return sorted.slice(0, 24)
   753→   753→      }
   754→   754→    } catch { /* EZTV unavailable -> TPB */ }
   755→   755→  }
   756→   756→  if (!title) return []
   757→   757→  const pad = (n?: number, w = 2) => (n != null ? String(n).padStart(w, '0') : '')
   758→   758→  const queries: string[] = []
   759→   759→  if (season && episode) {
   760→   760→    queries.push(`${title} S${pad(season)}E${pad(episode)}`)
   761→   761→    queries.push(`${title} ${season}x${pad(episode)}`)
   762→   762→  } else if (season) {
   763→   763→    queries.push(`${title} S${pad(season)}`)
   764→   764→  } else {
   765→   765→    queries.push(title)
   766→   766→  }
   767→   767→  for (const q of queries) {
   768→   768→    try {
   769→   769→      const rows = await apibaySearch(q)
   770→   770→      const tvCats = new Set(['205', '208'])
   771→   771→      const opts = rows
   772→   772→        .filter((r) => tvCats.has(r.categoryCode) || /\bS\d{1,2}E\d{1,2}\b/i.test(r.name))
   773→   773→        .map(apibayToTorrentOption)
   774→   774→      const sorted = sortTorrentsPlayableFirst(opts).slice(0, 12)
   775→   775→      if (sorted.length) return sorted
   776→   776→    } catch { /* try next */ }
   777→   777→  }
   778→   778→  // 1337x fallback (works when TPB/EZTV are blocked)
   779→   779→  if (season && episode) {
   780→   780→    try {
   781→   781→      const { leetxSearch } = await import('./leetx')
   782→   782→      const opts = await leetxSearch(`${title} S${pad(season)}E${pad(episode)}`, { category: 'tv', resolve: 8 })
   783→   783→      if (opts.length) return opts.slice(0, 12)
   784→   784→    } catch { /* next fallback */ }
   785→   785→  }
   786→   786→  // SolidTorrents fallback (DHT index, reachable when trackers are blocked)
   787→   787→  if (season && episode) {
   788→   788→    try {
   789→   789→      const { solidSearch } = await import('./solidtorrents')
   790→   790→      const opts = await solidSearch(`${title} S${pad(season)}E${pad(episode)}`, { videoOnly: true })
   791→   791→      const sorted = sortTorrentsPlayableFirst(opts).slice(0, 12)
   792→   792→      if (sorted.length) return sorted
   793→   793→    } catch { /* next fallback */ }
   794→   794→  }
   795→   795→  // RARBG archive fallback (JSON, infohash included)
   796→   796→  try {
   797→   797→    const { rarbgSearch } = await import('./rarbg')
   798→   798→    const opts = sortTorrentsPlayableFirst(await rarbgSearch(`${title} ${season && episode ? `S${pad(season)}E${pad(episode)}` : ''}`.trim()))
   799→   799→      .filter((t) => /\bS\d{1,2}E\d{1,2}\b/i.test(t.title) || !season)
   800→   800→    if (opts.length) return opts.slice(0, 12)
   801→   801→  } catch { /* next fallback */ }
   802→   802→  // LimeTorrents TV fallback (hash embedded in rows)
   803→   803→  if (season && episode) {
   804→   804→    try {
   805→   805→      const { limeSearch } = await import('./limetorrents')
   806→   806→      const opts = await limeSearch(`${title} S${pad(season)}E${pad(episode)}`, { category: 'tv' })
   807→   807→      if (opts.length) return opts.slice(0, 12)
   808→   808→    } catch { /* next fallback */ }
   809→   809→  }
   810→   810→  // TorrentGalaxy fallback (best-effort, mirror chain) — soft deadline so a
   811→   811→  // fully-blocked mirror chain can never stall the episode panel
